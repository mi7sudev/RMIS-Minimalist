    77→    77→function nextStepHint(stage: string): string {
    78→    78→  if (stage === "Rejected")
    79→    79→    return "Your application was not shortlisted. You may apply for other open positions.";
    80→    80→  if (stage === "Shortlisted")

   104→   104→        state === "current" && "motion-safe:animate-pulse"

   112→   112→/** 3-checkpoint tracking timeline: Submitted → Review → Decision (§7.3). */
   113→   113→function JourneyTimeline({ status }: { status: string }) {
   114→   114→  const stage = stageForStatus(status);
   115→   115→  const decided = stage === "Shortlisted" || stage === "Rejected";
   116→   116→
   117→   117→  const steps: { label: string; state: CheckpointState }[] = [
   118→   118→    { label: "Submitted", state: "done" },
   119→   119→    {
   120→   120→      label: "Review",
   121→   121→      state: stage === "Under Review" ? "current" : decided ? "done" : "pending",
   122→   122→    },
   123→   123→    {
   124→   124→      label:
   125→   125→        stage === "Shortlisted" ? "Shortlisted" : stage === "Rejected" ? "Not Shortlisted" : "Awaiting the shortlist decision",
   126→   126→      state: stage === "Shortlisted" ? "done" : stage === "Rejected" ? "failed" : "pending",
   127→   127→    },
   128→   128→  ];
   129→   129→
   130→   130→  return (
   131→   131→    <div className="mt-3 grid grid-cols-3 gap-1">
   132→   132→      {steps.map((s, i) => {
   133→   133→        const reached = s.state !== "pending";
   134→   134→        return (
   135→   135→          <div key={`${s.label}-${i}`} className="flex flex-col items-center gap-1.5 text-center">
   136→   136→            <div className="flex w-full items-center">
   137→   137→              {/* Traversed connectors pick up a subtle ok→warn gradient tint. */}
   138→   138→              <div
   139→   139→                className={cn(
   140→   140→                  "h-px flex-1",
   141→   141→                  i === 0 ? "bg-transparent" : reached ? "bg-gradient-to-r from-[var(--ok)]/60 to-[var(--warn)]/45" : "bg-divider"
   142→   142→                )}
   143→   143→              />
   144→   144→              <CheckpointDot state={s.state} />
   145→   145→              <div
   146→   146→                className={cn(
   147→   147→                  "h-px flex-1",
   148→   148→                  i === steps.length - 1
   149→   149→                    ? "bg-transparent"
   150→   150→                    : steps[i + 1].state !== "pending"
   151→   151→                      ? "bg-gradient-to-r from-[var(--ok)]/60 to-[var(--warn)]/45"
   152→   152→                      : "bg-divider"
   153→   153→                )}
   154→   154→              />
   155→   155→            </div>
   156→   156→            <span className={cn("text-[10px] leading-tight", s.state === "current" ? "font-medium text-ink" : "text-stone")}>
   157→   157→              {s.label}
   158→   158→            </span>
   159→   159→          </div>
   160→   160→        );
   161→   161→      })}
   162→   162→    </div>
   163→   163→  );
   164→   164→}

    62→    62→export async function loadApplicantEligibilities(applicantId: number) {
    63→    63→  const links = await db.applicantEligibilityLink.findMany({
    64→    64→    where: { applicantId },
    65→    65→    orderBy: { applicantEligibilityOrd: "asc" },
    66→    66→  });
    67→    67→  if (!links.length) return [];
    68→    68→  const ids = links.map((l) => l.applicantEligibilityId).filter((x): x is number => x != null);
    69→    69→  const [rows, categoryLinks] = await Promise.all([
    70→    70→    db.applicantEligibility.findMany({ where: { id: { in: ids } } }),
    71→    71→    db.applicantEligibilityCategoryLink.findMany({
    72→    72→      where: { applicantEligibilityId: { in: ids } },
    73→    73→    }),
    74→    74→  ]);
    75→    75→  // Map each applicant_eligibility → its first linked eligibility category name
    76→    76→  const eligibilityIds = Array.from(
    77→    77→    new Set(categoryLinks.map((c) => c.eligibilityId).filter((x): x is number => x != null))
    78→    78→  );
    79→    79→  const categories = eligibilityIds.length
    80→    80→    ? await db.eligibility.findMany({ where: { id: { in: eligibilityIds } } })
    81→    81→    : [];
    82→    82→  const catById = new Map(categories.map((c) => [c.id, c.name]));
    83→    83→  const catLinkByApplElig = new Map<number, number | null>();
    84→    84→  for (const cl of categoryLinks) {
    85→    85→    if (cl.applicantEligibilityId == null) continue;
    86→    86→    if (!catLinkByApplElig.has(cl.applicantEligibilityId)) {
    87→    87→      catLinkByApplElig.set(cl.applicantEligibilityId, cl.eligibilityId);
    88→    88→    }
    89→    89→  }
    90→    90→  const byId = new Map(rows.map((r) => [r.id, r]));
    91→    91→  return links
    92→    92→    .map((l) => {
    93→    93→      if (!l.applicantEligibilityId) return undefined;
    94→    94→      const row = byId.get(l.applicantEligibilityId);
    95→    95→      if (!row) return undefined;
    96→    96→      const catId = catLinkByApplElig.get(row.id) ?? null;
    97→    97→      const eligibilityTitle = catId != null ? catById.get(catId) ?? null : null;
    98→    98→      return { ...row, eligibilityTitle };
    99→    99→    })
   100→   100→    .filter((x): x is NonNullable<typeof x> => x != null);
   101→   101→}
   102→   102→
   103→   103→/**
   104→   104→ * Backfill `eligibilityTitle` for eligibility items whose title is still null.
   105→   105→ *
   106→   106→ * Production stores the title link in `specific_eligibilities_applicant_eligibility_lnk`
   107→   107→ * (not in `applicant_eligibilities_eligibility_category_lnk`, which has 0 rows in
   108→   108→ * production). This resolves the title via the path:
   109→   109→ *
   110→   110→ *   specific_eligibilities_applicant_eligibility_lnk
   111→   111→ *     → specific_eligibilities (name)             ← prefer this
   112→   112→ *     → specific_eligibilities_eligibility_lnk
   113→   113→ *         → eligibilities (name)                  ← fallback
   114→   114→ *
   115→   115→ * Only items whose title is still null are touched, so newly-created entries
   116→   116→ * (which write to the category link) keep working. Patches items in place.
   117→   117→ */
   118→   118→export async function fillEligibilityTitleFromSpecificLinks<
   119→   119→  T extends { id: number; eligibilityTitle: string | null }
   120→   120→>(items: T[]): Promise<void> {
   121→   121→  const aeIds = items.filter((it) => it.eligibilityTitle == null).map((it) => it.id);
   122→   122→  if (!aeIds.length) return;
   123→   123→
   124→   124→  const specificLinks = await db.specificEligibilityApplicantEligibilityLink.findMany({
   125→   125→    where: { applicantEligibilityId: { in: aeIds } },
   126→   126→    orderBy: { specificEligibilityOrd: "asc" },
   127→   127→  });
   128→   128→  if (!specificLinks.length) return;
   129→   129→
   130→   130→  const specificIds = Array.from(
   131→   131→    new Set(
   132→   132→      specificLinks
   133→   133→        .map((l) => l.specificEligibilityId)
   134→   134→        .filter((x): x is number => x != null)
   135→   135→    )
   136→   136→  );
   137→   137→  const [specifics, specificEligLinks] = await Promise.all([
   138→   138→    specificIds.length
   139→   139→      ? db.specificEligibility.findMany({ where: { id: { in: specificIds } } })
   140→   140→      : [],
   141→   141→    specificIds.length
   142→   142→      ? db.specificEligibilityEligibilityLink.findMany({
   143→   143→          where: { specificEligibilityId: { in: specificIds } },
   144→   144→          orderBy: { specificEligibilityOrd: "asc" },
   145→   145→        })
   146→   146→      : [],
   147→   147→  ]);
   148→   148→
   149→   149→  const specNameById = new Map<number, string | null>(
   150→   150→    specifics.map((s) => [s.id, s.name] as [number, string | null])
   151→   151→  );
   152→   152→  // First specificEligibilityId per applicantEligibilityId (lowest ord).
   153→   153→  const firstSpecByAe = new Map<number, number | null>();
   154→   154→  for (const sl of specificLinks) {
   155→   155→    if (sl.applicantEligibilityId == null) continue;
   156→   156→    if (!firstSpecByAe.has(sl.applicantEligibilityId)) {
   157→   157→      firstSpecByAe.set(sl.applicantEligibilityId, sl.specificEligibilityId);
   158→   158→    }
   159→   159→  }
   160→   160→  // First eligibilityId per specificEligibilityId (lowest ord).
   161→   161→  const firstEligBySpec = new Map<number, number | null>();
   162→   162→  for (const sel of specificEligLinks) {
   163→   163→    if (sel.specificEligibilityId == null) continue;
   164→   164→    if (!firstEligBySpec.has(sel.specificEligibilityId)) {
   165→   165→      firstEligBySpec.set(sel.specificEligibilityId, sel.eligibilityId);
   166→   166→    }
   167→   167→  }
   168→   168→  const eligIds = Array.from(
   169→   169→    new Set(
   170→   170→      Array.from(firstEligBySpec.values()).filter(
   171→   171→        (x): x is number => x != null
   172→   172→      )
   173→   173→    )
   174→   174→  );
   175→   175→  const eligibilities = eligIds.length
   176→   176→    ? await db.eligibility.findMany({ where: { id: { in: eligIds } } })
   177→   177→    : [];
   178→   178→  const eligNameById = new Map(eligibilities.map((e) => [e.id, e.name]));
   179→   179→
   180→   180→  for (const it of items) {
   181→   181→    if (it.eligibilityTitle != null) continue;
   182→   182→    const specId = firstSpecByAe.get(it.id) ?? null;
   183→   183→    if (specId == null) continue;
   184→   184→    // Prefer specific_eligibilities.name (e.g. "RA7883 Barangay Health Worker").
   185→   185→    const specName = specNameById.get(specId) ?? null;
   186→   186→    if (specName) {
   187→   187→      it.eligibilityTitle = specName;
   188→   188→      continue;
   189→   189→    }
   190→   190→    // Fallback: chain through specific_eligibilities_eligibility_lnk → eligibilities.name.
   191→   191→    const eligId = firstEligBySpec.get(specId) ?? null;
   192→   192→    if (eligId != null) {
   193→   193→      it.eligibilityTitle = eligNameById.get(eligId) ?? null;
   194→   194→    }
   195→   195→  }
   196→   196→}

   335→   335→/**
   336→   336→ * Find-or-create the `eligibilities` reference row that carries an applicant
   337→   337→ * eligibility's TITLE (the ApplicantEligibility table has no title column —
   338→   338→ * the title lives on the linked Eligibility row via the category link).
   339→   339→ *
   340→   340→ * Previously the lookup-only version silently DROPPED any title that wasn't
   341→   341→ * already in the reference vocabulary, so titles typed through the profile
   342→   342→ * form (or extracted from a PDS) vanished from the list after a reload.
   343→   343→ * This now creates the missing reference row, so every title persists.
   344→   344→ *
   345→   345→ * Created rows are DRAFTS (publishedAt left null): they resolve for the
   346→   346→ * owner's own list (loadApplicantEligibilities does not filter published)
   347→   347→ * but do NOT leak into the public /api/reference dropdown other applicants
   348→   348→ * see. Published vocabulary rows are only ever matched, never duplicated.
   349→   349→ */
   350→   350→export async function findOrCreateEligibilityReference(
   351→   351→  name: string
   352→   352→): Promise<number> {
   353→   353→  const trimmed = name.trim();
   354→   354→  const existing = await db.eligibility.findFirst({ where: { name: trimmed } });
   355→   355→  if (existing) return existing.id;
   356→   356→  const now = new Date();
   357→   357→  const created = await db.eligibility.create({
   358→   358→    data: { name: trimmed, createdAt: now, updatedAt: now },
   359→   359→  });
   360→   360→  return created.id;
   361→   361→}

   363→   363→// ---- Bulk section clearers (used by auto-apply OVERRIDE behavior) ----
   364→   364→//
   365→   365→// When an applicant uploads a new PDS / document, the extracted data is the
   366→   366→// LATEST source of truth for matching form fields. Before writing the new
   367→   367→// entries we must CLEAR the existing child rows + their junction links for
   368→   368→// that section so the new document fully REPLACES the old data (instead of
   369→   369→// appending duplicates).
   370→   370→//
   371→   371→// Each clearer:
   372→   372→//   1. Loads all junction links for the applicant (to get the child row ids).
   373→   373→//   2. Deletes the junction links.
   374→   374→//   3. Deletes the child rows themselves.

   383→   383→  await db.applicantEducation.deleteMany({ where: { id: { in: ids } } });
   384→   384→  return ids.length;

   562→   562→// ---- Batch loaders (eliminate N+1 in list endpoints) ----
   563→   563→
   564→   564→/**
   565→   565→ * Batch-load the first position for each of multiple job postings.
   566→   566→ * Returns a Map<jobPostingId, Position> for O(1) lookup.
   567→   567→ * Uses 2 queries total (junction + positions) instead of 2N.
   568→   568→ */
   569→   569→export async function batchLoadJobPostingPositions(jobPostingIds: number[]) {
   570→   570→  const result = new Map<number, Awaited<ReturnType<typeof db.position.findFirst>>>();
   571→   571→  if (!jobPostingIds.length) return result;
   572→   572→
   573→   573→  const links = await db.jobPostingPositionLink.findMany({
   574→   574→    where: { jobpostingId: { in: jobPostingIds } },
   575→   575→    orderBy: { postionOrd: "asc" },
   576→   576→  });
   577→   577→  const positionIds = links
   578→   578→    .map((l) => l.postionId)
   579→   579→    .filter((x): x is number => x != null);
   580→   580→  if (!positionIds.length) return result;
   581→   581→
   582→   582→  const positions = await db.position.findMany({ where: { id: { in: positionIds } } });
   583→   583→  const positionById = new Map(positions.map((p) => [p.id, p]));
   584→   584→
   585→   585→  // For each job posting, take the first linked position (lowest ord)
   586→   586→  const seen = new Set<number>();
   587→   587→  for (const link of links) {
   588→   588→    if (link.jobpostingId == null) continue;
   589→   589→    if (seen.has(link.jobpostingId)) continue;
   590→   590→    seen.add(link.jobpostingId);
   591→   591→    if (link.postionId != null) {
   592→   592→      const pos = positionById.get(link.postionId);
   593→   593→      if (pos) result.set(link.jobpostingId, pos);
   594→   594→    }
   595→   595→  }
   596→   596→  return result;
   597→   597→}
   598→   598→
   599→   599→/**
   600→   600→ * Batch-load the first place of assignment for each of multiple positions.
   601→   601→ * Returns a Map<positionId, {id, name}> for O(1) lookup.
   602→   602→ * Uses 2 queries total instead of 2N.
   603→   603→ */
   604→   604→export async function batchLoadPositionPlaceOfAssignment(positionIds: number[]) {
   605→   605→  const result = new Map<number, { id: number; name: string | null }>();
   606→   606→  if (!positionIds.length) return result;
   607→   607→
   608→   608→  const links = await db.positionPlaceOfAssignmentLink.findMany({
   609→   609→    where: { postionId: { in: positionIds } },
   610→   610→  });
   611→   611→  const placeIds = links
   612→   612→    .map((l) => l.placeOfAssignmentId)
   613→   613→    .filter((x): x is number => x != null);
   614→   614→  if (!placeIds.length) return result;
   615→   615→
   616→   616→  const places = await db.placeOfAssignment.findMany({ where: { id: { in: placeIds } } });
   617→   617→  const placeById = new Map(places.map((p) => [p.id, p]));
   618→   618→
   619→   619→  const seen = new Set<number>();
   620→   620→  for (const link of links) {
   621→   621→    if (link.postionId == null) continue;
   622→   622→    if (seen.has(link.postionId)) continue;
   623→   623→    seen.add(link.postionId);
   624→   624→    if (link.placeOfAssignmentId != null) {
   625→   625→      const place = placeById.get(link.placeOfAssignmentId);
   626→   626→      if (place) result.set(link.postionId, { id: place.id, name: place.name });
   627→   627→    }
   628→   628→  }
   629→   629→  return result;
   630→   630→}

   398→   398→export default function Recruitment() {
   399→   399→  const [jobs, setJobs] = useState<JobWire[] | null>(null);
   400→   400→  const [error, setError] = useState<string | null>(null);
   401→   401→  const [search, setSearch] = useState("");
   402→   402→  const [debouncedSearch, setDebouncedSearch] = useState("");
   403→   403→  const [statusFilter, setStatusFilter] = useState<"all" | "open" | "closed">("all");
   404→   404→  const [sortBy, setSortBy] = useState<"recent" | "deadline" | "applications">("recent");
   405→   405→  const [page, setPage] = useState(1);
   406→   406→  const [formOpen, setFormOpen] = useState(false);
   407→   407→  const [spin, setSpin] = useState(false);
   408→   408→
   409→   409→  const load = useCallback(async (silent = false) => {
   410→   410→    if (!silent) setError(null);
   411→   411→    try {
   412→   412→      const data = await apiFetch<JobWire[]>("/api/jobs?limit=200");
   413→   413→      setJobs(data);
   414→   414→    } catch (e) {
   415→   415→      if (!silent) setError(e instanceof Error ? e.message : "Failed to load job postings");
   416→   416→    }
   417→   417→  }, []);
   418→   418→
   419→   419→  useEffect(() => {
   420→   420→    void load();
   421→   421→  }, [load]);
   422→   422→
   423→   423→  // 30 s silent poll + refetch-on-focus.
   424→   424→  useEffect(() => {
   425→   425→    const id = setInterval(() => {
   426→   426→      if (!document.hidden) void load(true);
   427→   427→    }, 30_000);
   428→   428→    return () => clearInterval(id);
   429→   429→  }, [load]);
   430→   430→  useEffect(() => {
   431→   431→    const onFocus = () => void load(true);
   432→   432→    window.addEventListener("focus", onFocus);
   433→   433→    document.addEventListener("visibilitychange", onFocus);
   434→   434→    return () => {
   435→   435→      window.removeEventListener("focus", onFocus);
   436→   436→      document.removeEventListener("visibilitychange", onFocus);
   437→   437→    };
   438→   438→  }, [load]);
   439→   439→
   440→   440→  // Debounced search (350 ms).
   441→   441→  useEffect(() => {
   442→   442→    const t = setTimeout(() => {
   443→   443→      setDebouncedSearch(search.trim().toLowerCase());
   444→   444→      setPage(1);
   445→   445→    }, 350);
   446→   446→    return () => clearTimeout(t);
   447→   447→  }, [search]);
   448→   448→
   449→   449→  const filtered = useMemo(() => {
   450→   450→    let rows = jobs ?? [];
   451→   451→    if (debouncedSearch) {
   452→   452→      rows = rows.filter((j) => {

   454→   454→          j.title,
   455→   455→          j.position?.itemNumber,

   457→   457→          j.positionType,
   458→   458→        ]
   459→   459→          .filter(Boolean)
   460→   460→          .join(" ")
   461→   461→          .toLowerCase();
   462→   462→        return hay.includes(debouncedSearch);
   463→   463→      });
   464→   464→    }
   465→   465→    if (statusFilter === "open") rows = rows.filter(isOpenJob);
   466→   466→    else if (statusFilter === "closed") rows = rows.filter((j) => !isOpenJob(j));
   467→   467→    const sorted = [...rows];
   468→   468→    if (sortBy === "recent") {
   469→   469→      sorted.sort((a, b) => new Date(b.publishedAt).getTime() - new Date(a.publishedAt).getTime());
   470→   470→    } else if (sortBy === "deadline") {
   471→   471→      sorted.sort((a, b) => {
   472→   472→        const av = a.deadlineDate ? new Date(a.deadlineDate).getTime() : Number.POSITIVE_INFINITY;
   473→   473→        const bv = b.deadlineDate ? new Date(b.deadlineDate).getTime() : Number.POSITIVE_INFINITY;
   474→   474→        return av - bv;
   475→   475→      });
   476→   476→    } else {
   477→   477→      sorted.sort((a, b) => b.applicationCount - a.applicationCount);
   478→   478→    }
   479→   479→    return sorted;
   480→   480→  }, [jobs, debouncedSearch, statusFilter, sortBy]);
   481→   481→
   482→   482→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE));
   483→   483→  const safePage = Math.min(page, totalPages);
   484→   484→  const pageRows = filtered.slice((safePage - 1) * PAGE_SIZE, safePage * PAGE_SIZE);
   485→   485→
   486→   486→  const manualRefresh = useCallback(() => {
   487→   487→    setSpin(true);
   488→   488→    void load(true).finally(() => setTimeout(() => setSpin(false), 500));
   489→   489→  }, [load]);
   490→   490→
   491→   491→  return (
   492→   492→    <div className="space-y-6">
   493→   493→      <div className="flex flex-wrap items-center gap-3">

   495→   495→        <span className="rounded-full bg-fog text-ink text-xs px-2.5 py-1">
   496→   496→          {filtered.length} posting{filtered.length === 1 ? "" : "s"}
   497→   497→        </span>
   498→   498→        <div className="ml-auto flex items-center gap-2">
   499→   499→          <button type="button" className={ghostBtn} onClick={manualRefresh}>
   500→   500→            <RefreshCw className={`h-4 w-4 ${spin ? "animate-spin" : ""}`} aria-hidden />
   501→   501→            Refresh
   502→   502→          </button>
   503→   503→          <button type="button" className={ctaBtn} onClick={() => setFormOpen(true)}>
   504→   504→            <Plus className="h-4 w-4" aria-hidden />
   505→   505→            Create Job
   506→   506→          </button>
   507→   507→        </div>
   508→   508→      </div>
   509→   509→
   510→   510→      {/* Filter bar */}
   511→   511→      <div className="dlg-card p-4 grid gap-3 sm:grid-cols-[1fr_180px_200px]">
   512→   512→        <Input
   513→   513→          className="dlg-input"
   514→   514→          placeholder="Search title, item no., or place…"
   515→   515→          value={search}
   516→   516→          onChange={(e) => setSearch(e.target.value)}
   517→   517→          aria-label="Search job postings"
   518→   518→        />
   519→   519→        <Select
   520→   520→          value={statusFilter}
   521→   521→          onValueChange={(v) => {
   522→   522→            setStatusFilter(v as typeof statusFilter);
   523→   523→            setPage(1);
   524→   524→          }}
   525→   525→        >
   526→   526→          <SelectTrigger className="dlg-input min-h-[44px] w-full" aria-label="Status filter">
   527→   527→            <SelectValue />
   528→   528→          </SelectTrigger>
   529→   529→          <SelectContent>
   530→   530→            <SelectItem value="all">All statuses</SelectItem>
   531→   531→            <SelectItem value="open">Open</SelectItem>
   532→   532→            <SelectItem value="closed">Closed</SelectItem>
   533→   533→          </SelectContent>
   534→   534→        </Select>
   535→   535→        <Select
   536→   536→          value={sortBy}
   537→   537→          onValueChange={(v) => {
   538→   538→            setSortBy(v as typeof sortBy);
   539→   539→            setPage(1);
   540→   540→          }}
   541→   541→        >
   542→   542→          <SelectTrigger className="dlg-input min-h-[44px] w-full" aria-label="Sort">
   543→   543→            <SelectValue />
   544→   544→          </SelectTrigger>
   545→   545→          <SelectContent>
   546→   546→            <SelectItem value="recent">Recently posted</SelectItem>
   547→   547→            <SelectItem value="deadline">Deadline</SelectItem>
   548→   548→            <SelectItem value="applications">Applications</SelectItem>
   549→   549→          </SelectContent>
   550→   550→        </Select>
   551→   551→      </div>
   552→   552→
   553→   553→      {error ? (
   554→   554→        <div className="dlg-card p-8 text-center space-y-4">
   555→   555→          <AlertTriangle className="h-8 w-8 text-dusty-rose mx-auto" aria-hidden />
   556→   556→          <p className="text-sm text-stone">{error}</p>
   557→   557→          <button type="button" className={ghostBtn} onClick={() => void load()}>
   558→   558→            Retry
   559→   559→          </button>
   560→   560→        </div>
   561→   561→      ) : jobs === null ? (
   562→   562→        <div className="dlg-card p-6 space-y-3">
   563→   563→          {Array.from({ length: 6 }).map((_, i) => (
   564→   564→            <Skeleton key={i} className="h-12 w-full rounded-[12px]" />
   565→   565→          ))}
   566→   566→        </div>
   567→   567→      ) : pageRows.length === 0 ? (
   568→   568→        <div className="dlg-card p-10 text-center space-y-3">
   569→   569→          <p className="text-sm text-pebble">No job postings match the current filters.</p>
   570→   570→          <button type="button" className={ctaBtn} onClick={() => setFormOpen(true)}>
   571→   571→            <Plus className="h-4 w-4" aria-hidden />
   572→   572→            Create Job
   573→   573→          </button>
   574→   574→        </div>
   575→   575→      ) : (
   576→   576→        <div className="dlg-card overflow-x-auto">
   577→   577→          <table className="w-full text-sm">
   578→   578→            <thead>
   579→   579→              <tr className="border-b border-[#ececec] text-left text-xs text-pebble">
   580→   580→                <th className="px-5 py-3 font-medium">Position</th>
   581→   581→                <th className="px-4 py-3 font-medium">Vacancies</th>
   582→   582→                <th className="px-4 py-3 font-medium">Monthly salary</th>
   583→   583→                <th className="px-4 py-3 font-medium">Applications</th>
   584→   584→                <th className="px-4 py-3 font-medium">Deadline</th>
   585→   585→                <th className="px-5 py-3 font-medium">Status</th>
   586→   586→              </tr>
   587→   587→            </thead>
   588→   588→            <tbody>
   589→   589→              {pageRows.map((j) => {
   590→   590→                const dl = deadlineState(j.deadlineDate);
   591→   591→                return (
   592→   592→                  <tr
   593→   593→                    key={j.id}
   594→   594→                    className="border-b border-[#ececec] last:border-0 cursor-pointer hover:bg-fog/60 transition-colors"
   595→   595→                    onClick={() => navigate("job", { id: String(j.id) })}

   608→   608→                    <td className={`px-4 py-3.5 ${dl.overdue ? "text-dusty-rose" : "text-stone"}`}>

   610→   610→                    </td>
   611→   611→                    <td className="px-5 py-3.5">
   612→   612→                      <OpenClosedPill active={isOpenJob(j)} />
   613→   613→                    </td>
   614→   614→                  </tr>
   615→   615→                );
   616→   616→              })}
   617→   617→            </tbody>
   618→   618→          </table>
   619→   619→        </div>
   620→   620→      )}
   621→   621→
   622→   622→      {/* Pagination */}
   623→   623→      {jobs !== null && filtered.length > 0 && (
   624→   624→        <div className="flex items-center justify-between gap-3">
   625→   625→          <p className="text-xs text-pebble">
   626→   626→            {(safePage - 1) * PAGE_SIZE + 1}–{Math.min(safePage * PAGE_SIZE, filtered.length)} of {filtered.length}
   627→   627→          </p>
   628→   628→          <div className="flex items-center gap-2">
   629→   629→            <button type="button" className={ghostBtn + " min-h-[36px] px-3 text-xs"} disabled={safePage <= 1} onClick={() => setPage(safePage - 1)}>
   630→   630→              Previous
   631→   631→            </button>
   632→   632→            <span className="text-xs text-stone tabular-nums">
   633→   633→              Page {safePage} of {totalPages}
   634→   634→            </span>
   635→   635→            <button type="button" className={ghostBtn + " min-h-[36px] px-3 text-xs"} disabled={safePage >= totalPages} onClick={() => setPage(safePage + 1)}>
   636→   636→              Next

   638→   638→          </div>
   639→   639→        </div>
   640→   640→      )}
   641→   641→
   642→   642→      <JobFormDialog open={formOpen} onOpenChange={setFormOpen} job={null} onSaved={() => void load(true)} />
   643→   643→    </div>
   644→   644→  );

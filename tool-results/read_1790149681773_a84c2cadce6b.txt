    57→function divisionShort(code: string | null | undefined): string {
    58→  if (!code) return "—";

    62→function stripHtml(html: string): string {

    66→function sgLabel(grade: string | null | undefined, step: string | null | undefined): string {
    67→  if (!grade) return "—";
    68→  return step ? `SG${grade}/${step}` : `SG${grade}`;

    71→function pageWindow(current: number, total: number): (number | "…")[] {
    72→  if (total <= 7) return Array.from({ length: total }, (_, i) => i + 1);
    73→  const out: (number | "…")[] = [1];
    74→  const lo = Math.max(2, current - 1);
    75→  const hi = Math.min(total - 1, current + 1);
    76→  if (lo > 2) out.push("…");
    77→  for (let i = lo; i <= hi; i++) out.push(i);
    78→  if (hi < total - 1) out.push("…");
    79→  out.push(total);
    80→  return out;
    81→}

    83→/** % of the application window elapsed between publish and deadline (0–100). */
    84→function deadlineProgress(job: JobWire): number {
    85→  const start = new Date(job.publishDate ?? job.publishedAt).getTime();
    86→  const end = job.deadlineDate ? new Date(job.deadlineDate).getTime() : Number.NaN;
    87→  if (Number.isNaN(start) || Number.isNaN(end) || end <= start) return job.deadlineDate ? 100 : 0;
    88→  const pct = ((Date.now() - start) / (end - start)) * 100;
    89→  return Math.min(100, Math.max(0, pct));
    90→}

   109→function RichOrPlain({ html, text }: { html?: string | null; text?: string | null }) {
   110→  if (html && html.trim()) {
   111→    return (
   112→      <div
   113→        className="rich-text text-sm leading-relaxed text-stone"
   114→        dangerouslySetInnerHTML={{ __html: html }}
   115→      />
   116→    );
   117→  }
   118→  if (text && text.trim()) {
   119→    return <p className="whitespace-pre-line text-sm leading-relaxed text-stone">{text}</p>;
   120→  }
   121→  return <p className="text-sm text-pebble">Not provided.</p>;

   142→function QuickView({
   143→  job,
   144→  onApply,
   145→  onDetail,
   146→}: {
   147→  job: JobWire;
   148→  onApply: (job: JobWire) => void;
   149→  onDetail: (job: JobWire) => void;
   150→}) {
   151→  const pos = job.position;
   152→  const dl = deadlineState(job.deadlineDate);
   153→  const brief = job.briefDescription?.trim()
   154→    ? job.briefDescription
   155→    : job.briefDescriptionHtml
   156→      ? stripHtml(job.briefDescriptionHtml)
   157→      : "";
   158→  const facts: [string, string][] = [
   159→    ["Item No.", pos?.itemNumber || "—"],
   160→    ["Vacancies", String(job.numberOfVacancy)],
   161→    ["Salary Grade", sgLabel(pos?.salaryGrade, pos?.salaryStep)],
   162→    ["Monthly Salary", pos?.salaryAmount != null ? formatCurrency(pos.salaryAmount) : "—"],
   163→    ["Published", formatDate(job.publishDate ?? job.publishedAt)],
   164→    ["Deadline", formatDate(job.deadlineDate)],
   165→  ];
   166→  return (
   167→    <div className="mt-4 space-y-4 border-t border-border pt-4">
   168→      {brief ? (
   169→        <p className="line-clamp-4 text-sm leading-relaxed text-stone">{brief}</p>
   170→      ) : (
   171→        <p className="text-sm text-pebble">No brief description provided.</p>
   172→      )}
   173→      <div className="grid grid-cols-2 gap-3 sm:grid-cols-3">
   174→        {facts.map(([label, value]) => (
   175→          <div key={label} className="rounded-none bg-fog p-3">
   176→            <p className="text-xs text-pebble">{label}</p>
   177→            <p className="mt-0.5 truncate text-sm text-ink" title={value}>
   178→              {value}
   179→            </p>
   180→          </div>
   181→        ))}
   182→      </div>
   183→      <div className="flex flex-wrap gap-3">
   184→        <button
   185→          onClick={() => onApply(job)}
   186→          disabled={dl.overdue}
   187→          className="dlg-cta min-h-[44px] px-6 text-sm disabled:pointer-events-none disabled:opacity-50"
   188→        >
   189→          Apply now
   190→        </button>
   191→        <button onClick={() => onDetail(job)} className="dlg-ghost min-h-[44px] px-6 text-sm">
   192→          Read full description
   193→        </button>
   194→      </div>
   195→    </div>
   196→  );
   197→}

   538→export default function JobsView() {
   539→  const route = useHashRoute();
   540→  const detailParam = route.params.job ?? null;
   541→
   542→  const [jobs, setJobs] = useState<JobWire[] | null>(null);
   543→  const [error, setError] = useState<string | null>(null);
   544→  const [query, setQuery] = useState("");
   545→  const [divisionFilter, setDivisionFilter] = useState<string[]>([]);
   546→  const [sort, setSort] = useState<SortKey>("newest");
   547→  const [page, setPage] = useState(1);
   548→  const [expandedId, setExpandedId] = useState<number | null>(null);
   549→  const [filtersOpen, setFiltersOpen] = useState(false);
   550→  const [cancelling, setCancelling] = useState(false);
   551→  const lastJson = useRef("");
   552→
   553→  const { requestApply, dialogs } = useApplyFlow();
   554→
   555→  const load = useCallback(async (silent: boolean) => {
   556→    try {
   557→      const data = await apiFetch<JobWire[]>("/api/jobs?limit=200");
   558→      const json = JSON.stringify(data);
   559→      if (!silent || json !== lastJson.current) {
   560→        lastJson.current = json;
   561→        setJobs(data);
   562→      }
   563→      setError(null);
   564→    } catch (e) {
   565→      if (!silent) setError(e instanceof Error ? e.message : "Failed to load positions");
   566→    }
   567→  }, []);
   568→
   569→  // Mount + 20s silent poll + refetch on focus (spec §13 cadence).
   570→  useEffect(() => {
   571→    void load(false);
   572→    const timer = setInterval(() => {
   573→      if (!document.hidden) void load(true);
   574→    }, 20_000);
   575→    const onFocus = () => void load(true);
   576→    window.addEventListener("focus", onFocus);
   577→    return () => {
   578→      clearInterval(timer);
   579→      window.removeEventListener("focus", onFocus);
   580→    };
   581→  }, [load]);
   582→
   583→  // Facet counts over the full corpus.
   584→  const facets = useMemo(() => {
   585→    const map = new Map<string, number>();
   586→    for (const j of jobs ?? []) {
   587→      const code = j.position?.division;
   588→      if (!code) continue;
   589→      map.set(code, (map.get(code) ?? 0) + 1);
   590→    }
   591→    return [...map.entries()].sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0]));
   592→  }, [jobs]);
   593→
   594→  // Filtering + sorting.
   595→  const filtered = useMemo(() => {
   596→    let list = jobs ?? [];
   597→    const needle = query.trim().toLowerCase();
   598→    if (needle) {
   599→      list = list.filter((j) =>
   600→        `${j.title} ${j.position?.positionTitle ?? ""} ${j.position?.itemNumber ?? ""}`
   601→          .toLowerCase()
   602→          .includes(needle)
   603→      );
   604→    }
   605→    if (divisionFilter.length > 0) {
   606→      list = list.filter((j) => divisionFilter.includes(j.position?.division ?? ""));
   607→    }
   608→    const deadlineTime = (j: JobWire) =>
   609→      j.deadlineDate ? new Date(j.deadlineDate).getTime() : Number.POSITIVE_INFINITY;
   610→    const sorted = [...list];
   611→    if (sort === "newest") {
   612→      sorted.sort(
   613→        (a, b) => new Date(b.publishedAt).getTime() - new Date(a.publishedAt).getTime()
   614→      );
   615→    } else if (sort === "deadline") {
   616→      sorted.sort((a, b) => deadlineTime(a) - deadlineTime(b));
   617→    } else {
   618→      sorted.sort((a, b) => (b.position?.salaryAmount ?? -1) - (a.position?.salaryAmount ?? -1));
   619→    }
   620→    return sorted;
   621→  }, [jobs, query, divisionFilter, sort]);
   622→
   623→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE));
   624→  const current = Math.min(page, totalPages);
   625→  const pageItems = filtered.slice((current - 1) * PAGE_SIZE, current * PAGE_SIZE);
   626→
   627→  const changeQuery = (v: string) => {
   628→    setQuery(v);
   629→    setPage(1);
   630→  };
   631→  const toggleDivision = (code: string) => {
   632→    setDivisionFilter((prev) =>
   633→      prev.includes(code) ? prev.filter((c) => c !== code) : [...prev, code]
   634→    );
   635→    setPage(1);
   636→  };
   637→  const clearFilters = () => {
   638→    setQuery("");
   639→    setDivisionFilter([]);
   640→    setPage(1);
   641→  };
   642→  const toggleExpanded = (id: number) => setExpandedId((prev) => (prev === id ? null : id));
   643→  const openDetail = (job: JobWire) => navigate("jobs", { job: String(job.id) });
   644→
   645→  const applyTo = useCallback(
   646→    (job: JobWire) => {
   647→      requestApply(job);
   648→    },
   649→    [requestApply]
   650→  );
   651→
   652→  // Cancel application (only while status "Applied" — spec §7.7).
   653→  const appliedId = useMemo(() => {
   654→    const detail = detailParam ? (jobs ?? []).find((j) => j.id === Number(detailParam)) : null;
   655→    return detail?.applications?.[0]?.id ?? null;
   656→  }, [detailParam, jobs]);
   657→
   658→  const cancelApplication = useCallback(async () => {
   659→    if (appliedId == null) return;
   660→    setCancelling(true);
   661→    try {
   662→      await apiFetch(`/api/applications/${appliedId}`, { method: "DELETE" });
   663→      toast.success("Application cancelled", {
   664→        description: "Your application was withdrawn. You can apply again while the position is open.",
   665→      });
   666→      await load(false);
   667→    } catch (e) {
   668→      toast.error(e instanceof Error ? e.message : "Could not cancel the application");
   669→    } finally {
   670→      setCancelling(false);
   671→    }
   672→  }, [appliedId, load]);
   673→
   674→  // ── Detail mode ───────────────────────────────────────────────────────────
   675→  if (detailParam && /^\d+$/.test(detailParam)) {
   676→    const detailId = Number(detailParam);
   677→    const detail = jobs?.find((j) => j.id === detailId) ?? null;
   678→
   679→    return (
   680→      <div className="mx-auto w-full max-w-[1600px] px-4 py-6 sm:px-6 sm:py-8 lg:px-8">
   681→        <button
   682→          onClick={() => navigate("jobs")}
   683→          className="mb-5 flex min-h-[44px] items-center gap-2 text-sm text-stone hover:text-ink"
   684→        >
   685→          <ArrowLeft className="h-4 w-4" aria-hidden="true" />
   686→          Back to Positions
   687→        </button>
   688→
   689→        {jobs === null && error === null ? (
   690→          <div className="space-y-6">
   691→            <div className="dlg-card space-y-4 p-8">
   692→              <Skeleton className="h-6 w-32" />
   693→              <Skeleton className="h-10 w-2/3" />
   694→              <Skeleton className="h-4 w-1/2" />
   695→            </div>
   696→            <div className="dlg-card space-y-3 p-8">
   697→              <Skeleton className="h-5 w-48" />
   698→              <Skeleton className="h-4 w-full" />
   699→              <Skeleton className="h-4 w-full" />
   700→              <Skeleton className="h-4 w-3/4" />
   701→            </div>
   702→          </div>
   703→        ) : error !== null ? (
   704→          <ErrorCard message={error} onRetry={() => void load(false)} />
   705→        ) : detail === null ? (
   706→          <div className="dlg-card p-10 text-center">
   707→            <p className="font-display text-xl text-carbon">Position not found</p>
   708→            <p className="mt-2 text-sm text-stone">
   709→              This position may have closed or is no longer published.
   710→            </p>
   711→            <button onClick={() => navigate("jobs")} className="dlg-cta mt-5 min-h-[44px] px-7 text-sm">
   712→              Back to Positions
   713→            </button>
   714→          </div>
   715→        ) : (
   716→          <div className="grid grid-cols-1 items-start gap-6 xl:grid-cols-[1fr_340px]">
   717→            <DetailBody job={detail} />
   718→            <aside className="xl:sticky xl:top-6">
   719→              <SummaryCard
   720→                job={detail}
   721→                onApply={applyTo}
   722→                onCancel={() => void cancelApplication()}
   723→                cancelling={cancelling}
   724→              />
   725→            </aside>
   726→          </div>
   727→        )}
   728→
   729→        {dialogs}
   730→      </div>
   731→    );
   732→  }
   733→
   734→  // ── List mode ─────────────────────────────────────────────────────────────
   735→  const activeFilterCount = (query.trim() ? 1 : 0) + divisionFilter.length;
   736→
   737→  const filterRail = (
   738→    <div className="dlg-card p-4">
   739→      <label htmlFor="job-search" className="sr-only">
   740→        Search positions
   741→      </label>
   742→      <div className="relative">
   743→        <Search
   744→          className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-pebble"
   745→          aria-hidden="true"
   746→        />
   747→        <Input
   748→          id="job-search"
   749→          value={query}
   750→          onChange={(e) => changeQuery(e.target.value)}
   751→          placeholder="Search positions…"
   752→          className="dlg-input min-h-[44px] pl-9"
   753→        />
   754→      </div>
   755→
   756→      <p className="mt-5 px-1 text-xs font-medium uppercase tracking-wider text-pebble">Division</p>
   757→      <div className="mt-2 space-y-0.5">
   758→        {facets.map(([code, count]) => (
   759→          <label
   760→            key={code}
   761→            className="flex min-h-[44px] cursor-pointer items-center gap-3 rounded-none px-2 hover:bg-fog"
   762→          >
   763→            <Checkbox
   764→              checked={divisionFilter.includes(code)}
   765→              onCheckedChange={() => toggleDivision(code)}
   766→            />
   767→            <span className="min-w-0 flex-1 truncate text-sm text-stone" title={divisionName(code)}>
   768→              {divisionShort(code)}
   769→            </span>
   770→            <span className="text-xs text-pebble">{count}</span>
   771→          </label>
   772→        ))}
   773→        {facets.length === 0 && <p className="px-2 py-2 text-sm text-pebble">No divisions on file</p>}
   774→      </div>
   775→
   776→      {(query.trim() || divisionFilter.length > 0) && (
   777→        <button
   778→          onClick={clearFilters}
   779→          className="min-h-[44px] px-2 text-sm text-stone underline underline-offset-4 hover:text-ink"
   780→        >
   781→          Clear filters
   782→        </button>
   783→      )}
   784→    </div>
   785→  );
   786→
   787→  let content: React.ReactNode;
   788→  if (jobs === null && error === null) {
   789→    content = (
   790→      <div className="space-y-4">
   791→        {[0, 1, 2, 3].map((i) => (
   792→          <div key={i} className="dlg-card p-6">
   793→            <Skeleton className="h-5 w-24" />
   794→            <Skeleton className="mt-3 h-6 w-2/3" />
   795→            <Skeleton className="mt-2 h-4 w-full" />
   796→          </div>
   797→        ))}
   798→      </div>
   799→    );
   800→  } else if (error !== null && jobs === null) {
   801→    content = <ErrorCard message={error} onRetry={() => void load(false)} />;
   802→  } else if ((jobs ?? []).length === 0) {
   803→    content = (
   804→      <div className="dlg-card p-10 text-center">
   805→        <p className="font-display text-xl text-carbon">No open positions</p>
   806→        <p className="mt-2 text-sm text-stone">
   807→          There are no published postings at the moment — check back soon.
   808→        </p>
   809→      </div>
   810→    );
   811→  } else if (filtered.length === 0) {
   812→    content = (
   813→      <div className="dlg-card p-10 text-center">
   814→        <p className="font-display text-xl text-carbon">No positions match your filters</p>
   815→        <p className="mt-2 text-sm text-stone">Try a different search term or clear the filters.</p>
   816→        <button onClick={clearFilters} className="dlg-ghost mt-5 min-h-[44px] px-6 text-sm">
   817→          Clear filters
   818→        </button>
   819→      </div>
   820→    );
   821→  } else {
   822→    content = (
   823→      <>
   824→        <div className="space-y-4">
   825→          {pageItems.map((job) => (
   826→            <JobCard
   827→              key={job.id}
   828→              job={job}
   829→              expanded={expandedId === job.id}
   830→              onToggle={toggleExpanded}
   831→              onApply={applyTo}
   832→              onDetail={openDetail}
   833→            />
   834→          ))}
   835→        </div>
   836→
   837→        {totalPages > 1 && (
   838→          <nav className="mt-6 flex items-center justify-center gap-1.5" aria-label="Pagination">
   839→            <button
   840→              onClick={() => setPage((p) => Math.max(1, p - 1))}
   841→              disabled={current === 1}
   842→              aria-label="Previous page"
   843→              className="dlg-ghost grid h-11 w-11 place-items-center disabled:opacity-40"
   844→            >
   845→              <ChevronLeft className="h-4 w-4" aria-hidden="true" />
   846→            </button>
   847→            {pageWindow(current, totalPages).map((p, i) =>
   848→              p === "…" ? (
   849→                <span key={`ellipsis-${i}`} className="grid h-11 w-8 place-items-center text-sm text-pebble">
   850→                  …
   851→                </span>
   852→              ) : (
   853→                <button
   854→                  key={p}
   855→                  onClick={() => setPage(p)}
   856→                  aria-current={p === current ? "page" : undefined}
   857→                  className={
   858→                    p === current
   859→                      ? "grid h-11 min-w-11 place-items-center bg-ink px-3 text-sm font-medium text-white"
   860→                      : "grid h-11 min-w-11 place-items-center px-3 text-sm text-stone hover:bg-fog hover:text-ink"
   861→                  }
   862→                >
   863→                  {p}
   864→                </button>
   865→              )
   866→            )}
   867→            <button
   868→              onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   869→              disabled={current === totalPages}
   870→              aria-label="Next page"
   871→              className="dlg-ghost grid h-11 w-11 place-items-center disabled:opacity-40"
   872→            >
   873→              <ChevronRight className="h-4 w-4" aria-hidden="true" />
   874→            </button>
   875→          </nav>
   876→        )}
   877→      </>
   878→    );
   879→  }
   880→
   881→  return (
   882→    <div className="mx-auto w-full max-w-[1600px] px-4 py-6 sm:px-6 sm:py-8 lg:px-8">
   883→      <div className="mb-5 max-w-2xl">
   884→        <h1 className="text-heading-lg">Positions</h1>
   885→        <p className="mt-2 text-sm leading-relaxed text-stone">
   886→          Browse published vacancies at DOST-MIRDC and submit your application online.
   887→        </p>
   888→      </div>
   889→
   890→      <div className="items-start lg:flex lg:gap-6">
   891→        {/* Search rail — sticky on desktop, collapsible on mobile */}
   892→        <aside className="mb-4 w-full lg:sticky lg:top-20 lg:mb-0 lg:w-72 lg:shrink-0">
   893→          <div className="lg:hidden">
   894→            <Collapsible open={filtersOpen} onOpenChange={setFiltersOpen}>
   895→              <CollapsibleTrigger asChild>
   896→                <button className="dlg-ghost flex min-h-[44px] w-full items-center justify-between px-5 text-sm">
   897→                  <span className="flex items-center gap-2">
   898→                    <SlidersHorizontal className="h-4 w-4" aria-hidden="true" />
   899→                    Filters
   900→                    {activeFilterCount > 0 && (
   901→                      <span className="grid h-5 min-w-5 place-items-center bg-ink px-1.5 text-[11px] font-medium text-white">
   902→                        {activeFilterCount}
   903→                      </span>
   904→                    )}
   905→                  </span>
   906→                  <ChevronDown
   907→                    className={`h-4 w-4 transition-transform ${filtersOpen ? "rotate-180" : ""}`}
   908→                    aria-hidden="true"
   909→                  />
   910→                </button>
   911→              </CollapsibleTrigger>
   912→              <CollapsibleContent>
   913→                <div className="pt-3">{filterRail}</div>
   914→              </CollapsibleContent>
   915→            </Collapsible>
   916→          </div>
   917→          <div className="hidden lg:block">{filterRail}</div>
   918→        </aside>
   919→
   920→        {/* Results */}
   921→        <div className="min-w-0 flex-1">
   922→          <div className="mb-4 flex flex-wrap items-center justify-between gap-3">
   923→            <p className="text-sm text-stone">
   924→              {filtered.length} Result{filtered.length === 1 ? "" : "s"}
   925→            </p>
   926→            <div className="flex items-center gap-2">
   927→              <span className="hidden text-sm text-stone sm:inline">Sort</span>
   928→              <Select
   929→                value={sort}
   930→                onValueChange={(v) => {
   931→                  setSort(v as SortKey);
   932→                  setPage(1);
   933→                }}
   934→              >
   935→                <SelectTrigger className="dlg-input min-h-[44px] w-[170px]" aria-label="Sort results">
   936→                  <SelectValue />
   937→                </SelectTrigger>
   938→                <SelectContent className="rounded-none">
   939→                  <SelectItem value="newest">Newest</SelectItem>
   940→                  <SelectItem value="deadline">Deadline</SelectItem>
   941→                  <SelectItem value="salary">Salary desc</SelectItem>
   942→                </SelectContent>
   943→              </Select>
   944→            </div>
   945→          </div>
   946→
   947→          {/* Active-filter chips — presentation of existing filter state */}
   948→          {activeFilterCount > 0 && (
   949→            <div className="mb-4 flex flex-wrap items-center gap-2">
   950→              {query.trim() && (
   951→                <span className="dlg-pill inline-flex items-center gap-1.5 bg-white px-3 py-1 text-xs font-medium text-graphite">
   952→                  &ldquo;{query.trim()}&rdquo;
   953→                  <button
   954→                    onClick={() => changeQuery("")}
   955→                    aria-label="Clear search filter"
   956→                    className="focus-ring -mr-1 grid h-5 w-5 place-items-center text-stone hover:bg-fog hover:text-ink"
   957→                  >
   958→                    <X className="h-3 w-3" aria-hidden="true" />
   959→                  </button>
   960→                </span>
   961→              )}
   962→              {divisionFilter.map((code) => (
   963→                <span
   964→                  key={code}
   965→                  className="dlg-pill inline-flex items-center gap-1.5 bg-white px-3 py-1 text-xs font-medium text-graphite"
   966→                >
   967→                  {divisionShort(code)}
   968→                  <button
   969→                    onClick={() => toggleDivision(code)}
   970→                    aria-label={`Remove ${divisionShort(code)} filter`}
   971→                    className="focus-ring -mr-1 grid h-5 w-5 place-items-center text-stone hover:bg-fog hover:text-ink"
   972→                  >
   973→                    <X className="h-3 w-3" aria-hidden="true" />
   974→                  </button>
   975→                </span>
   976→              ))}
   977→              <button
   978→                onClick={clearFilters}
   979→                className="text-xs text-stone underline underline-offset-4 hover:text-ink"
   980→              >
   981→                Clear all
   982→              </button>
   983→            </div>
   984→          )}
   985→
   986→          {content}
   987→        </div>
   988→      </div>
   989→
   990→      {dialogs}
   991→    </div>
   992→  );
   993→}
   994→

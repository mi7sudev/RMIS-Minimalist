   421→const PAGE_SIZE = 10;
   422→
   423→export default function Recruitment() {
   424→  const [jobs, setJobs] = useState<JobWire[] | null>(null);
   425→  const [error, setError] = useState<string | null>(null);
   426→  const [search, setSearch] = useState("");
   427→  const [debouncedSearch, setDebouncedSearch] = useState("");
   428→  const [statusFilter, setStatusFilter] = useState<"all" | "open" | "closed">("all");
   429→  const [sortBy, setSortBy] = useState<"recent" | "deadline" | "applications">("recent");
   430→  const [page, setPage] = useState(1);
   431→  const [formOpen, setFormOpen] = useState(false);
   432→  const [editJob, setEditJob] = useState<JobWire | null>(null);
   433→  const [spin, setSpin] = useState(false);
   434→
   435→  const openCreate = () => {
   436→    setEditJob(null);
   437→    setFormOpen(true);
   438→  };
   439→  const openEdit = (j: JobWire) => {
   440→    setEditJob(j);
   441→    setFormOpen(true);
   442→  };
   443→
   444→  const load = useCallback(async (silent = false) => {
   445→    if (!silent) setError(null);
   446→    try {
   447→      const data = await apiFetch<JobWire[]>("/api/jobs?limit=200");
   448→      setJobs(data);
   449→    } catch (e) {
   450→      if (!silent) setError(e instanceof Error ? e.message : "Failed to load job postings");
   451→    }
   452→  }, []);
   453→
   454→  useEffect(() => {
   455→    void load();
   456→  }, [load]);
   457→
   458→  // 30 s silent poll + refetch-on-focus.
   459→  useEffect(() => {
   460→    const id = setInterval(() => {
   461→      if (!document.hidden) void load(true);
   462→    }, 30_000);
   463→    return () => clearInterval(id);
   464→  }, [load]);
   465→  useEffect(() => {
   466→    const onFocus = () => void load(true);
   467→    window.addEventListener("focus", onFocus);
   468→    document.addEventListener("visibilitychange", onFocus);
   469→    return () => {
   470→      window.removeEventListener("focus", onFocus);
   471→      document.removeEventListener("visibilitychange", onFocus);
   472→    };
   473→  }, [load]);
   474→
   475→  // Debounced search (350 ms).
   476→  useEffect(() => {
   477→    const t = setTimeout(() => {
   478→      setDebouncedSearch(search.trim().toLowerCase());
   479→      setPage(1);
   480→    }, 350);
   481→    return () => clearTimeout(t);
   482→  }, [search]);
   483→
   484→  const filtered = useMemo(() => {
   485→    let rows = jobs ?? [];
   486→    if (debouncedSearch) {
   487→      rows = rows.filter((j) => {
   488→        const hay = [
   489→          j.title,
   490→          j.position?.itemNumber,
   491→          j.position?.placeOfAssignment,
   492→          j.positionType,
   493→        ]
   494→          .filter(Boolean)
   495→          .join(" ")
   496→          .toLowerCase();
   497→        return hay.includes(debouncedSearch);
   498→      });
   499→    }
   500→    if (statusFilter === "open") rows = rows.filter(isOpenJob);
   501→    else if (statusFilter === "closed") rows = rows.filter((j) => !isOpenJob(j));
   502→    const sorted = [...rows];
   503→    if (sortBy === "recent") {
   504→      sorted.sort((a, b) => new Date(b.publishedAt).getTime() - new Date(a.publishedAt).getTime());
   505→    } else if (sortBy === "deadline") {
   506→      sorted.sort((a, b) => {
   507→        const av = a.deadlineDate ? new Date(a.deadlineDate).getTime() : Number.POSITIVE_INFINITY;
   508→        const bv = b.deadlineDate ? new Date(b.deadlineDate).getTime() : Number.POSITIVE_INFINITY;
   509→        return av - bv;
   510→      });
   511→    } else {
   512→      sorted.sort((a, b) => b.applicationCount - a.applicationCount);
   513→    }
   514→    return sorted;
   515→  }, [jobs, debouncedSearch, statusFilter, sortBy]);
   516→
   517→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE));
   518→  const safePage = Math.min(page, totalPages);
   519→  const pageRows = filtered.slice((safePage - 1) * PAGE_SIZE, safePage * PAGE_SIZE);
   520→
   521→  const manualRefresh = useCallback(() => {
   522→    setSpin(true);
   523→    void load(true).finally(() => setTimeout(() => setSpin(false), 500));
   524→  }, [load]);
   525→
   526→  return (
   527→    <div className="space-y-6">
   528→      <PageHeader
   529→        title="Recruitment"
   530→        description={`${filtered.length} posting${filtered.length === 1 ? "" : "s"}`}
   531→        actions={
   532→          <>
   533→            <button type="button" className={ghostBtn} onClick={manualRefresh}>

    21→  const byId = new Map(rows.map((r) => [r.id, r]));

    23→    .map((l) => (l.applicantEducationId ? byId.get(l.applicantEducationId) : undefined))

    62→export async function loadApplicantEligibilities(applicantId: number) {
    63→  const links = await db.applicantEligibilityLink.findMany({
    64→    where: { applicantId },
    65→    orderBy: { applicantEligibilityOrd: "asc" },
    66→  });
    67→  if (!links.length) return [];
    68→  const ids = links.map((l) => l.applicantEligibilityId).filter((x): x is number => x != null);
    69→  const [rows, categoryLinks] = await Promise.all([
    70→    db.applicantEligibility.findMany({ where: { id: { in: ids } } }),
    71→    db.applicantEligibilityCategoryLink.findMany({
    72→      where: { applicantEligibilityId: { in: ids } },
    73→    }),
    74→  ]);
    75→  // Map each applicant_eligibility → its first linked eligibility category name
    76→  const eligibilityIds = Array.from(
    77→    new Set(categoryLinks.map((c) => c.eligibilityId).filter((x): x is number => x != null))
    78→  );
    79→  const categories = eligibilityIds.length
    80→    ? await db.eligibility.findMany({ where: { id: { in: eligibilityIds } } })
    81→    : [];
    82→  const catById = new Map(categories.map((c) => [c.id, c.name]));
    83→  const catLinkByApplElig = new Map<number, number | null>();
    84→  for (const cl of categoryLinks) {
    85→    if (cl.applicantEligibilityId == null) continue;
    86→    if (!catLinkByApplElig.has(cl.applicantEligibilityId)) {
    87→      catLinkByApplElig.set(cl.applicantEligibilityId, cl.eligibilityId);
    88→    }
    89→  }
    90→  const byId = new Map(rows.map((r) => [r.id, r]));
    91→  return links
    92→    .map((l) => {
    93→      if (!l.applicantEligibilityId) return undefined;
    94→      const row = byId.get(l.applicantEligibilityId);
    95→      if (!row) return undefined;
    96→      const catId = catLinkByApplElig.get(row.id) ?? null;
    97→      const eligibilityTitle = catId != null ? catById.get(catId) ?? null : null;
    98→      return { ...row, eligibilityTitle };
    99→    })
   100→    .filter((x): x is NonNullable<typeof x> => x != null);
   101→}

   103→/**
   104→ * Backfill `eligibilityTitle` for eligibility items whose title is still null.
   105→ *
   106→ * Production stores the title link in `specific_eligibilities_applicant_eligibility_lnk`
   107→ * (not in `applicant_eligibilities_eligibility_category_lnk`, which has 0 rows in
   108→ * production). This resolves the title via the path:
   109→ *
   110→ *   specific_eligibilities_applicant_eligibility_lnk
   111→ *     → specific_eligibilities (name)             ← prefer this
   112→ *     → specific_eligibilities_eligibility_lnk
   113→ *         → eligibilities (name)                  ← fallback
   114→ *
   115→ * Only items whose title is still null are touched, so newly-created entries
   116→ * (which write to the category link) keep working. Patches items in place.
   117→ */
   118→export async function fillEligibilityTitleFromSpecificLinks<
   119→  T extends { id: number; eligibilityTitle: string | null }
   120→>(items: T[]): Promise<void> {
   121→  const aeIds = items.filter((it) => it.eligibilityTitle == null).map((it) => it.id);
   122→  if (!aeIds.length) return;
   123→
   124→  const specificLinks = await db.specificEligibilityApplicantEligibilityLink.findMany({
   125→    where: { applicantEligibilityId: { in: aeIds } },
   126→    orderBy: { specificEligibilityOrd: "asc" },
   127→  });
   128→  if (!specificLinks.length) return;
   129→
   130→  const specificIds = Array.from(
   131→    new Set(
   132→      specificLinks
   133→        .map((l) => l.specificEligibilityId)
   134→        .filter((x): x is number => x != null)
   135→    )
   136→  );
   137→  const [specifics, specificEligLinks] = await Promise.all([
   138→    specificIds.length
   139→      ? db.specificEligibility.findMany({ where: { id: { in: specificIds } } })
   140→      : [],
   141→    specificIds.length
   142→      ? db.specificEligibilityEligibilityLink.findMany({
   143→          where: { specificEligibilityId: { in: specificIds } },
   144→          orderBy: { specificEligibilityOrd: "asc" },
   145→        })
   146→      : [],
   147→  ]);
   148→
   149→  const specNameById = new Map<number, string | null>(
   150→    specifics.map((s) => [s.id, s.name] as [number, string | null])
   151→  );
   152→  // First specificEligibilityId per applicantEligibilityId (lowest ord).
   153→  const firstSpecByAe = new Map<number, number | null>();
   154→  for (const sl of specificLinks) {
   155→    if (sl.applicantEligibilityId == null) continue;
   156→    if (!firstSpecByAe.has(sl.applicantEligibilityId)) {
   157→      firstSpecByAe.set(sl.applicantEligibilityId, sl.specificEligibilityId);
   158→    }
   159→  }
   160→  // First eligibilityId per specificEligibilityId (lowest ord).
   161→  const firstEligBySpec = new Map<number, number | null>();
   162→  for (const sel of specificEligLinks) {
   163→    if (sel.specificEligibilityId == null) continue;
   164→    if (!firstEligBySpec.has(sel.specificEligibilityId)) {
   165→      firstEligBySpec.set(sel.specificEligibilityId, sel.eligibilityId);
   166→    }
   167→  }
   168→  const eligIds = Array.from(
   169→    new Set(
   170→      Array.from(firstEligBySpec.values()).filter(
   171→        (x): x is number => x != null
   172→      )
   173→    )
   174→  );
   175→  const eligibilities = eligIds.length
   176→    ? await db.eligibility.findMany({ where: { id: { in: eligIds } } })
   177→    : [];
   178→  const eligNameById = new Map(eligibilities.map((e) => [e.id, e.name]));
   179→
   180→  for (const it of items) {
   181→    if (it.eligibilityTitle != null) continue;
   182→    const specId = firstSpecByAe.get(it.id) ?? null;
   183→    if (specId == null) continue;
   184→    // Prefer specific_eligibilities.name (e.g. "RA7883 Barangay Health Worker").
   185→    const specName = specNameById.get(specId) ?? null;
   186→    if (specName) {
   187→      it.eligibilityTitle = specName;
   188→      continue;
   189→    }
   190→    // Fallback: chain through specific_eligibilities_eligibility_lnk → eligibilities.name.
   191→    const eligId = firstEligBySpec.get(specId) ?? null;
   192→    if (eligId != null) {
   193→      it.eligibilityTitle = eligNameById.get(eligId) ?? null;
   194→    }
   195→  }
   196→}

   248→export async function loadApplicantFullProfile(applicantId: number): Promise<ApplicantFullProfile | null> {
   249→  const applicant = await db.applicant.findUnique({ where: { id: applicantId } });
   250→  if (!applicant) return null;
   251→  // Use allSettled for the same resilience as the profile route — a single
   252→  // corrupted section should not prevent the full-profile snapshot from loading.
   253→  const settled = await Promise.allSettled([
   254→    loadApplicantEducations(applicantId),
   255→    loadApplicantWorkExperiences(applicantId),
   256→    loadApplicantTrainings(applicantId),
   257→    loadApplicantEligibilities(applicantId),
   258→    loadApplicantAwards(applicantId),
   259→  ]);
   260→  const educations = settled[0].status === "fulfilled" ? settled[0].value : [];
   261→  const workExperiences = settled[1].status === "fulfilled" ? settled[1].value : [];
   262→  const trainings = settled[2].status === "fulfilled" ? settled[2].value : [];
   263→  const eligibilities = settled[3].status === "fulfilled" ? settled[3].value : [];
   264→  const awards = settled[4].status === "fulfilled" ? settled[4].value : [];
   265→  // raw-json functions are synchronous (better-sqlite3 is sync)
   266→  const characterReferenceRaw = getApplicantCharacterReference(applicantId);
   267→  return {
   268→    id: applicant.id,
   269→    firstName: applicant.firstName,
   270→    middleName: applicant.middleName,
   271→    lastName: applicant.lastName,
   272→    extensionName: applicant.extensionName,
   273→    emailAddress: applicant.emailAddress,
   274→    contactNumber: applicant.contactNumber,
   275→    mobileNumber: applicant.mobileNumber,
   276→    gender: applicant.gender,
   277→    civilStatus: applicant.civilStatus,
   278→    citizenship: applicant.citizenship,
   279→    birthDate: applicant.birthDate,
   280→    birthPlace: applicant.birthPlace,
   281→    presentAddress: applicant.presentAddress,
   282→    city: applicant.city,
   283→    province: applicant.province,
   284→    country: applicant.country,
   285→    zipCode: applicant.zipCode,

   335→/**
   336→ * Find-or-create the `eligibilities` reference row that carries an applicant
   337→ * eligibility's TITLE (the ApplicantEligibility table has no title column —
   338→ * the title lives on the linked Eligibility row via the category link).
   339→ *
   340→ * Previously the lookup-only version silently DROPPED any title that wasn't
   341→ * already in the reference vocabulary, so titles typed through the profile
   342→ * form (or extracted from a PDS) vanished from the list after a reload.
   343→ * This now creates the missing reference row, so every title persists.
   344→ *
   345→ * Created rows are DRAFTS (publishedAt left null): they resolve for the
   346→ * owner's own list (loadApplicantEligibilities does not filter published)
   347→ * but do NOT leak into the public /api/reference dropdown other applicants
   348→ * see. Published vocabulary rows are only ever matched, never duplicated.
   349→ */
   350→export async function findOrCreateEligibilityReference(
   351→  name: string
   352→): Promise<number> {
   353→  const trimmed = name.trim();
   354→  const existing = await db.eligibility.findFirst({ where: { name: trimmed } });
   355→  if (existing) return existing.id;
   356→  const now = new Date();
   357→  const created = await db.eligibility.create({
   358→    data: { name: trimmed, createdAt: now, updatedAt: now },
   359→  });
   360→  return created.id;
   361→}

   363→// ---- Bulk section clearers (used by auto-apply OVERRIDE behavior) ----

   365→// When an applicant uploads a new PDS / document, the extracted data is the
   366→// LATEST source of truth for matching form fields. Before writing the new

   372→//   1. Loads all junction links for the applicant (to get the child row ids).
   373→//   2. Deletes the junction links.

   471→// ---- JobPosting ↔ Position (junction: jobpostings_postions_lnk) ----
   472→export async function loadJobPostingPositions(jobpostingId: number) {
   473→  const links = await db.jobPostingPositionLink.findMany({
   474→    where: { jobpostingId },
   475→    orderBy: { postionOrd: "asc" },
   476→  });
   477→  if (!links.length) return [];
   478→  const ids = links.map((l) => l.postionId).filter((x): x is number => x != null);
   479→  const positions = await db.position.findMany({ where: { id: { in: ids } } });
   480→  const byId = new Map(positions.map((p) => [p.id, p]));
   481→  return links
   482→    .map((l) => (l.postionId ? byId.get(l.postionId) : undefined))
   483→    .filter((x): x is NonNullable<typeof x> => x != null);
   484→}

   506→// ---- Application ↔ Applicant / JobPosting (junction tables) ----
   507→export async function findApplicationForApplicantJob(applicantId: number, jobpostingId: number) {

   509→  // also linked to the given job posting.

   511→  const applicationIds = applLinks
   512→    .map((l) => l.applicationId)
   513→    .filter((x): x is number => x != null);
   514→  if (!applicationIds.length) return null;
   515→  const jobLinks = await db.applicationJobLink.findMany({
   516→    where: { applicationId: { in: applicationIds }, jobpostingId },
   517→  });
   518→  if (!jobLinks.length) return null;
   519→  const applicationId = jobLinks[0].applicationId;
   520→  return db.application.findUnique({ where: { id: applicationId! } });
   521→}

   562→// ---- Batch loaders (eliminate N+1 in list endpoints) ----
   563→
   564→/**
   565→ * Batch-load the first position for each of multiple job postings.
   566→ * Returns a Map<jobPostingId, Position> for O(1) lookup.
   567→ * Uses 2 queries total (junction + positions) instead of 2N.
   568→ */
   569→export async function batchLoadJobPostingPositions(jobPostingIds: number[]) {
   570→  const result = new Map<number, Awaited<ReturnType<typeof db.position.findFirst>>>();
   571→  if (!jobPostingIds.length) return result;
   572→
   573→  const links = await db.jobPostingPositionLink.findMany({
   574→    where: { jobpostingId: { in: jobPostingIds } },
   575→    orderBy: { postionOrd: "asc" },
   576→  });
   577→  const positionIds = links
   578→    .map((l) => l.postionId)
   579→    .filter((x): x is number => x != null);
   580→  if (!positionIds.length) return result;
   581→
   582→  const positions = await db.position.findMany({ where: { id: { in: positionIds } } });
   583→  const positionById = new Map(positions.map((p) => [p.id, p]));
   584→
   585→  // For each job posting, take the first linked position (lowest ord)
   586→  const seen = new Set<number>();
   587→  for (const link of links) {
   588→    if (link.jobpostingId == null) continue;
   589→    if (seen.has(link.jobpostingId)) continue;
   590→    seen.add(link.jobpostingId);
   591→    if (link.postionId != null) {
   592→      const pos = positionById.get(link.postionId);
   593→      if (pos) result.set(link.jobpostingId, pos);
   594→    }
   595→  }
   596→  return result;
   597→}

   599→/**
   600→ * Batch-load the first place of assignment for each of multiple positions.
   601→ * Returns a Map<positionId, {id, name}> for O(1) lookup.
   602→ * Uses 2 queries total instead of 2N.
   603→ */
   604→export async function batchLoadPositionPlaceOfAssignment(positionIds: number[]) {
   605→  const result = new Map<number, { id: number; name: string | null }>();
   606→  if (!positionIds.length) return result;
   607→
   608→  const links = await db.positionPlaceOfAssignmentLink.findMany({
   609→    where: { postionId: { in: positionIds } },
   610→  });
   611→  const placeIds = links
   612→    .map((l) => l.placeOfAssignmentId)
   613→    .filter((x): x is number => x != null);
   614→  if (!placeIds.length) return result;
   615→
   616→  const places = await db.placeOfAssignment.findMany({ where: { id: { in: placeIds } } });
   617→  const placeById = new Map(places.map((p) => [p.id, p]));
   618→
   619→  const seen = new Set<number>();
   620→  for (const link of links) {
   621→    if (link.postionId == null) continue;
   622→    if (seen.has(link.postionId)) continue;
   623→    seen.add(link.postionId);
   624→    if (link.placeOfAssignmentId != null) {
   625→      const place = placeById.get(link.placeOfAssignmentId);
   626→      if (place) result.set(link.postionId, { id: place.id, name: place.name });
   627→    }
   628→  }
   629→  return result;
   630→}

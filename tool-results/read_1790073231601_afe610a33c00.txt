   396→const PAGE_SIZE = 10;
   397→
   398→export default function Recruitment() {
   399→  const [jobs, setJobs] = useState<JobWire[] | null>(null);
   400→  const [error, setError] = useState<string | null>(null);
   401→  const [search, setSearch] = useState("");
   402→  const [debouncedSearch, setDebouncedSearch] = useState("");
   403→  const [statusFilter, setStatusFilter] = useState<"all" | "open" | "closed">("all");
   404→  const [sortBy, setSortBy] = useState<"recent" | "deadline" | "applications">("recent");
   405→  const [page, setPage] = useState(1);
   406→  const [formOpen, setFormOpen] = useState(false);
   407→  const [spin, setSpin] = useState(false);
   408→
   409→  const load = useCallback(async (silent = false) => {
   410→    if (!silent) setError(null);
   411→    try {
   412→      const data = await apiFetch<JobWire[]>("/api/jobs?limit=200");
   413→      setJobs(data);
   414→    } catch (e) {
   415→      if (!silent) setError(e instanceof Error ? e.message : "Failed to load job postings");
   416→    }
   417→  }, []);
   418→
   419→  useEffect(() => {
   420→    void load();
   421→  }, [load]);
   422→
   423→  // 30 s silent poll + refetch-on-focus.
   424→  useEffect(() => {
   425→    const id = setInterval(() => {
   426→      if (!document.hidden) void load(true);
   427→    }, 30_000);
   428→    return () => clearInterval(id);
   429→  }, [load]);
   430→  useEffect(() => {
   431→    const onFocus = () => void load(true);
   432→    window.addEventListener("focus", onFocus);
   433→    document.addEventListener("visibilitychange", onFocus);
   434→    return () => {
   435→      window.removeEventListener("focus", onFocus);
   436→      document.removeEventListener("visibilitychange", onFocus);
   437→    };
   438→  }, [load]);
   439→
   440→  // Debounced search (350 ms).
   441→  useEffect(() => {
   442→    const t = setTimeout(() => {
   443→      setDebouncedSearch(search.trim().toLowerCase());
   444→      setPage(1);
   445→    }, 350);
   446→    return () => clearTimeout(t);
   447→  }, [search]);
   448→
   449→  const filtered = useMemo(() => {
   450→    let rows = jobs ?? [];
   451→    if (debouncedSearch) {
   452→      rows = rows.filter((j) => {
   453→        const hay = [
   454→          j.title,
   455→          j.position?.itemNumber,
   456→          j.position?.placeOfAssignment,
   457→          j.positionType,
   458→        ]
   459→          .filter(Boolean)
   460→          .join(" ")
   461→          .toLowerCase();
   462→        return hay.includes(debouncedSearch);
   463→      });
   464→    }
   465→    if (statusFilter === "open") rows = rows.filter(isOpenJob);
   466→    else if (statusFilter === "closed") rows = rows.filter((j) => !isOpenJob(j));
   467→    const sorted = [...rows];
   468→    if (sortBy === "recent") {
   469→      sorted.sort((a, b) => new Date(b.publishedAt).getTime() - new Date(a.publishedAt).getTime());
   470→    } else if (sortBy === "deadline") {
   471→      sorted.sort((a, b) => {
   472→        const av = a.deadlineDate ? new Date(a.deadlineDate).getTime() : Number.POSITIVE_INFINITY;
   473→        const bv = b.deadlineDate ? new Date(b.deadlineDate).getTime() : Number.POSITIVE_INFINITY;
   474→        return av - bv;
   475→      });
   476→    } else {
   477→      sorted.sort((a, b) => b.applicationCount - a.applicationCount);
   478→    }
   479→    return sorted;
   480→  }, [jobs, debouncedSearch, statusFilter, sortBy]);
   481→
   482→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE));
   483→  const safePage = Math.min(page, totalPages);
   484→  const pageRows = filtered.slice((safePage - 1) * PAGE_SIZE, safePage * PAGE_SIZE);
   485→
   486→  const manualRefresh = useCallback(() => {
   487→    setSpin(true);
   488→    void load(true).finally(() => setTimeout(() => setSpin(false), 500));
   489→  }, [load]);
   490→
   491→  return (
   492→    <div className="space-y-6">
   493→      <div className="flex flex-wrap items-center gap-3">
   494→        <h1 className="font-display text-2xl text-ink">Recruitment</h1>
   495→        <span className="rounded-full bg-fog text-ink text-xs px-2.5 py-1">
   496→          {filtered.length} posting{filtered.length === 1 ? "" : "s"}
   497→        </span>
   498→        <div className="ml-auto flex items-center gap-2">
   499→          <button type="button" className={ghostBtn} onClick={manualRefresh}>
   500→            <RefreshCw className={`h-4 w-4 ${spin ? "animate-spin" : ""}`} aria-hidden />
   501→            Refresh
   502→          </button>
   503→          <button type="button" className={ctaBtn} onClick={() => setFormOpen(true)}>
   504→            <Plus className="h-4 w-4" aria-hidden />
   505→            Create Job
   506→          </button>
   507→        </div>
   508→      </div>
   509→
   510→      {/* Filter bar */}
   511→      <div className="dlg-card p-4 grid gap-3 sm:grid-cols-[1fr_180px_200px]">
   512→        <Input
   513→          className="dlg-input"
   514→          placeholder="Search title, item no., or place…"
   515→          value={search}
   516→          onChange={(e) => setSearch(e.target.value)}
   517→          aria-label="Search job postings"
   518→        />
   519→        <Select
   520→          value={statusFilter}
   521→          onValueChange={(v) => {
   522→            setStatusFilter(v as typeof statusFilter);
   523→            setPage(1);
   524→          }}
   525→        >
   526→          <SelectTrigger className="dlg-input min-h-[44px] w-full" aria-label="Status filter">
   527→            <SelectValue />
   528→          </SelectTrigger>
   529→          <SelectContent>
   530→            <SelectItem value="all">All statuses</SelectItem>
   531→            <SelectItem value="open">Open</SelectItem>
   532→            <SelectItem value="closed">Closed</SelectItem>
   533→          </SelectContent>
   534→        </Select>
   535→        <Select
   536→          value={sortBy}
   537→          onValueChange={(v) => {
   538→            setSortBy(v as typeof sortBy);
   539→            setPage(1);
   540→          }}

   622→      {/* Pagination */}

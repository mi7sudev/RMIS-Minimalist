   127→/** Gradient monogram avatar (wave-3) with initials derived from the name. */
   128→function MonogramAvatar({ name, size = 36, warm = false }: { name: string; size?: number; warm?: boolean }) {
   129→  const initials = name
   130→    .split(/\s+/)
   131→    .filter(Boolean)
   132→    .slice(0, 2)
   133→    .map((w) => w[0]?.toUpperCase() ?? "")
   134→    .join("");
   135→  return (
   136→    <span aria-hidden="true">
   137→      <Monogram size={size} warm={warm}>{initials || "?"}</Monogram>
   138→    </span>
   139→  );
   140→}
   141→
   142→function useFocusRefresh(fn: () => void) {
   143→  useEffect(() => {
   144→    const onFocus = () => fn();
   145→    window.addEventListener("focus", onFocus);
   146→    document.addEventListener("visibilitychange", onFocus);
   147→    return () => {
   148→      window.removeEventListener("focus", onFocus);
   149→      document.removeEventListener("visibilitychange", onFocus);
   150→    };
   151→  }, [fn]);
   152→}

   186→export default function ReviewQueue() {
   187→  const [queue, setQueue] = useState<QueueRow[] | null>(null);
   188→  const [roster, setRoster] = useState<RosterRow[] | null>(null);
   189→  const [error, setError] = useState<string | null>(null);
   190→  const [qualifiedOnly, setQualifiedOnly] = useState(false);
   191→  const [mode, setMode] = useState<"kanban" | "list">("kanban");
   192→  const [listStage, setListStage] = useState<"All" | StageKey>("All");
   193→  const [selectedId, setSelectedId] = useState<number | null>(null);
   194→  const [regretOpen, setRegretOpen] = useState(false);
   195→  const [regretBusy, setRegretBusy] = useState(false);
   196→  const [spin, setSpin] = useState(false);
   197→
   198→  const load = useCallback(async (silent = false) => {
   199→    if (!silent) setError(null);
   200→    try {
   201→      const [q, r] = await Promise.all([
   202→        apiFetch<{ data: QueueRow[] }>("/api/evaluator/queue?pageSize=100"),
   203→        apiFetch<{ data: RosterRow[] }>("/api/admin/applicants?pageSize=100"),
   204→      ]);
   205→      setQueue(q.data);
   206→      setRoster(r.data);
   207→      if (!silent) setError(null);
   208→    } catch (e) {
   209→      if (!silent) setError(e instanceof Error ? e.message : "Failed to load the review queue");
   210→    }
   211→  }, []);
   212→
   213→  useEffect(() => {
   214→    void load();
   215→  }, [load]);
   216→
   217→  // 15 s silent poll + refetch-on-focus (spec §13).
   218→  useEffect(() => {
   219→    const id = setInterval(() => {
   220→      if (!document.hidden) void load(true);
   221→    }, 15_000);
   222→    return () => clearInterval(id);
   223→  }, [load]);
   224→  useFocusRefresh(() => void load(true));
   225→
   226→  const manualRefresh = useCallback(() => {
   227→    setSpin(true);
   228→    void load(true).finally(() => setTimeout(() => setSpin(false), 500));
   229→  }, [load]);
   230→
   231→  const applications = useMemo(() => queue ?? [], [queue]);
   232→  const filtered = useMemo(
   233→    () => (qualifiedOnly ? applications.filter((r) => r.match?.verdict === "ALL_MET") : applications),
   234→    [applications, qualifiedOnly]
   235→  );
   236→  const listRows = useMemo(
   237→    () => (listStage === "All" ? filtered : filtered.filter((r) => stageForStatus(r.status) === listStage)),
   238→    [filtered, listStage]
   239→  );
   240→  const rejectedIds = useMemo(
   241→    () => filtered.filter((r) => stageForStatus(r.status) === "Rejected").map((r) => r.id),
   242→    [filtered]
   243→  );
   244→
   245→  const kanbanColumns = useMemo(() => {
   246→    const map: Record<StageKey, QueueRow[]> = { "Applied": [], "Under Review": [], "Shortlisted": [], "Rejected": [] };
   247→    for (const r of filtered) map[stageForStatus(r.status)].push(r);
   248→    return map;
   249→  }, [filtered]);
   250→
   251→  const rosterCards = useMemo(() => (roster ?? []).filter((r) => r.applicationCount === 0), [roster]);
   252→
   253→  const sendRegrets = async () => {
   254→    setRegretBusy(true);
   255→    try {
   256→      const res = await apiFetch<{ summary: RegretSummary }>("/api/evaluator/applications/regrets", {
   257→        method: "POST",
   258→        body: { applicationIds: rejectedIds },
   259→      });
   260→      const s = res.summary;
   261→      const parts = [
   262→        `${s.sent} sent`,
   263→        `${s.alreadySent} already sent`,
   264→        s.shortlisted ? `${s.shortlisted} shortlisted skipped` : "",
   265→        s.failed ? `${s.failed} failed` : "",
   266→        s.notFound ? `${s.notFound} not found` : "",
   267→      ].filter(Boolean);
   268→      if (s.sent > 0) {
   269→        toast.success(`Regret letters: ${s.sent} sent`, { description: parts.join(" · ") });
   270→      } else {
   271→        toast.info("No new regret letters sent", { description: parts.join(" · ") });
   272→      }
   273→      setRegretOpen(false);
   274→      await load(true);
   275→    } catch (e) {
   276→      toast.error(e instanceof Error ? e.message : "Failed to send regret letters");
   277→    } finally {
   278→      setRegretBusy(false);
   279→    }
   280→  };
   281→
   282→  return (
   283→    <div className="space-y-6">
   284→      {/* Header + toolbar */}
   285→      <PageHeader
   286→        title="Review Queue"
   287→        description="Screen applications, verify minimum qualifications, and advance candidates through the pipeline."
   288→        actions={
   289→          <>
   290→            <span className="status-pill status-neutral num">
   291→              {filtered.length} application{filtered.length === 1 ? "" : "s"}
   292→            </span>
   293→            <label className="focus-ring inline-flex min-h-[44px] cursor-pointer items-center gap-2 rounded-full border border-border bg-white px-3.5 text-sm text-stone transition-colors hover:border-ink/10">
   294→              <Switch checked={qualifiedOnly} onCheckedChange={setQualifiedOnly} aria-label="Qualified only" />
   295→              Qualified only
   296→            </label>
   297→            <div className="flex items-center gap-1 rounded-full bg-fog p-1" role="group" aria-label="View mode">
   298→              <button
   299→                type="button"
   300→                className={`inline-flex min-h-[36px] items-center gap-1.5 rounded-full px-3 text-xs font-medium transition-colors ${mode === "kanban" ? "bg-ink text-white" : "text-stone hover:text-ink"}`}
   301→                onClick={() => setMode("kanban")}
   302→                aria-pressed={mode === "kanban"}
   303→              >
   304→                <LayoutGrid className="h-3.5 w-3.5" aria-hidden />
   305→                Kanban
   306→              </button>
   307→              <button
   308→                type="button"
   309→                className={`inline-flex min-h-[36px] items-center gap-1.5 rounded-full px-3 text-xs font-medium transition-colors ${mode === "list" ? "bg-ink text-white" : "text-stone hover:text-ink"}`}
   310→                onClick={() => setMode("list")}
   311→                aria-pressed={mode === "list"}
   312→              >
   313→                <List className="h-3.5 w-3.5" aria-hidden />
   314→                List
   315→              </button>
   316→            </div>
   317→            <button type="button" className={ghostBtn} onClick={manualRefresh}>
   318→              <RefreshCw className={`h-4 w-4 ${spin ? "animate-spin" : ""}`} aria-hidden />
   319→              Refresh
   320→            </button>
   321→          </>
   322→        }
   323→      />
   324→
   325→      {/* Stage tabs (list mode only) */}
   326→      {mode === "list" && (
   327→        <div className="flex flex-wrap items-center gap-2">
   328→          {(["All", ...PIPELINE_STAGES] as const).map((s) => {
   329→            const count = s === "All" ? filtered.length : filtered.filter((r) => stageForStatus(r.status) === s).length;
   330→            const active = listStage === s;
   331→            return (
   332→              <button
   333→                key={s}
   334→                type="button"
   335→                className={`inline-flex min-h-[36px] items-center gap-2 rounded-full px-3.5 text-xs font-medium transition-colors ${active ? "bg-ink text-white" : "dlg-ghost"}`}
   336→                onClick={() => setListStage(s)}
   337→                aria-pressed={active}
   338→              >
   339→                {s !== "All" && <span className={dotClass(STAGE_DOT[s])} aria-hidden />}
   340→                {s}
   341→                <span className={`num rounded-full px-1.5 ${active ? "bg-white/20" : "bg-fog"}`}>{count}</span>
   342→              </button>
   343→            );
   344→          })}
   345→          {listStage === "Rejected" && rejectedIds.length > 0 && (
   346→            <button
   347→              type="button"
   348→              className={ghostBtn + " border-[var(--bad)]/30 text-[var(--bad)]"}
   349→              onClick={() => setRegretOpen(true)}
   350→            >
   351→              <MailX className="h-4 w-4" aria-hidden />
   352→              Send regret letters ({rejectedIds.length})
   353→            </button>
   354→          )}
   355→        </div>
   356→      )}
   357→
   358→      {/* Error */}
   359→      {error && (
   360→        <div className="dlg-card p-8">
   361→          <EmptyState
   362→            icon={AlertTriangle}
   363→            tone="rose"
   364→            title="Couldn't load the review queue"
   365→            description={error}
   366→            action={
   367→              <button type="button" className={ghostBtn} onClick={() => void load()}>
   368→                Retry
   369→              </button>
   370→            }
   371→          />
   372→        </div>
   373→      )}
   374→
   375→      {/* Loading skeleton */}
   376→      {!error && (queue === null || roster === null) && (
   377→        <>
   378→          <SkeletonKpis count={4} />
   379→          {mode === "kanban" ? <SkeletonKanban columns={5} /> : <SkeletonRows rows={8} rowClassName="h-16" />}
   380→        </>
   381→      )}
   382→
   383→      {/* Stage KPIs (wave-3) */}
   384→      {!error && queue !== null && roster !== null && (
   385→        <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   386→          <KpiCard label="Applied" value={kanbanColumns["Applied"].length} icon={Inbox} tone="info" hint="awaiting start" />
   387→          <KpiCard label="Under Review" value={kanbanColumns["Under Review"].length} icon={ScanSearch} tone="warn" hint="being screened" />
   388→          <KpiCard label="Shortlisted" value={kanbanColumns["Shortlisted"].length} icon={Star} tone="ok" hint="advancing" />
   389→          <KpiCard label="Rejected" value={kanbanColumns["Rejected"].length} icon={XCircle} tone="bad" hint="not qualified" />
   390→        </div>
   391→      )}
   392→
   393→      {/* Kanban */}
   394→      {!error && queue !== null && roster !== null && mode === "kanban" && (
   395→        <div className="flex snap-x snap-proximity gap-4 overflow-x-auto scroll-thin pb-2 lg:grid lg:grid-cols-5 lg:overflow-visible lg:pb-0">
   396→          {/* Applicants column (roster) */}
   397→          <KanbanColumn label="Applicants" icon={STAGE_ICON["Applicants"]} tone={STAGE_TONE["Applicants"]} count={rosterCards.length}>
   398→            {rosterCards.length === 0 ? (
   399→              <EmptyState icon={Users} tone="slate" title="No applicants" description="Registered applicants waiting to apply appear here." compact />
   400→            ) : (
   401→              rosterCards.map((r) => (
   402→                <button
   403→                  key={r.id}
   404→                  type="button"
   405→                  className={cardCls + " focus-ring w-full cursor-pointer p-4 text-left"}
   406→                  onClick={() => navigate("candidate", { id: String(r.id) })}
   407→                >
   408→                  <div className="flex items-start gap-2.5">
   409→                    <MonogramAvatar name={fullName(r)} />
   410→                    <div className="min-w-0 flex-1">
   411→                      <p className="min-w-0 truncate text-sm font-medium text-ink">{fullName(r)}</p>
   412→                      <p className="mt-0.5 truncate text-xs text-stone">{r.emailAddress || "No email on record"}</p>
   413→                    </div>
   414→                  </div>
   415→                  <div className="mt-2.5">
   416→                    <StatusPill
   417→                      status={r.isProfileComplete ? "Profile complete" : "Profile incomplete"}
   418→                      variant={variantForCompletion(r.isProfileComplete)}
   419→                    />
   420→                  </div>
   421→                </button>
   422→              ))
   423→            )}
   424→          </KanbanColumn>
   425→
   426→          {/* Pipeline columns */}
   427→          {PIPELINE_STAGES.map((stage) => {
   428→            const rows = kanbanColumns[stage];
   429→            return (
   430→              <KanbanColumn key={stage} label={stage} icon={STAGE_ICON[stage]} tone={STAGE_TONE[stage]} count={rows.length}>
   431→                {rows.length === 0 ? (
   432→                  <EmptyState icon={STAGE_ICON[stage]} tone={STAGE_TONE[stage]} title="No applications" description="Nothing in this stage right now." compact />
   433→                ) : (
   434→                  rows.map((row) => {
   435→                    const position = row.job.position?.positionTitle || row.job.title;
   436→                    const place = row.job.position?.placeOfAssignment ?? null;
   437→                    return (
   438→                      <div
   439→                        key={row.id}
   440→                        role="button"
   441→                        tabIndex={0}
   442→                        className={cardCls + " focus-ring cursor-pointer p-4"}
   443→                        onClick={() => setSelectedId(row.id)}
   444→                        onKeyDown={(e) => e.key === "Enter" && setSelectedId(row.id)}
   445→                        aria-label={`Review application from ${fullName(row.applicant)}`}
   446→                      >
   447→                        <div className="flex items-start gap-2.5">
   448→                          <MonogramAvatar name={fullName(row.applicant)} />
   449→                          <p className="min-w-0 flex-1 truncate text-sm font-medium text-ink">{fullName(row.applicant)}</p>
   450→                          <button
   451→                            type="button"
   452→                            className={iconBtn + " -mr-1.5 -mt-1 h-8 w-8 shrink-0"}
   453→                            aria-label={`Open ${fullName(row.applicant)}'s profile`}
   454→                            onClick={(e) => {
   455→                              e.stopPropagation();
   456→                              navigate("candidate", { id: String(row.applicantId) });
   457→                            }}
   458→                          >
   459→                            <UserRound className="h-4 w-4" aria-hidden />
   460→                          </button>
   461→                        </div>
   462→                        <p className="num mt-0.5 whitespace-nowrap text-xs text-pebble">Applied {formatDate(row.dateApplied)}</p>
   463→                        <p className="mt-1 truncate text-xs text-stone">
   464→                          {[humanize(position), place].filter(Boolean).join(" · ")}
   465→                        </p>
   466→                        <div className="mt-2.5 flex flex-wrap items-center gap-1.5">
   467→                          <StatusPill status={row.status} />
   468→                          <VerdictPill verdict={row.match?.verdict} />
   469→                        </div>
   470→                      </div>
   471→                    );
   472→                  })
   473→                )}
   474→              </KanbanColumn>
   475→            );
   476→          })}
   477→        </div>
   478→      )}
   479→
   480→      {/* List */}
   481→      {!error && queue !== null && roster !== null && mode === "list" && (
   482→        <div className="space-y-3">
   483→          {listRows.length === 0 ? (
   484→            <div className="dlg-card py-6">
   485→              <EmptyState
   486→                icon={Inbox}
   487→                title="No applications in the queue"
   488→                description={listStage === "All" ? "Applications appear here once candidates apply." : `No applications in the ${listStage} stage yet.`}
   489→              />
   490→            </div>
   491→          ) : (
   492→            <div className="dlg-card overflow-hidden">
   493→              <div className="overflow-x-auto scroll-thin">
   494→                <table className="w-full min-w-[760px] border-collapse text-sm">
   495→                  <thead>
   496→                    <tr className="border-b border-border text-left">
   497→                      <th scope="col" className="px-5 py-3 text-[11px] font-medium uppercase tracking-[0.08em] text-stone">Candidate</th>
   498→                      <th scope="col" className="px-4 py-3 text-[11px] font-medium uppercase tracking-[0.08em] text-stone">Position</th>
   499→                      <th scope="col" className="px-4 py-3 text-[11px] font-medium uppercase tracking-[0.08em] text-stone">Applied</th>
   500→                      <th scope="col" className="px-4 py-3 text-[11px] font-medium uppercase tracking-[0.08em] text-stone">Status</th>
   501→                      <th scope="col" className="px-4 py-3 text-[11px] font-medium uppercase tracking-[0.08em] text-stone">Verdict</th>
   502→                      <th scope="col" className="px-4 py-3"><span className="sr-only">Open</span></th>

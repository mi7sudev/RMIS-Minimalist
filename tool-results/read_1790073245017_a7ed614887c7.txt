   127→  if (v === null || v === undefined) return "";
   128→  if (typeof v === "boolean") return v ? "Yes" : "No";
   129→  const s = String(v).trim();
   130→  return s;

   146→function snapshotDisplayDate(v: unknown): string {
   147→  const s = str(v);
   148→  if (!s) return "";

   150→  return Number.isNaN(d.getTime()) ? s : formatDate(d);

   155→function verdictMeta(verdict: string): { chip: string; cls: string; meaning: string } {
   156→  switch (verdict) {
   157→    case "ALL_MET":
   158→      return { chip: "Qualified", cls: "bg-ink text-white", meaning: "Meets the minimum requirements" };
   159→    case "PARTIAL":
   160→      return { chip: "Partial", cls: "bg-fog text-ink", meaning: "Partially meets the minimum requirements" };
   161→    case "NONE_MET":
   162→      return { chip: "Not qualified", cls: "bg-dusty-rose/15 text-dusty-rose", meaning: "Does not meet the minimum requirements" };
   163→    case "NEEDS_REVIEW":
   164→      return { chip: "Verify", cls: "bg-fog text-ink", meaning: "Needs manual verification" };
   165→    case "NO_REQUIREMENTS":
   166→      return { chip: "No reqs", cls: "bg-fog text-stone", meaning: "No published requirements" };
   167→    default:
   168→      return { chip: verdict, cls: "bg-fog text-stone", meaning: "" };
   169→  }
   170→}

   364→function stateBanner(stage: string): { text: string; cls: string } {

   366→  if (stage === "Rejected") return { text: "Applicant was not shortlisted.", cls: "text-dusty-rose" };

   373→function NoticeFormDialog({
   374→  applicationId,
   375→  kind,
   376→  onClose,
   377→  onSent,
   378→}: {
   379→  applicationId: number;
   380→  kind: "interview" | "skills_exam";
   381→  onClose: () => void;
   382→  onSent: () => void;
   383→}) {
   384→  const [examType, setExamType] = useState("");
   385→  const [date, setDate] = useState("");
   386→  const [time, setTime] = useState("");
   387→  const [venue, setVenue] = useState("");
   388→  const [contact, setContact] = useState("");
   389→  const [notes, setNotes] = useState("");
   390→  const [busy, setBusy] = useState(false);
   391→
   392→  const ready = date.trim() !== "" && time.trim() !== "" && venue.trim() !== "";
   393→
   394→  const submit = async () => {
   395→    setBusy(true);
   396→    try {
   397→      const res = await apiFetch<NoticeSendResult>(`/api/evaluator/applications/${applicationId}/notice`, {
   398→        method: "POST",
   399→        body: {
   400→          type: kind,
   401→          date: date || undefined,
   402→          time: time || undefined,
   403→          venue: venue || undefined,
   404→          contact: contact || undefined,
   405→          notes: notes || undefined,
   406→          examType: kind === "skills_exam" ? examType || undefined : undefined,
   407→        },
   408→      });
   409→      const label = kind === "interview" ? "Interview invitation" : "Skills-exam notice";
   410→      toast.success(`${label} sent`, {
   411→        description: `Email: ${res.email?.status ?? "unknown"} · SMS: ${res.sms?.status ?? "unknown"}`,
   412→      });
   413→      onSent();
   414→      onClose();
   415→    } catch (e) {
   416→      toast.error(e instanceof Error ? e.message : "Failed to send the notice");
   417→    } finally {
   418→      setBusy(false);
   419→    }
   420→  };
   421→
   422→  return (
   423→    <Dialog open onOpenChange={(v) => !v && onClose()}>
   424→      <DialogContent className="sm:max-w-md">
   425→        <DialogHeader>
   426→          <DialogTitle className="font-display text-xl">
   427→            {kind === "interview" ? "Send interview invitation" : "Send skills-exam notice"}
   428→          </DialogTitle>
   429→          <DialogDescription>
   430→            The notice fans out as an email plus a companion SMS to the applicant.
   431→          </DialogDescription>
   432→        </DialogHeader>
   433→        <div className="space-y-4">
   434→          {kind === "skills_exam" && (
   435→            <div className="space-y-1.5">
   436→              <Label htmlFor="notice-exam-type" className="text-xs text-stone">
   437→                Exam type
   438→              </Label>
   439→              <Input
   440→                id="notice-exam-type"
   441→                className="dlg-input"
   442→                value={examType}
   443→                onChange={(e) => setExamType(e.target.value)}
   444→                placeholder="e.g. Encoding / trade test"
   445→              />
   446→            </div>
   447→          )}
   448→          <div className="grid grid-cols-2 gap-3">
   449→            <div className="space-y-1.5">
   450→              <Label htmlFor="notice-date" className="text-xs text-stone">
   451→                Date *
   452→              </Label>
   453→              <Input id="notice-date" type="date" className="dlg-input" value={date} onChange={(e) => setDate(e.target.value)} />
   454→            </div>
   455→            <div className="space-y-1.5">
   456→              <Label htmlFor="notice-time" className="text-xs text-stone">
   457→                Time *
   458→              </Label>
   459→              <Input id="notice-time" type="time" className="dlg-input" value={time} onChange={(e) => setTime(e.target.value)} />
   460→            </div>
   461→          </div>
   462→          <div className="space-y-1.5">
   463→            <Label htmlFor="notice-venue" className="text-xs text-stone">
   464→              Venue *
   465→            </Label>
   466→            <Input
   467→              id="notice-venue"
   468→              className="dlg-input"
   469→              value={venue}
   470→              onChange={(e) => setVenue(e.target.value)}
   471→              placeholder="Building / room"
   472→            />
   473→          </div>
   474→          <div className="space-y-1.5">
   475→            <Label htmlFor="notice-contact" className="text-xs text-stone">
   476→              HR contact
   477→            </Label>
   478→            <Input
   479→              id="notice-contact"
   480→              className="dlg-input"
   481→              value={contact}
   482→              onChange={(e) => setContact(e.target.value)}
   483→              placeholder="Name / number for questions"
   484→            />
   485→          </div>
   486→          <div className="space-y-1.5">
   487→            <Label htmlFor="notice-notes" className="text-xs text-stone">
   488→              Notes
   489→            </Label>
   490→            <Textarea id="notice-notes" className="dlg-input min-h-[80px]" value={notes} onChange={(e) => setNotes(e.target.value)} />
   491→          </div>
   492→        </div>
   493→        <DialogFooter>
   494→          <button type="button" className={ghostBtn} onClick={onClose}>
   495→            Cancel
   496→          </button>
   497→          <button type="button" className={ctaBtn} disabled={!ready || busy} onClick={() => void submit()}>
   498→            <Send className="h-4 w-4" aria-hidden />
   499→            {busy ? "Sending…" : "Send notice"}
   500→          </button>
   501→        </DialogFooter>
   502→      </DialogContent>
   503→    </Dialog>
   504→  );
   505→}

   509→function parseAttachments(json: string | null): { name: string; bytes: number }[] {
   510→  if (!json) return [];
   511→  try {
   512→    const parsed: unknown = JSON.parse(json);
   513→    return Array.isArray(parsed) ? (parsed as { name: string; bytes: number }[]) : [];
   514→  } catch {
   515→    return [];
   516→  }
   517→}

   519→function DirectEmailCard({
   520→  applicationId,
   521→  emails,
   522→  onSent,
   523→}: {
   524→  applicationId: number;
   525→  emails: DirectEmailRow[] | null;
   526→  onSent: () => void;
   527→}) {
   528→  const [subject, setSubject] = useState("");
   529→  const [message, setMessage] = useState("");
   530→  const [files, setFiles] = useState<File[]>([]);
   531→  const [busy, setBusy] = useState(false);
   532→  const fileRef = useRef<HTMLInputElement>(null);
   533→
   534→  const pickFiles = (list: FileList | null) => {
   535→    if (!list) return;
   536→    const next = Array.from(list);
   537→    if (next.length + files.length > 3) {
   538→      toast.error("A maximum of 3 attachments is allowed");
   539→      return;
   540→    }
   541→    if (next.some((f) => f.size > 5 * 1024 * 1024)) {
   542→      toast.error("Each attachment must be 5 MB or smaller");
   543→      return;
   544→    }
   545→    setFiles((prev) => [...prev, ...next].slice(0, 3));
   546→    if (fileRef.current) fileRef.current.value = "";
   547→  };
   548→
   549→  const send = async () => {
   550→    if (!message.trim()) return;
   551→    setBusy(true);
   552→    try {
   553→      const fd = new FormData();
   554→      fd.set("subject", subject.trim());
   555→      fd.set("message", message.trim());
   556→      for (const f of files) fd.append("attachments", f);
   557→      const res = await apiFetch<{ email: { status: string; error?: string | null }; attachments: number }>(
   558→        `/api/evaluator/applications/${applicationId}/email`,
   559→        { method: "POST", formData: fd }
   560→      );
   561→      if (res.email?.status === "failed") {
   562→        toast.error("Email delivery failed", { description: res.email.error ?? undefined });
   563→      } else {
   564→        toast.success(`Direct email ${res.email?.status}`, {
   565→          description: res.attachments ? `${res.attachments} attachment(s) included.` : undefined,
   566→        });
   567→      }
   568→      setSubject("");
   569→      setMessage("");
   570→      setFiles([]);
   571→      onSent();
   572→    } catch (e) {
   573→      toast.error(e instanceof Error ? e.message : "Failed to send the email");
   574→    } finally {
   575→      setBusy(false);
   576→    }
   577→  };
   578→
   579→  return (
   580→    <div className="dlg-card p-4">
   581→      <div className="flex items-center gap-2">
   582→        <Mail className="h-4 w-4 text-stone" aria-hidden />
   583→        <h3 className="font-display text-lg text-ink">Direct email</h3>
   584→      </div>
   585→      <p className="text-xs text-stone mt-1">Recipient is resolved from the applicant record — attachments ≤ 3 files × 5 MB.</p>
   586→      <div className="space-y-3 mt-3">
   587→        <Input
   588→          className="dlg-input"
   589→          placeholder="Subject"
   590→          value={subject}
   591→          maxLength={200}
   592→          onChange={(e) => setSubject(e.target.value)}
   593→          aria-label="Email subject"
   594→        />
   595→        <Textarea
   596→          className="dlg-input min-h-[100px]"
   597→          placeholder="Message *"
   598→          value={message}
   599→          maxLength={5000}
   600→          onChange={(e) => setMessage(e.target.value)}
   601→          aria-label="Email message"
   602→        />
   603→        <div>
   604→          <label className="inline-flex min-h-[44px] cursor-pointer items-center gap-2 rounded-full border border-[#ececec] bg-white px-4 text-sm hover:bg-fog">
   605→            <Paperclip className="h-4 w-4" aria-hidden />
   606→            Attach files
   607→            <input
   608→              ref={fileRef}
   609→              type="file"
   610→              multiple
   611→              className="hidden"
   612→              onChange={(e) => pickFiles(e.target.files)}
   613→              aria-label="Attach files"
   614→            />
   615→          </label>
   616→          {files.length > 0 && (
   617→            <ul className="mt-2 space-y-1">
   618→              {files.map((f) => (
   619→                <li key={f.name} className="flex items-center justify-between gap-2 text-xs text-stone">
   620→                  <span className="truncate">{f.name}</span>
   621→                  <button
   622→                    type="button"
   623→                    className="text-dusty-rose shrink-0"
   624→                    onClick={() => setFiles((prev) => prev.filter((x) => x !== f))}
   625→                  >
   626→                    Remove
   627→                  </button>
   628→                </li>
   629→              ))}
   630→            </ul>
   631→          )}
   632→        </div>
   633→        <button type="button" className={ctaBtn + " w-full"} disabled={!message.trim() || busy} onClick={() => void send()}>
   634→          <Send className="h-4 w-4" aria-hidden />
   635→          {busy ? "Sending…" : "Send email"}
   636→        </button>
   637→      </div>
   638→      <div className="mt-4 border-t border-[#ececec] pt-3">
   639→        <p className="text-xs font-medium text-stone">History</p>
   640→        <div className="mt-2 max-h-48 overflow-y-auto scroll-thin space-y-1.5">
   641→          {emails === null ? (
   642→            <p className="text-xs text-pebble">Loading…</p>
   643→          ) : emails.length === 0 ? (
   644→            <p className="text-xs text-pebble">No direct emails sent yet.</p>
   645→          ) : (
   646→            emails.map((m) => {
   647→              const atts = parseAttachments(m.attachments);
   648→              return (
   649→                <div key={m.id} className="flex items-center justify-between gap-2 text-xs">
   650→                  <span className="truncate text-ink">{m.subject || "(no subject)"}</span>
   651→                  <span className="shrink-0 flex items-center gap-2 text-pebble">
   652→                    {atts.length > 0 && <span>{atts.length} att.</span>}
   653→                    <span>{timeAgo(m.createdAt)}</span>
   654→                    <span className={`rounded-full px-2 py-0.5 ${m.status === "sent" || m.status === "mock" ? "bg-fog text-ink" : "bg-dusty-rose/15 text-dusty-rose"}`}>
   655→                      {m.status}
   656→                    </span>
   657→                  </span>
   658→                </div>
   659→              );
   660→            })
   661→          )}
   662→        </div>
   663→      </div>
   664→    </div>
   665→  );
   666→}

  1025→                Send Skills-Exam Notice
  1026→              </button>
  1027→            </div>
  1028→          )}
  1029→          {stage === "Rejected" && (
  1030→            <button
  1031→              type="button"
  1032→              className={ghostBtn + " w-full text-dusty-rose"}
  1033→              disabled={regretAlreadySent || regretBusy}
  1034→              onClick={() => setRegretOpen(true)}
  1035→            >
  1036→              {regretAlreadySent ? "Regret letter sent" : regretBusy ? "Sending…" : "Send Regret Letter"}

  1039→        </div>
  1040→
  1041→        <DirectEmailCard applicationId={applicationId} emails={emails} onSent={refetch} />

  1044→      {/* Decision confirm */}
  1045→      <AlertDialog open={confirmAction !== null} onOpenChange={(v) => !v && setConfirmAction(null)}>
  1046→        <AlertDialogContent>
  1047→          <AlertDialogHeader>
  1048→            <AlertDialogTitle>{confirmAction?.title}</AlertDialogTitle>
  1049→            <AlertDialogDescription>{confirmAction?.description}</AlertDialogDescription>

  1052→            <AlertDialogCancel>Cancel</AlertDialogCancel>

  1054→              className={confirmAction?.destructive ? "bg-dusty-rose/10 text-dusty-rose hover:bg-dusty-rose/20" : ""}
  1055→              onClick={(e) => {
  1056→                e.preventDefault();
  1057→                if (confirmAction) void decide(confirmAction.status);
  1058→              }}
  1059→            >
  1060→              {busy ? "Working…" : "Confirm"}
  1061→            </AlertDialogAction>
  1062→          </AlertDialogFooter>
  1063→        </AlertDialogContent>
  1064→      </AlertDialog>
  1065→
  1066→      {/* Regret confirm */}
  1067→      <AlertDialog open={regretOpen} onOpenChange={setRegretOpen}>
  1068→        <AlertDialogContent>
  1069→          <AlertDialogHeader>

  1071→            <AlertDialogDescription>

  1073→                ? `A formal regret letter will be emailed to ${email}.`
  1074→                : "No email on record — the send will be logged as skipped and HR will contact the applicant directly."}
  1075→            </AlertDialogDescription>
  1076→          </AlertDialogHeader>
  1077→          <AlertDialogFooter>
  1078→            <AlertDialogCancel>Cancel</AlertDialogCancel>
  1079→            <AlertDialogAction
  1080→              className="bg-dusty-rose/10 text-dusty-rose hover:bg-dusty-rose/20"
  1081→              onClick={(e) => {
  1082→                e.preventDefault();
  1083→                void sendRegret();
  1084→              }}
  1085→            >
  1086→              Send regret letter
  1087→            </AlertDialogAction>
  1088→          </AlertDialogFooter>

  1091→
  1092→      {/* Notice form */}
  1093→      {noticeKind && (
  1094→        <NoticeFormDialog
  1095→          applicationId={applicationId}
  1096→          kind={noticeKind}
  1097→          onClose={() => setNoticeKind(null)}
  1098→          onSent={refetch}
  1099→        />
  1100→      )}
  1101→    </div>
  1102→  );
  1103→}

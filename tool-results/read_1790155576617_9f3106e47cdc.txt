    58→function notRequired(v: string | null | undefined): boolean {
    59→  if (!v) return true;
    60→  const s = v.trim().toLowerCase();
    61→  return s === "" || s === "n/a" || s === "na" || s === "none" || s === "none required" || s === "not required";
    62→}

    64→/** Applicant-facing stage label → functional status variant (§2 status system). */
    65→const STAGE_VARIANT: Record<string, "ok" | "warn" | "bad" | "info"> = {

    67→  "In Review": "warn",
    68→  Shortlisted: "ok",
    69→  "Not Selected": "bad",

    76→/** §7.3 Next Step hint paragraph per pipeline stage. */
    77→function nextStepHint(stage: string): string {
    78→  if (stage === "Rejected")
    79→    return "Your application was not shortlisted. You may apply for other open positions.";
    80→  if (stage === "Shortlisted")
    81→    return "A notice has been sent to your registered email. HR will contact you for the next steps.";
    82→  if (stage === "Under Review") return "HR is currently evaluating your application.";
    83→  return "Your application has been received and is awaiting evaluation.";
    84→}

    86→type CheckpointState = "done" | "current" | "pending" | "failed";

    88→/** Status-colored checkpoint dot resting in a small white ring with a soft
    89→ *  glow (glow ships in the .stage-dot primitives). The current stage gets a
    90→ *  pulsing halo — motion-safe only. */
    91→function CheckpointDot({ state }: { state: CheckpointState }) {
    92→  const dot =
    93→    state === "done"
    94→      ? "dot-ok"
    95→      : state === "current"
    96→        ? "dot-warn"
    97→        : state === "failed"
    98→          ? "dot-bad"
    99→          : "dot-neutral";
   100→  return (
   101→    <span
   102→      className={cn(
   103→        "inline-flex h-[18px] w-[18px] shrink-0 items-center justify-center rounded-full bg-white shadow-[0_1px_3px_rgba(24,24,37,0.16),0_0_0_1px_rgba(24,24,37,0.05)]",
   104→        state === "current" && "motion-safe:animate-pulse"
   105→      )}
   106→    >
   107→      <span className={cn("stage-dot", dot)} />
   108→    </span>
   109→  );
   110→}

   112→/** 3-checkpoint tracking timeline: Submitted → Review → Decision (§7.3). */
   113→function JourneyTimeline({ status }: { status: string }) {
   114→  const stage = stageForStatus(status);
   115→  const decided = stage === "Shortlisted" || stage === "Rejected";
   116→
   117→  const steps: { label: string; short?: string; state: CheckpointState }[] = [
   118→    { label: "Submitted", state: "done" },
   119→    {
   120→      label: "Review",
   121→      state: stage === "Under Review" ? "current" : decided ? "done" : "pending",
   122→    },
   123→    {
   124→      label:
   125→        stage === "Shortlisted" ? "Shortlisted" : stage === "Rejected" ? "Not Shortlisted" : "Awaiting the shortlist decision",
   126→      // Responsive copy swap (3-d, presentation-only): the full label wraps to 4+ lines
   127→      // inside a 3-up timeline at 320px; "Shortlisted"/"Not Shortlisted" stay verbatim.
   128→      short: stage === "Shortlisted" || stage === "Rejected" ? undefined : "Awaiting decision",
   129→      state: stage === "Shortlisted" ? "done" : stage === "Rejected" ? "failed" : "pending",
   130→    },
   131→  ];
   132→
   133→  return (
   134→    <div className="mt-3 grid grid-cols-3 gap-1">
   135→      {steps.map((s, i) => {
   136→        const reached = s.state !== "pending";
   137→        return (
   138→          <div key={`${s.label}-${i}`} className="flex flex-col items-center gap-1.5 text-center">
   139→            <div className="flex w-full items-center">
   140→              {/* Traversed connectors pick up a subtle ok→warn gradient tint. */}
   141→              <div
   142→                className={cn(
   143→                  "h-px flex-1",
   144→                  i === 0 ? "bg-transparent" : reached ? "bg-gradient-to-r from-[var(--ok)]/60 to-[var(--warn)]/45" : "bg-divider"
   145→                )}
   146→              />
   147→              <CheckpointDot state={s.state} />
   148→              <div
   149→                className={cn(
   150→                  "h-px flex-1",
   151→                  i === steps.length - 1
   152→                    ? "bg-transparent"
   153→                    : steps[i + 1].state !== "pending"
   154→                      ? "bg-gradient-to-r from-[var(--ok)]/60 to-[var(--warn)]/45"
   155→                      : "bg-divider"
   156→                )}
   157→              />
   158→            </div>
   159→            <span className={cn("text-[10px] leading-tight", s.state === "current" ? "font-medium text-ink" : "text-stone")}>
   160→              {s.short ? (
   161→                <>
   162→                  <span className="sm:hidden">{s.short}</span>
   163→                  <span className="hidden sm:inline">{s.label}</span>
   164→                </>
   165→              ) : (
   166→                s.label
   167→              )}
   168→            </span>
   169→          </div>
   170→        );
   171→      })}
   172→    </div>
   173→  );
   174→}

   240→  if (!position) return null;

   242→    { label: "Education", value: position.cscEducation },

   265→function ApplicationDetailModal({
   266→  application,
   267→  onClose,
   268→  onCancelled,
   269→}: {
   270→  application: ApplicationWire;
   271→  onClose: () => void;
   272→  onCancelled: () => void;
   273→}) {
   274→  const [confirmOpen, setConfirmOpen] = useState(false);
   275→  const [cancelling, setCancelling] = useState(false);
   276→  const job = application.job ?? null;
   277→  const position = job?.position ?? null;
   278→  const label = currentStageLabel(application.status);
   279→  const cancellable = label === "Submitted";
   280→  const dl = deadlineState(job?.deadlineDate ?? null);
   281→
   282→  const cancelApplication = async () => {
   283→    setCancelling(true);
   284→    try {
   285→      await apiFetch(`/api/applications/${application.id}`, { method: "DELETE" });
   286→      toast.success("Application cancelled", { description: "Your application has been withdrawn." });
   287→      setConfirmOpen(false);
   288→      onClose();
   289→      onCancelled();
   290→    } catch (e) {
   291→      // Progressed applications: surface the server's guidance (e.g. "contact HR") as-is.
   292→      toast.error(e instanceof Error ? e.message : "Unable to cancel this application");
   293→      setConfirmOpen(false);
   294→    } finally {
   295→      setCancelling(false);
   296→    }
   297→  };
   298→
   299→  const facts: [string, string][] = [
   300→    ["Place of Assignment", position?.placeOfAssignment ?? "—"],
   301→    ["Employment Type", job?.positionType ? humanize(job.positionType) : position?.positionType ? humanize(position.positionType) : "—"],
   302→    ["Vacancies", job ? String(job.numberOfVacancy) : "—"],
   303→    ["Monthly Salary", formatCurrency(position?.salaryAmount ?? null)],
   304→    ["Salary Grade", position?.salaryGrade ? `SG-${position.salaryGrade}${position.salaryStep ? ` Step ${position.salaryStep}` : ""}` : "—"],
   305→    ["Item No.", position?.itemNumber ?? "—"],
   306→    ["Published", formatDate(job?.publishDate ?? null)],
   307→    ["Deadline", job ? dl.label : "—"],
   308→  ];
   309→
   310→  return (
   311→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   312→      <DialogContent className="dlg-card-plain max-h-[90vh] w-full overflow-y-auto scroll-thin p-0 shadow-e4 sm:max-w-3xl">
   313→        <div className="p-6">
   314→          <DialogHeader className="space-y-2 text-left">
   315→            <div className="flex flex-wrap items-center gap-2">
   316→              <StatusPill status={label} variant={stageVariant(label)} />
   317→              <span className="num bg-fog px-3 py-1 text-xs font-medium text-graphite">
   318→                Application #{application.id}
   319→              </span>
   320→            </div>
   321→            <DialogTitle className="font-display text-2xl leading-tight text-ink">
   322→              {job?.title ?? "Application"}
   323→            </DialogTitle>
   324→          </DialogHeader>
   325→
   326→          {/* Successfully-applied strip (§7.7) */}
   327→          <div className="mt-4 flex items-center gap-3 rounded-none bg-fog p-3">
   328→            <IconChip icon={BadgeCheck} tone="emerald" size={36} iconSize={17} />
   329→            <div>
   330→              <p className="text-sm font-medium text-ink">Successfully Applied</p>
   331→              <p className="text-xs text-stone">
   332→                Applied <span className="num">{formatDate(application.dateApplied)}</span>
   333→              </p>
   334→            </div>
   335→          </div>
   336→
   337→          {/* Vitals summary grid */}
   338→          <div className="mt-5 grid grid-cols-2 gap-2 sm:grid-cols-4">
   339→            {facts.map(([k, v]) => (
   340→              <div key={k} className="rounded-none bg-fog p-3">
   341→                <p className="text-[11px] text-pebble">{k}</p>
   342→                <p className="num mt-0.5 truncate text-sm text-ink" title={v}>
   343→                  {v}
   344→                </p>
   345→              </div>
   346→            ))}
   347→          </div>
   348→
   349→          <div className="mt-6 space-y-6">
   350→            <MqrLedger position={position} />
   351→            <RichSection title="Brief Description" text={job?.briefDescription} html={job?.briefDescriptionHtml} />
   352→            <RichSection title="Duties & Responsibilities" text={job?.dutiesResponsibilities} html={job?.dutiesHtml} />
   353→            <RichSection title="Compensation Package" text={job?.compensationPackage} html={job?.compensationHtml} />
   354→            <RichSection title="Other Qualifications" text={job?.otherQualifications} html={job?.otherQualificationsHtml} />
   355→          </div>
   356→
   357→          <div className="mt-8 flex flex-col gap-3 border-t border-border pt-5 sm:flex-row sm:justify-end">
   358→            {cancellable && (
   359→              <button
   360→                type="button"
   361→                onClick={() => setConfirmOpen(true)}
   362→                disabled={cancelling}
   363→                className="min-h-[44px] border border-[var(--bad)]/30 bg-white px-6 py-2.5 text-sm font-medium text-[var(--bad)] transition-colors hover:bg-fog disabled:opacity-50"
   364→              >
   365→                Cancel Application
   366→              </button>
   367→            )}
   368→            <button type="button" onClick={onClose} className="dlg-ghost min-h-[44px] px-6 py-2.5 text-sm">
   369→              Close
   370→            </button>
   371→          </div>
   372→        </div>
   373→
   374→        {/* Nested destructive confirm (§7.7) */}
   375→        <AlertDialog open={confirmOpen} onOpenChange={setConfirmOpen}>
   376→          <AlertDialogContent className="dlg-card-plain">
   377→            <AlertDialogHeader>
   378→              <AlertDialogTitle className="font-display text-xl text-ink">Cancel this application?</AlertDialogTitle>
   379→              <AlertDialogDescription className="text-sm leading-relaxed text-stone">
   380→                Your application for {job?.title ?? "this position"} will be withdrawn. You may apply again while the
   381→                position is still open. Once review has started, applications can no longer be cancelled.
   382→              </AlertDialogDescription>
   383→            </AlertDialogHeader>
   384→            <AlertDialogFooter>
   385→              <AlertDialogCancel className="dlg-ghost min-h-[44px] px-5 py-2.5 text-sm">Keep it</AlertDialogCancel>
   386→              <AlertDialogAction
   387→                onClick={(e) => {
   388→                  e.preventDefault();
   389→                  void cancelApplication();
   390→                }}
   391→                disabled={cancelling}
   392→                className="min-h-[44px] border border-[var(--bad)]/30 bg-white px-5 py-2.5 text-sm font-medium text-[var(--bad)] transition-colors hover:bg-fog disabled:opacity-50"
   393→              >
   394→                {cancelling ? "Cancelling…" : "Cancel Application"}
   395→              </AlertDialogAction>
   396→            </AlertDialogFooter>
   397→          </AlertDialogContent>
   398→        </AlertDialog>
   399→      </DialogContent>
   400→    </Dialog>
   401→  );
   402→}

   435→        type="button"

   437→        className="dlg-ghost inline-flex min-h-[44px] w-fit items-center gap-2 px-5 py-2 text-sm"
   438→      >
   439→        View details <ArrowRight className="h-3.5 w-3.5" />

   495→    }
   496→  }, []);
   497→
   498→  useEffect(() => {
   499→    void reload(false);
   500→    const interval = setInterval(() => {
   501→      if (!document.hidden) void reload(true);
   502→    }, 15_000);
   503→    const onFocus = () => void reload(true);
   504→    const onVisible = () => {
   505→      if (!document.hidden) void reload(true);
   506→    };
   507→    window.addEventListener("focus", onFocus);
   508→    document.addEventListener("visibilitychange", onVisible);
   509→    return () => {
   510→      clearInterval(interval);
   511→      window.removeEventListener("focus", onFocus);
   512→      document.removeEventListener("visibilitychange", onVisible);
   513→    };
   514→  }, [reload]);
   515→
   516→  const loading = apps === null && jobs === null;
   517→  const today = startOfToday();
   518→  const openJobs = (jobs ?? [])
   519→    .filter((j) => !j.deadlineDate || new Date(j.deadlineDate) >= today)
   520→    .sort((a, b) => Date.parse(b.publishedAt) - Date.parse(a.publishedAt))
   521→    .slice(0, 6);
   522→  const sortedApps = (apps ?? []).sort((a, b) => Date.parse(b.dateApplied) - Date.parse(a.dateApplied));
   523→  const profileComplete = user?.applicant?.isProfileComplete ?? false;
   524→  const inProgress = sortedApps.filter((a) => {
   525→    const s = stageForStatus(a.status);

   527→  }).length;
   528→
   529→  return (

   531→      {/* Page header (§3 PageHeader pattern) */}

   534→        description="Track your applications and discover the latest openings across DOST-MIRDC."
   535→      />
   536→
   537→      {/* Complete Your Profile banner while incomplete */}
   538→      {!profileComplete && (
   539→        <div className="dlg-card mt-6 flex flex-col gap-3 border border-dashed border-divider p-4 sm:flex-row sm:items-center sm:justify-between">
   540→          <div className="flex items-start gap-3">
   541→            <IconChip icon={AlertCircle} tone="gold" size={36} iconSize={17} className="mt-0.5" />
   542→            <div>
   543→              <p className="font-display text-lg leading-tight text-ink">Complete Your Profile</p>

   545→                Finish the 7-part profile to unlock applications — upload your PDS and we&apos;ll fill it for you.
   546→              </p>
   547→            </div>
   548→          </div>
   549→          <button
   550→            type="button"
   551→            onClick={() => navigate("profile")}
   552→            className="dlg-cta min-h-[44px] w-fit shrink-0 px-6 py-2.5 text-sm"
   553→          >
   554→            Complete profile
   555→          </button>
   556→        </div>
   557→      )}
   558→
   559→      {/* Error state with retry */}
   560→      {error && apps === null && (
   561→        <div className="dlg-card mt-6 flex flex-col items-center gap-3 p-8 text-center">
   562→          <IconChip icon={AlertCircle} tone="rose" size={44} />
   563→          <p className="text-sm text-stone">{error}</p>
   564→          <button type="button" onClick={() => void reload(false)} className="dlg-ghost min-h-[44px] px-6 py-2.5 text-sm">
   565→            <RefreshCw className="mr-2 inline h-3.5 w-3.5" /> Retry
   566→          </button>
   567→        </div>
   568→      )}
   569→
   570→      {!loading && (
   571→        /* KPI-style summary — counts derived from the same wire data as below. */
   572→        <div className="mt-6 grid animate-in fade-in slide-in-from-bottom-2 grid-cols-1 gap-4 duration-300 sm:grid-cols-3">
   573→          <SummaryCard label="Open Positions" value={openJobs.length} hint="published now" icon={Briefcase} tone="amber" />
   574→          <SummaryCard label="My Applications" value={sortedApps.length} hint="total submitted" icon={Inbox} tone="plum" />
   575→          <SummaryCard label="In Progress" value={inProgress} hint="awaiting decision" icon={ClipboardList} tone="emerald" />
   576→        </div>
   577→      )}
   578→
   579→      {loading ? (
   580→        /* First-load skeletons — never stale data, never a blank pane. */
   581→        <div className="mt-6 space-y-10">
   582→          <SkeletonKpis count={3} className="grid-cols-1 sm:grid-cols-3 lg:grid-cols-3" />
   583→          <div className="grid gap-10 lg:grid-cols-[minmax(0,1fr)_minmax(320px,400px)] lg:items-start lg:gap-10 xl:gap-14">
   584→            <SkeletonKpis count={4} className="grid-cols-1 sm:grid-cols-2 lg:grid-cols-1 xl:grid-cols-2 [&>div]:h-44" />
   585→            <div className="space-y-4">
   586→              <SkeletonRows rows={1} rowClassName="h-44" />
   587→              <SkeletonRows rows={1} rowClassName="h-44" />
   588→            </div>
   589→          </div>
   590→        </div>
   591→      ) : (
   592→        <div className="mt-6 grid animate-in fade-in slide-in-from-bottom-2 gap-10 duration-300 lg:grid-cols-[minmax(0,1fr)_minmax(320px,400px)] lg:items-start lg:gap-10 xl:gap-14">
   593→          {/* LEFT — Open positions */}
   594→          <section className="lg:col-start-1 lg:row-start-1">

   601→                onClick={() => navigate("jobs")}
   602→                className="focus-ring inline-flex min-h-[44px] items-center gap-1.5 text-sm text-stone underline-offset-4 hover:text-ink hover:underline"

   605→              </button>
   606→            </div>
   607→            {openJobs.length === 0 ? (
   608→              <div className="dlg-card">
   609→                <EmptyState
   610→                  icon={Briefcase}
   611→                  tone="amber"
   612→                  title="No open positions right now"
   613→                  description="New openings appear here as soon as they are published."
   614→                  action={

   616→                      type="button"
   617→                      onClick={() => navigate("jobs")}
   618→                      className="dlg-ghost inline-flex min-h-[44px] items-center px-5 py-2 text-sm"

   627→                {openJobs.map((j) => (
   628→                  <OpenJobCard key={j.id} job={j} />
   629→                ))}
   630→              </div>

   632→          </section>
   633→
   634→          {/* RIGHT — Your applications (first on mobile, sticky rail on desktop) */}
   635→          <section className="order-first lg:order-none lg:sticky lg:top-6 lg:col-start-2 lg:row-start-1 lg:max-h-[calc(100vh-3rem)] lg:overflow-y-auto scroll-thin">
   636→            <div className="mb-4 flex items-center gap-3">
   637→              <h2 className="font-display text-xl text-ink">Your Applications</h2>
   638→              {sortedApps.length > 0 && (
   639→                <span className="status-pill status-neutral num">{sortedApps.length}</span>

   641→              <span className="hidden h-px flex-1 bg-divider sm:block" aria-hidden />
   642→            </div>

   650→                  action={
   651→                    <button
   652→                      type="button"
   653→                      onClick={() => navigate("jobs")}

   663→                {sortedApps.map((app, i) => (

   671→
   672→      {detail && (

   675→          onClose={() => setDetail(null)}
   676→          onCancelled={() => {
   677→            setDetail(null);
   678→            void reload(true);
   679→          }}
   680→        />
   681→      )}
   682→    </div>
   683→  );

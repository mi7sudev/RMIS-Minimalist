     1→// ============================================================================
     2→// email.ts — provider-agnostic EMAIL sender for RMIS notifications.
     3→//
     4→// Mirrors the design of sms.ts (same contract, same audit story):
     5→//
     6→//   1. "mock" (default, dev) — FREE forever. No external service: every send
     7→//      is logged to the `email_logs` table so the whole flow (shortlist
     8→//      notice included) is testable end-to-end and auditable in the admin
     9→//      Email panel.
    10→//   2. "resend" — production-grade HTTP API. Free tier (100 emails/day),
    11→//      no SDK needed — plain fetch. Set RESEND_API_KEY + (optional)
    12→//      EMAIL_FROM. Perfect fit: shortlist notices are low-volume, so the
    13→//      free tier realistically costs ₱0 for this office.
    14→//
    15→// CONFIG (env vars):
    16→//   EMAIL_PROVIDER=mock|resend
    17→//   # resend:
    18→//   RESEND_API_KEY=<key from resend.com>
    19→//   EMAIL_FROM="DOST-MIRDC Recruitment <recruitment@mirdc.gov.ph>"
    20→//
    21→// Design rules (identical to sms.ts):
    22→//   - sendEmail() NEVER throws — callers fire it after their own success
    23→//     path and a provider outage must never break a status update.
    24→//     Results are always persisted to `email_logs`.
    25→//   - Templates: the SHORTLIST notice is the flagship (rich HTML + text);
    26→//     every other status change gets a simple, honest plain-text notice.
    27→// ============================================================================

    84→export type SendEmailInput = {
    85→  to: string | null | undefined;
    86→  subject: string;
    87→  /** plain-text body — always required (email clients + deliverability) */
    88→  text: string;
    89→  /** optional HTML body (inline styles only — no external assets) */
    90→  html?: string;

    92→  related?: { type: string; id: number };
    93→  /**
    94→   * Optional file attachments (base64 content). Only the ACTIVE provider
    95→   * receives the bytes — email_logs persists metadata (name + size) only,
    96→   * never the file content (PII / DB-size discipline).
    97→   */
    98→  attachments?: EmailAttachment[];

   101→/** A single email attachment. `content` is base64-encoded file bytes. */

   103→  filename: string;

   105→  contentType?: string;

   108→export type SendEmailResult = {
   109→  attempted: boolean;
   110→  status: "sent" | "failed" | "mock" | "skipped";
   111→  provider: EmailProviderId;
   112→  providerRef?: string;
   113→  error?: string;
   114→  to?: string;

   117→const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/;

   119→/** Fire an email through the active provider. Never throws. */
   120→export async function sendEmail(input: SendEmailInput): Promise<SendEmailResult> {
   121→  const provider = getEmailProviderId();
   122→  const to = (input.to ?? "").trim();
   123→
   124→  // Unreachable/absent address — record the skip so admins can see why an
   125→  // applicant got no email.
   126→  if (!to || !EMAIL_RE.test(to)) {
   127→    await logEmail({
   128→      to: to || "(none)",
   129→      subject: input.subject,
   130→      bodyText: input.text,
   131→      bodyHtml: input.html ?? null,
   132→      provider,
   133→      status: "skipped",
   134→      error: "No valid email address on record",
   135→      relatedType: input.related?.type,
   136→      relatedId: input.related?.id,
   137→      attachmentsMeta: attachmentsMeta(input.attachments),
   138→    }).catch(() => {});
   139→    return { attempted: false, status: "skipped", provider };
   140→  }
   141→
   142→  let result: { ok: boolean; ref?: string; error?: string; status: "sent" | "failed" | "mock" };
   143→  try {
   144→    switch (provider) {
   145→      case "resend":
   146→        result = await sendViaResend(
   147→          to,
   148→          input.subject,
   149→          input.text,
   150→          input.html,
   151→          input.attachments
   152→        );
   153→        break;
   154→      default:
   155→        // mock: nothing is delivered — attachment metadata is still logged
   156→        // below so the audit trail shows what WOULD have been sent.
   157→        result = { ok: true, status: "mock", ref: "mock" };
   158→    }
   159→  } catch (e) {
   160→    result = {
   161→      ok: false,
   162→      status: "failed",
   163→      error: e instanceof Error ? e.message : String(e),
   164→    };
   165→  }
   166→
   167→  await logEmail({
   168→    to,
   169→    subject: input.subject,
   170→    bodyText: input.text,
   171→    bodyHtml: input.html ?? null,
   172→    provider,
   173→    status: result.status,
   174→    providerRef: result.ref,
   175→    error: result.error,
   176→    relatedType: input.related?.type,
   177→    relatedId: input.related?.id,
   178→    attachmentsMeta: attachmentsMeta(input.attachments),
   179→  }).catch(() => {});
   180→
   181→  return {
   182→    attempted: true,
   183→    status: result.status,
   184→    provider,
   185→    providerRef: result.ref,
   186→    error: result.error,
   187→    to,
   188→  };
   189→}
   190→
   191→/** JSON metadata persisted to email_logs.attachments — name + size only. */
   192→function attachmentsMeta(attachments?: EmailAttachment[]): string | null {
   193→  if (!attachments || attachments.length === 0) return null;
   194→  try {
   195→    return JSON.stringify(
   196→      attachments.map((a) => ({
   197→        name: a.filename,
   198→        bytes: Math.floor((a.content.length * 3) / 4),
   199→      }))
   200→    );
   201→  } catch {
   202→    return null;
   203→  }
   204→}

   206→async function logEmail(row: {
   207→  to: string;
   208→  subject: string;
   209→  bodyText: string;
   210→  bodyHtml: string | null;
   211→  provider: EmailProviderId;
   212→  status: string;
   213→  providerRef?: string;
   214→  error?: string;
   215→  relatedType?: string;
   216→  relatedId?: number;
   217→  attachmentsMeta?: string | null;
   218→}) {
   219→  await db.emailLog.create({
   220→    data: {
   221→      to: row.to,
   222→      subject: row.subject,
   223→      bodyText: row.bodyText,
   224→      bodyHtml: row.bodyHtml,
   225→      provider: row.provider,
   226→      status: row.status,
   227→      providerRef: row.providerRef,
   228→      error: row.error,
   229→      relatedType: row.relatedType,
   230→      relatedId: row.relatedId,
   231→      attachments: row.attachmentsMeta ?? null,
   232→    },
   233→  });
   234→}

   236→// --- Driver: Resend (HTTP API — no SDK) ------------------------------------
   237→
   238→async function sendViaResend(
   239→  to: string,
   240→  subject: string,
   241→  text: string,
   242→  html?: string,
   243→  attachments?: EmailAttachment[]
   244→): Promise<{ ok: boolean; ref?: string; error?: string; status: "sent" | "failed" }> {
   245→  const key = process.env.RESEND_API_KEY;
   246→  if (!key) return { ok: false, status: "failed", error: "RESEND_API_KEY not set" };
   247→
   248→  const res = await fetch("https://api.resend.com/emails", {
   249→    method: "POST",
   250→    headers: {
   251→      "Content-Type": "application/json",
   252→      Authorization: `Bearer ${key}`,
   253→    },
   254→    body: JSON.stringify({
   255→      from: DEFAULT_FROM,
   256→      to: [to],
   257→      subject,
   258→      text,
   259→      ...(html ? { html } : {}),
   260→      ...(attachments && attachments.length > 0
   261→        ? {
   262→            attachments: attachments.map((a) => ({
   263→              filename: a.filename,
   264→              content: a.content,
   265→              ...(a.contentType ? { content_type: a.contentType } : {}),
   266→            })),
   267→          }
   268→        : {}),
   269→    }),
   270→  });
   271→  if (!res.ok) {
   272→    return {
   273→      ok: false,
   274→      status: "failed",
   275→      error: `resend ${res.status}: ${await safeBody(res)}`,
   276→    };
   277→  }
   278→  const data = (await res.json().catch(() => ({}))) as { id?: string };
   279→  return { ok: true, status: "sent", ref: data?.id };
   280→}
   281→
   282→async function safeBody(res: Response): Promise<string> {
   283→  try {
   284→    return (await res.text()).slice(0, 200);
   285→  } catch {
   286→    return "";
   287→  }
   288→}

   322→function esc(s: string): string {
   323→  return s
   324→    .replace(/&/g, "&amp;")

   326→    .replace(/>/g, "&gt;")
   327→    .replace(/"/g, "&quot;");

   463→  positionTitle: string,
   464→  status: string,
   465→  reason?: string | null
   466→): { subject: string; text: string; html: string } {

   500→      Sign in to RMIS to view the full details of your application.
   501→    </p>`
   502→  );
   503→
   504→  return { subject, text, html };
   505→}

   527→export async function emailApplicationStatusChanged(opts: {
   528→  email: string | null | undefined;
   529→  firstName: string | null;
   530→  positionTitle: string;
   531→  status: string;
   532→  reason?: string | null;
   533→  applicationId: number;
   534→}): Promise<SendEmailResult> {
   535→  const key = opts.status.trim().replace(/\s+/g, "_").toUpperCase();
   536→  // The shortlist moment gets the flagship template; the explicit review
   537→  // start gets the dedicated under-review notice; everything else is a plain
   538→  // status notice.
   539→  if (key === "SHORTLISTED") {
   540→    return emailApplicationShortlisted({
   541→      email: opts.email,
   542→      firstName: opts.firstName,
   543→      positionTitle: opts.positionTitle,
   544→      applicationId: opts.applicationId,
   545→    });
   546→  }
   547→  if (key === "UNDER_REVIEW") {
   548→    const t = emailUnderReview(opts.firstName, opts.positionTitle);
   549→    return sendEmail({
   550→      to: opts.email,
   551→      subject: t.subject,
   552→      text: t.text,
   553→      html: t.html,
   554→      related: { type: "application", id: opts.applicationId },
   555→    });
   556→  }
   557→  const t = emailStatusChanged(opts.firstName, opts.positionTitle, opts.status, opts.reason);

    66→function UsersPanel() {
    67→  const [rows, setRows] = useState<UserRow[] | null>(null);
    68→  const [total, setTotal] = useState(0);
    69→  const [q, setQ] = useState("");
    70→  const [role, setRole] = useState("ALL");
    71→  const [page, setPage] = useState(1);
    72→  const pageSize = 15;
    73→  const [busyId, setBusyId] = useState<string | null>(null);
    74→
    75→  const [createOpen, setCreateOpen] = useState(false);
    76→  const [editUser, setEditUser] = useState<UserRow | null>(null);
    77→  const [hardTarget, setHardTarget] = useState<UserRow | null>(null);
    78→  const [disableTarget, setDisableTarget] = useState<UserRow | null>(null);
    79→
    80→  const load = useCallback(async (silent = false) => {
    81→    if (!silent) setRows(null);
    82→    try {
    83→      const params = new URLSearchParams({ page: String(page), pageSize: String(pageSize) });
    84→      if (q.trim()) params.set("q", q.trim());
    85→      if (role !== "ALL") params.set("role", role);
    86→      const data = await apiFetch<{ data: UserRow[]; total: number }>(`/api/admin/users?${params.toString()}`);
    87→      setRows(data.data);
    88→      setTotal(data.total);
    89→    } catch (e) {
    90→      if (!silent) toast.error(e instanceof Error ? e.message : "Failed to load users");
    91→      setRows([]);
    92→    }
    93→  }, [q, role, page]);
    94→
    95→  useEffect(() => {
    96→    const t = setTimeout(() => void load(), 250);
    97→    return () => clearTimeout(t);
    98→  }, [load]);
    99→
   100→  const toggleActive = async (u: UserRow) => {
   101→    setBusyId(u.id);
   102→    try {
   103→      if (!u.blocked) {
   104→        setDisableTarget(u); // confirm first
   105→      } else {
   106→        await apiFetch(`/api/admin/users/${u.id}`, { method: "PATCH", body: { isActive: true } });
   107→        toast.success(`${u.username} re-enabled`);
   108→        await load(true);
   109→      }
   110→    } catch (e) {
   111→      toast.error(e instanceof Error ? e.message : "Action failed");
   112→    } finally {
   113→      setBusyId(null);
   114→    }
   115→  };
   116→
   117→  const confirmDisable = async () => {
   118→    if (!disableTarget) return;
   119→    try {
   120→      await apiFetch(`/api/admin/users/${disableTarget.id}`, { method: "DELETE" });
   121→      toast.success(`${disableTarget.username} disabled`);
   122→      await load(true);
   123→    } catch (e) {
   124→      toast.error(e instanceof Error ? e.message : "Action failed");
   125→    }
   126→    setDisableTarget(null);
   127→  };
   128→
   129→  const confirmHardDelete = async () => {
   130→    if (!hardTarget) return;
   131→    try {
   132→      await apiFetch(`/api/admin/users/${hardTarget.id}?hard=1`, { method: "DELETE" });
   133→      toast.success(`${hardTarget.username} permanently deleted`);
   134→      await load(true);
   135→    } catch (e) {
   136→      toast.error(e instanceof Error ? e.message : "Delete failed");
   137→    }
   138→    setHardTarget(null);
   139→  };
   140→
   141→  const pages = Math.max(1, Math.ceil(total / pageSize));
   142→
   143→  return (
   144→    <div className="space-y-4">
   145→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   146→        <p className="text-sm text-stone">{total} account{total === 1 ? "" : "s"}</p>
   147→        <button type="button" className={ctaBtn} onClick={() => setCreateOpen(true)}>Create User</button>
   148→      </div>
   149→
   150→      <div className="dlg-card p-4 flex flex-col gap-3 sm:flex-row">
   151→        <div className="relative flex-1">
   152→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-pebble" aria-hidden />
   153→          <Input value={q} onChange={(e) => { setQ(e.target.value); setPage(1); }} placeholder="Search name, email or username…" className="dlg-input min-h-[44px] pl-9" />
   154→        </div>
   155→        <Select value={role} onValueChange={(v) => { setRole(v); setPage(1); }}>
   156→          <SelectTrigger className="dlg-input min-h-[44px] w-full sm:w-44"><SelectValue /></SelectTrigger>
   157→          <SelectContent>
   158→            <SelectItem value="ALL">All roles</SelectItem>
   159→            <SelectItem value="APPLICANT">Applicant</SelectItem>
   160→            <SelectItem value="EVALUATOR">Evaluator</SelectItem>
   161→            <SelectItem value="ADMIN">Admin</SelectItem>
   162→          </SelectContent>
   163→        </Select>
   164→      </div>
   165→
   166→      <div className="dlg-card overflow-hidden">
   167→        <div className="max-h-96 overflow-y-auto scroll-thin">
   168→          <table className="w-full text-sm">
   169→            <thead className="bg-fog text-left text-xs text-stone">
   170→              <tr>
   171→                <th className="px-4 py-3 font-medium">User</th>
   172→                <th className="px-4 py-3 font-medium">Role</th>
   173→                <th className="px-4 py-3 font-medium">Active</th>
   174→                <th className="px-4 py-3 font-medium text-right">Actions</th>
   175→              </tr>
   176→            </thead>
   177→            <tbody>

   186→                  <td className="px-4 py-3">

   195→                    <div className="flex justify-end gap-2">

   325→function EditUserDialog({ user, onOpenChange, onSaved }: { user: UserRow; onOpenChange: (o: boolean) => void; onSaved: () => void }) {
   326→  const [form, setForm] = useState({ firstName: user.firstName ?? "", lastName: user.lastName ?? "", role: user.role, password: "" });
   327→  const [busy, setBusy] = useState(false);
   328→
   329→  const submit = async () => {
   330→    setBusy(true);
   331→    try {
   332→      const body: Record<string, unknown> = { firstName: form.firstName, lastName: form.lastName, role: form.role };
   333→      if (form.password) body.password = form.password;
   334→      await apiFetch(`/api/admin/users/${user.id}`, { method: "PATCH", body });
   335→      toast.success("Account updated");
   336→      onOpenChange(false);
   337→      onSaved();
   338→    } catch (e) {
   339→      toast.error(e instanceof Error ? e.message : "Update failed");
   340→    } finally {
   341→      setBusy(false);
   342→    }
   343→  };
   344→
   345→  return (
   346→    <Dialog open onOpenChange={onOpenChange}>
   347→      <DialogContent className="dlg-card-plain sm:max-w-md">
   348→        <DialogHeader className="text-left">
   349→          <DialogTitle className="font-display text-2xl text-ink">Edit @{user.username}</DialogTitle>
   350→          <DialogDescription className="text-sm text-stone">{user.email}</DialogDescription>
   351→        </DialogHeader>
   352→        <div className="space-y-3">
   353→          <div className="grid grid-cols-2 gap-3">
   354→            <div><Label className={labelCls()}>First name</Label><Input className="dlg-input min-h-[44px]" value={form.firstName} onChange={(e) => setForm({ ...form, firstName: e.target.value })} /></div>
   355→            <div><Label className={labelCls()}>Last name</Label><Input className="dlg-input min-h-[44px]" value={form.lastName} onChange={(e) => setForm({ ...form, lastName: e.target.value })} /></div>
   356→          </div>

   399→function AuditPanel() {
   400→  const [rows, setRows] = useState<AuditRow[] | null>(null);
   401→  const [summary, setSummary] = useState<{ totalEvents: number; byRole: Record<string, number> } | null>(null);
   402→  const [actions, setActions] = useState<string[]>([]);
   403→  const [q, setQ] = useState("");
   404→  const [action, setAction] = useState("ALL");
   405→  const [page, setPage] = useState(1);
   406→  const pageSize = 25;
   407→
   408→  const load = useCallback(async (silent = false) => {
   409→    if (!silent) setRows(null);

   411→      const params = new URLSearchParams({ page: String(page), pageSize: String(pageSize) });
   412→      if (q.trim()) params.set("search", q.trim());
   413→      if (action !== "ALL") params.set("action", action);
   414→      const data = await apiFetch<{
   415→        data: AuditRow[];
   416→        total: number;
   417→        actions: string[];
   418→        summary: { totalEvents: number; onPage: number; topActions: { action: string; count: number }[]; byRole: Record<string, number> };

   423→    } catch (e) {
   424→      if (!silent) toast.error(e instanceof Error ? e.message : "Failed to load audit log");
   425→      setRows([]);
   426→    }
   427→  }, [q, action, page]);
   428→
   429→  useEffect(() => {
   430→    const t = setTimeout(() => void load(), 250);
   431→    return () => clearTimeout(t);
   432→  }, [load]);
   433→
   434→  const pages = Math.max(1, Math.ceil((rows === null ? 0 : summary?.totalEvents ?? 0) / pageSize));
   435→
   436→  return (
   437→    <div className="space-y-4">
   438→      <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   439→        <div className="dlg-card p-4">
   440→          <p className="text-xs text-pebble">Total events</p>
   441→          <p className="font-display text-3xl text-ink tabular-nums">{summary?.totalEvents ?? "—"}</p>
   442→        </div>
   443→        {(["ADMIN", "EVALUATOR", "APPLICANT"] as const).map((r) => (
   444→          <div key={r} className="dlg-card p-4">
   445→            <p className="text-xs text-pebble">{humanize(r)}</p>
   446→            <p className="font-display text-3xl text-ink tabular-nums">{summary?.byRole?.[r] ?? 0}</p>
   447→          </div>
   448→        ))}
   449→      </div>
   450→
   451→      <div className="dlg-card p-4 flex flex-col gap-3 sm:flex-row">
   452→        <div className="relative flex-1">
   453→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-pebble" aria-hidden />
   454→          <Input value={q} onChange={(e) => { setQ(e.target.value); setPage(1); }} placeholder="Search descriptions and actors…" className="dlg-input min-h-[44px] pl-9" />
   455→        </div>
   456→        <Select value={action} onValueChange={(v) => { setAction(v); setPage(1); }}>
   457→          <SelectTrigger className="dlg-input min-h-[44px] w-full sm:w-56"><SelectValue /></SelectTrigger>
   458→          <SelectContent>
   459→            <SelectItem value="ALL">All actions</SelectItem>
   460→            {actions.map((a) => <SelectItem key={a} value={a}>{humanize(a)}</SelectItem>)}
   461→          </SelectContent>
   462→        </Select>
   463→      </div>
   464→
   465→      <div className="dlg-card overflow-hidden">
   466→        <div className="max-h-96 overflow-y-auto scroll-thin">
   467→          <table className="w-full text-sm">
   468→            <thead className="bg-fog text-left text-xs text-stone">
   469→              <tr>
   470→                <th className="px-4 py-3 font-medium">Actor</th>
   471→                <th className="px-4 py-3 font-medium">Action</th>
   472→                <th className="px-4 py-3 font-medium">Detail</th>
   473→                <th className="px-4 py-3 font-medium">When</th>
   474→              </tr>
   475→            </thead>
   476→            <tbody>
   477→              {rows === null && <tr><td colSpan={4} className="p-4"><Skeleton className="h-16 w-full bg-fog" /></td></tr>}
   478→              {rows?.length === 0 && <tr><td colSpan={4} className="p-8 text-center text-sm text-pebble">No audit events match.</td></tr>}
   479→              {rows?.map((r) => (
   480→                <tr key={r.id} className="border-t border-border align-top">
   481→                  <td className="px-4 py-3">
   482→                    <p className="font-medium text-ink">{r.userLabel ?? "System"}</p>
   483→                    {r.userRole && <p className="text-xs text-pebble">{r.userRole}</p>}
   484→                  </td>
   485→                  <td className="px-4 py-3"><span className="inline-flex rounded-full bg-fog px-2.5 py-1 text-xs font-medium text-graphite">{humanize(r.action)}</span></td>
   486→                  <td className="px-4 py-3 text-stone">{r.description ?? "—"}</td>
   487→                  <td className="px-4 py-3 text-xs text-pebble">{timeAgo(r.timestamp)}</td>
   488→                </tr>
   489→              ))}
   490→            </tbody>
   491→          </table>
   492→        </div>
   493→      </div>
   494→
   495→      {pages > 1 && (
   496→        <div className="flex items-center justify-center gap-2">
   497→          <button type="button" className={ghostBtn} disabled={page <= 1} onClick={() => setPage((p) => p - 1)}>Previous</button>
   498→          <span className="text-sm text-stone">Page {page} of {pages}</span>
   499→          <button type="button" className={ghostBtn} disabled={page >= pages} onClick={() => setPage((p) => p + 1)}>Next</button>
   500→        </div>
   501→      )}
   502→    </div>
   503→  );
   504→}

   508→type MsgLog = {
   509→  id: number;
   510→  to: string;
   511→  subject?: string;
   512→  message?: string;
   513→  provider: string;
   514→  status: string;
   515→  error: string | null;
   516→  createdAt: string;
   517→  attachments?: string | null;
   518→};
   519→
   520→type MsgPanelData = {
   521→  provider: string;
   522→  configured: boolean;
   523→  stats: { total: number; sent: number; failed: number; skipped?: number; mock?: number; last24h: number };
   524→  logs: MsgLog[];
   525→};
   526→
   527→function StatusChip({ status }: { status: string }) {
   528→  const cls =
   529→    status === "sent" ? "bg-ink text-white"
   530→    : status === "failed" ? "bg-dusty-rose/15 text-dusty-rose"
   531→    : status === "skipped" ? "bg-fog text-pebble"
   532→    : "bg-fog text-stone";
   533→  return <span className={`inline-flex rounded-full px-2.5 py-1 text-xs font-medium ${cls}`}>{status}</span>;
   534→}
   535→
   536→function MessagingPanel({ kind }: { kind: "sms" | "email" }) {
   537→  const [data, setData] = useState<MsgPanelData | null>(null);
   538→  const [to, setTo] = useState("");
   539→  const [subject, setSubject] = useState("");
   540→  const [message, setMessage] = useState("");
   541→  const [busy, setBusy] = useState(false);
   542→
   543→  const load = useCallback(async (silent = false) => {
   544→    if (!silent) setData(null);
   545→    try {
   546→      setData(await apiFetch<MsgPanelData>(`/api/admin/${kind}`));
   547→    } catch (e) {
   548→      if (!silent) toast.error(e instanceof Error ? e.message : "Failed to load");
   549→      setData({ provider: "mock", configured: false, stats: { total: 0, sent: 0, failed: 0, last24h: 0 }, logs: [] });
   550→    }
   551→  }, [kind]);
   552→
   553→  useEffect(() => { void load(); }, [load]);
   554→
   555→  const testSend = async () => {
   556→    setBusy(true);
   557→    try {
   558→      const body = kind === "sms" ? { to, message: message || undefined } : { to, subject: subject || undefined, message: message || undefined };
   559→      const res = await apiFetch<{ email?: { status: string }; sms?: { status: string } }>(`/api/admin/${kind}`, { method: "POST", body });
   560→      const status = res.email?.status ?? res.sms?.status ?? "sent";
   561→      if (status === "sent") toast.success(`Test ${kind} sent`);
   562→      else if (status === "mock") toast.info(`Test ${kind} logged in mock mode`, { description: "Configure a provider to deliver for real." });
   563→      else if (status === "skipped") toast.warning("Recipient skipped", { description: "Check the address or mobile number format." });
   564→      else toast.error(`Delivery failed — check the logs`);
   565→      await load(true);
   566→    } catch (e) {
   567→      toast.error(e instanceof Error ? e.message : "Send failed");
   568→    } finally {
   569→      setBusy(false);
   570→    }
   571→  };
   572→
   573→  const tiles: [string, number | undefined][] = [
   574→    ["Total", data?.stats.total],
   575→    ["Sent", data?.stats.sent],
   576→    ["Failed", data?.stats.failed],
   577→    ["Last 24h", data?.stats.last24h],
   578→  ];
   579→
   580→  return (
   581→    <div className="space-y-4">
   582→      <div className="dlg-card p-6">
   583→        <div className="flex flex-wrap items-center justify-between gap-3">
   584→          <div>
   585→            <p className="text-xs text-pebble">Active provider</p>
   586→            <p className="font-display text-xl text-ink">{humanize(data?.provider ?? "…")}</p>
   587→          </div>
   588→          <span className={`inline-flex items-center rounded-full px-3 py-1 text-xs font-medium ${data?.provider === "mock" ? "bg-fog text-stone" : "bg-ink text-white"}`}>
   589→            {data?.provider === "mock" ? "Mock mode — messages are logged only" : data?.configured ? "Configured" : "Not configured"}
   590→          </span>
   591→        </div>
   592→        <div className="mt-4 grid grid-cols-2 gap-3 lg:grid-cols-4">
   593→          {tiles.map(([label, value]) => (
   594→            <div key={label} className="rounded-[12px] bg-fog p-3">
   595→              <p className="text-xs text-pebble">{label}</p>
   596→              <p className="font-display text-2xl text-ink tabular-nums">{value ?? "—"}</p>
   597→            </div>
   598→          ))}
   599→        </div>
   600→      </div>
   601→
   602→      <div className="dlg-card p-6">
   603→        <p className="mb-4 text-sm font-medium text-ink">Test send</p>
   604→        <div className="grid gap-3 sm:grid-cols-2">
   605→          <div>
   606→            <Label className={labelCls()}>{kind === "sms" ? "Mobile number (09…)" : "Recipient email"}</Label>
   607→            <Input className="dlg-input min-h-[44px]" value={to} onChange={(e) => setTo(e.target.value)} placeholder={kind === "sms" ? "09171234567" : "name@example.com"} />
   608→          </div>
   609→          {kind === "email" && (
   610→            <div>
   611→              <Label className={labelCls()}>Subject</Label>
   612→              <Input className="dlg-input min-h-[44px]" value={subject} onChange={(e) => setSubject(e.target.value)} placeholder="RMIS test email" />
   613→            </div>
   614→          )}
   615→          <div className={kind === "email" ? "sm:col-span-2" : ""}>
   616→            <Label className={labelCls()}>Message</Label>
   617→            <Input className="dlg-input min-h-[44px]" value={message} onChange={(e) => setMessage(e.target.value)} placeholder="Optional message" />
   618→          </div>
   619→        </div>
   620→        <button type="button" className={`${ctaBtn} mt-4`} disabled={busy || !to.trim()} onClick={() => void testSend()}>
   621→          {busy ? "Sending…" : `Send test ${kind}`}
   622→        </button>
   623→      </div>
   624→
   625→      <div className="dlg-card overflow-hidden">
   626→        <div className="max-h-96 overflow-y-auto scroll-thin">
   627→          <table className="w-full text-sm">
   628→            <thead className="bg-fog text-left text-xs text-stone">
   629→              <tr>
   630→                <th className="px-4 py-3 font-medium">To</th>
   631→                <th className="px-4 py-3 font-medium">{kind === "sms" ? "Message" : "Subject"}</th>
   632→                <th className="px-4 py-3 font-medium">Status</th>
   633→                <th className="px-4 py-3 font-medium">When</th>
   634→              </tr>
   635→            </thead>
   636→            <tbody>
   637→              {data === null && <tr><td colSpan={4} className="p-4"><Skeleton className="h-16 w-full bg-fog" /></td></tr>}
   638→              {data?.logs.length === 0 && <tr><td colSpan={4} className="p-8 text-center text-sm text-pebble">Nothing sent yet.</td></tr>}
   639→              {data?.logs.map((l) => (
   640→                <tr key={l.id} className="border-t border-border align-top">
   641→                  <td className="px-4 py-3 font-medium text-ink">{l.to}</td>
   642→                  <td className="max-w-[280px] truncate px-4 py-3 text-stone">{l.subject ?? l.message ?? "—"}</td>
   643→                  <td className="px-4 py-3"><StatusChip status={l.status} />{l.error && <p className="mt-1 text-xs text-pebble">{l.error}</p>}</td>
   644→                  <td className="px-4 py-3 text-xs text-pebble">{timeAgo(l.createdAt)}</td>
   645→                </tr>
   646→              ))}
   647→            </tbody>
   648→          </table>
   649→        </div>
   650→      </div>
   651→    </div>
   652→  );
   653→}

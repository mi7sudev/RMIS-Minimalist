   111→/** Legacy tone→class bridge (kept for existing importers). */
   112→export function toneClass(tone: Tone): string {
   113→  const map: Record<Tone, StatusVariant> = {

   115→    warning: "warn",
   116→    danger: "bad",
   117→    info: "info",

   124→/**
   125→ * Canonical stage/status pill. Displays the normalized status label; the
   126→ * variant comes from variantForStatus unless explicitly overridden (used for
   127→ * completion / verdict labels that are not application statuses).
   128→ */
   129→export function StatusPill({
   130→  status,
   131→  variant,
   132→  className = "",
   133→}: {
   134→  status: string | null | undefined;
   135→  variant?: StatusVariant;
   136→  className?: string;
   137→}) {
   138→  const label = getStatusMeta(status).label;
   139→  const v: StatusVariant = variant ?? variantForStatus(status);
   140→  return <span className={cn(pillClass(v), "shrink-0", className)}>{label}</span>;
   141→}

   143→function verdictMeta(verdict: string): { chip: string; variant: StatusVariant; meaning: string } {
   144→  switch (verdict) {
   145→    case "ALL_MET":
   146→      return { chip: "Qualified", variant: variantForVerdict("Qualified"), meaning: "Meets the minimum requirements" };
   147→    case "PARTIAL":
   148→      return { chip: "Partial", variant: variantForVerdict("Partial"), meaning: "Partially meets the minimum requirements" };
   149→    case "NONE_MET":
   150→      return { chip: "Not qualified", variant: variantForVerdict("Not qualified"), meaning: "Does not meet the minimum requirements" };
   151→    case "NEEDS_REVIEW":
   152→      return { chip: "Verify", variant: variantForVerdict("Verify"), meaning: "Needs manual verification" };
   153→    case "NO_REQUIREMENTS":
   154→      return { chip: "No reqs", variant: variantForVerdict("No reqs"), meaning: "No published requirements" };
   155→    default:
   156→      return { chip: verdict, variant: variantForVerdict(verdict), meaning: "" };
   157→  }
   158→}

   430→function NoticeFormDialog({
   431→  applicationId,
   432→  kind,
   433→  onClose,
   434→  onSent,
   435→}: {
   436→  applicationId: number;
   437→  kind: "interview" | "skills_exam";
   438→  onClose: () => void;
   439→  onSent: () => void;
   440→}) {
   441→  const [examType, setExamType] = useState("");
   442→  const [date, setDate] = useState("");
   443→  const [time, setTime] = useState("");
   444→  const [venue, setVenue] = useState("");
   445→  const [contact, setContact] = useState("");
   446→  const [notes, setNotes] = useState("");
   447→  const [busy, setBusy] = useState(false);
   448→
   449→  const ready = date.trim() !== "" && time.trim() !== "" && venue.trim() !== "";
   450→
   451→  const submit = async () => {
   452→    setBusy(true);
   453→    try {
   454→      const res = await apiFetch<NoticeSendResult>(`/api/evaluator/applications/${applicationId}/notice`, {
   455→        method: "POST",
   456→        body: {
   457→          type: kind,
   458→          date: date || undefined,
   459→          time: time || undefined,
   460→          venue: venue || undefined,
   461→          contact: contact || undefined,
   462→          notes: notes || undefined,
   463→          examType: kind === "skills_exam" ? examType || undefined : undefined,
   464→        },
   465→      });
   466→      const label = kind === "interview" ? "Interview invitation" : "Skills-exam notice";
   467→      toast.success(`${label} sent`, {
   468→        description: `Email: ${res.email?.status ?? "unknown"} · SMS: ${res.sms?.status ?? "unknown"}`,
   469→      });
   470→      onSent();
   471→      onClose();
   472→    } catch (e) {
   473→      toast.error(e instanceof Error ? e.message : "Failed to send the notice");
   474→    } finally {
   475→      setBusy(false);
   476→    }
   477→  };
   478→
   479→  return (
   480→    <Dialog open onOpenChange={(v) => !v && onClose()}>
   481→      <DialogContent className="sm:max-w-md">
   482→        <DialogHeader>
   483→          <DialogTitle className="font-display text-xl">
   484→            {kind === "interview" ? "Send interview invitation" : "Send skills-exam notice"}
   485→          </DialogTitle>
   486→          <DialogDescription>
   487→            The notice fans out as an email plus a companion SMS to the applicant.
   488→          </DialogDescription>
   489→        </DialogHeader>
   490→        <div className="space-y-4">
   491→          {kind === "skills_exam" && (
   492→            <div className="space-y-1.5">
   493→              <Label htmlFor="notice-exam-type" className="text-xs text-stone">
   494→                Exam type
   495→              </Label>
   496→              <Input
   497→                id="notice-exam-type"
   498→                className="dlg-input"
   499→                value={examType}
   500→                onChange={(e) => setExamType(e.target.value)}
   501→                placeholder="e.g. Encoding / trade test"
   502→              />
   503→            </div>

   566→function parseAttachments(json: string | null): { name: string; bytes: number }[] {
   567→  if (!json) return [];
   568→  try {
   569→    const parsed: unknown = JSON.parse(json);
   570→    return Array.isArray(parsed) ? (parsed as { name: string; bytes: number }[]) : [];
   571→  } catch {
   572→    return [];
   573→  }

   580→function DirectEmailCard({
   581→  applicationId,
   582→  emails,
   583→  onSent,
   584→}: {
   585→  applicationId: number;
   586→  emails: DirectEmailRow[] | null;
   587→  onSent: () => void;
   588→}) {
   589→  const [subject, setSubject] = useState("");
   590→  const [message, setMessage] = useState("");
   591→  const [files, setFiles] = useState<File[]>([]);
   592→  const [busy, setBusy] = useState(false);
   593→  const fileRef = useRef<HTMLInputElement>(null);
   594→
   595→  const pickFiles = (list: FileList | null) => {
   596→    if (!list) return;
   597→    const next = Array.from(list);
   598→    if (next.length + files.length > 3) {
   599→      toast.error("A maximum of 3 attachments is allowed");
   600→      return;
   601→    }
   602→    if (next.some((f) => f.size > 5 * 1024 * 1024)) {
   603→      toast.error("Each attachment must be 5 MB or smaller");
   604→      return;
   605→    }
   606→    setFiles((prev) => [...prev, ...next].slice(0, 3));
   607→    if (fileRef.current) fileRef.current.value = "";
   608→  };
   609→
   610→  const send = async () => {
   611→    if (!message.trim()) return;
   612→    setBusy(true);
   613→    try {
   614→      const fd = new FormData();
   615→      fd.set("subject", subject.trim());
   616→      fd.set("message", message.trim());
   617→      for (const f of files) fd.append("attachments", f);
   618→      const res = await apiFetch<{ email: { status: string; error?: string | null }; attachments: number }>(
   619→        `/api/evaluator/applications/${applicationId}/email`,
   620→        { method: "POST", formData: fd }
   621→      );
   622→      if (res.email?.status === "failed") {
   623→        toast.error("Email delivery failed", { description: res.email.error ?? undefined });
   624→      } else {
   625→        toast.success(`Direct email ${res.email?.status}`, {
   626→          description: res.attachments ? `${res.attachments} attachment(s) included.` : undefined,
   627→        });
   628→      }
   629→      setSubject("");
   630→      setMessage("");
   631→      setFiles([]);
   632→      onSent();
   633→    } catch (e) {
   634→      toast.error(e instanceof Error ? e.message : "Failed to send the email");
   635→    } finally {
   636→      setBusy(false);
   637→    }
   638→  };
   639→
   640→  return (
   641→    <div className="dlg-card p-4">
   642→      <div className="flex items-center gap-2">
   643→        <Mail className="h-4 w-4 text-stone" aria-hidden />
   644→        <h3 className="text-[15px] font-medium leading-6 text-ink">Direct email</h3>
   645→      </div>
   646→      <p className="mt-1 text-xs text-stone">Recipient is resolved from the applicant record — attachments ≤ 3 files × 5 MB.</p>
   647→      <div className="mt-3 space-y-3">
   648→        <Input
   649→          className="dlg-input"
   650→          placeholder="Subject"
   651→          value={subject}
   652→          maxLength={200}
   653→          onChange={(e) => setSubject(e.target.value)}
   654→          aria-label="Email subject"
   655→        />
   656→        <Textarea
   657→          className="dlg-input min-h-[100px]"
   658→          placeholder="Message *"
   659→          value={message}
   660→          maxLength={5000}
   661→          onChange={(e) => setMessage(e.target.value)}
   662→          aria-label="Email message"
   663→        />
   664→        <div>
   665→          <label className="dlg-ghost focus-ring inline-flex min-h-[44px] cursor-pointer items-center gap-2 rounded-full px-4 text-sm">
   666→            <Paperclip className="h-4 w-4" aria-hidden />
   667→            Attach files
   668→            <input
   669→              ref={fileRef}
   670→              type="file"
   671→              multiple
   672→              className="hidden"
   673→              onChange={(e) => pickFiles(e.target.files)}
   674→              aria-label="Attach files"
   675→            />
   676→          </label>
   677→          {files.length > 0 && (
   678→            <ul className="mt-2 space-y-1">
   679→              {files.map((f) => (
   680→                <li key={f.name} className="flex items-center justify-between gap-2 text-xs text-stone">
   681→                  <span className="truncate">{f.name}</span>
   682→                  <button
   683→                    type="button"
   684→                    className="focus-ring shrink-0 text-[var(--bad)] hover:underline"
   685→                    onClick={() => setFiles((prev) => prev.filter((x) => x !== f))}
   686→                  >
   687→                    Remove
   688→                  </button>
   689→                </li>
   690→              ))}
   691→            </ul>
   692→          )}
   693→        </div>
   694→        <button type="button" className={ctaBtn + " w-full"} disabled={!message.trim() || busy} onClick={() => void send()}>
   695→          <Send className="h-4 w-4" aria-hidden />
   696→          {busy ? "Sending…" : "Send email"}
   697→        </button>
   698→      </div>
   699→      <div className="mt-4 border-t border-[#ececec] pt-3">
   700→        <p className="text-xs font-medium text-stone">History</p>
   701→        <div className="mt-2 max-h-48 space-y-1.5 overflow-y-auto scroll-thin">
   702→          {emails === null ? (
   703→            <p className="text-xs text-pebble">Loading…</p>
   704→          ) : emails.length === 0 ? (
   705→            <p className="text-xs text-pebble">No direct emails sent yet.</p>
   706→          ) : (
   707→            emails.map((m) => {
   708→              const atts = parseAttachments(m.attachments);
   709→              return (
   710→                <div key={m.id} className="flex items-center justify-between gap-2 text-xs">
   711→                  <span className="truncate text-ink">{m.subject || "(no subject)"}</span>
   712→                  <span className="flex shrink-0 items-center gap-2 text-pebble">
   713→                    {atts.length > 0 && <span className="num">{atts.length} att.</span>}
   714→                    <span className="num">{timeAgo(m.createdAt)}</span>
   715→                    <span className={pillClass(emailStatusVariant(m.status))}>{m.status}</span>
   716→                  </span>
   717→                </div>
   718→              );
   719→            })
   720→          )}
   721→        </div>
   722→      </div>
   723→    </div>
   724→  );
   725→}

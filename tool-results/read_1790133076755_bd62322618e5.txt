   128→   128→function MonogramAvatar({ name, size = 36, warm = false }: { name: string; size?: number; warm?: boolean }) {
   129→   129→  const initials = name
   130→   130→    .split(/\s+/)
   131→   131→    .filter(Boolean)
   132→   132→    .slice(0, 2)
   133→   133→    .map((w) => w[0]?.toUpperCase() ?? "")
   134→   134→    .join("");
   135→   135→  return (
   136→   136→    <span aria-hidden="true">
   137→   137→      <Monogram size={size} warm={warm}>{initials || "?"}</Monogram>
   138→   138→    </span>
   139→   139→  );
   140→   140→}

   142→   142→function useFocusRefresh(fn: () => void) {
   143→   143→  useEffect(() => {
   144→   144→    const onFocus = () => fn();
   145→   145→    window.addEventListener("focus", onFocus);
   146→   146→    document.addEventListener("visibilitychange", onFocus);
   147→   147→    return () => {
   148→   148→      window.removeEventListener("focus", onFocus);
   149→   149→      document.removeEventListener("visibilitychange", onFocus);
   150→   150→    };
   151→   151→  }, [fn]);
   152→   152→}

   186→   186→export default function ReviewQueue() {
   187→   187→  const [queue, setQueue] = useState<QueueRow[] | null>(null);
   188→   188→  const [roster, setRoster] = useState<RosterRow[] | null>(null);
   189→   189→  const [error, setError] = useState<string | null>(null);
   190→   190→  const [qualifiedOnly, setQualifiedOnly] = useState(false);
   191→   191→  const [mode, setMode] = useState<"kanban" | "list">("kanban");
   192→   192→  const [listStage, setListStage] = useState<"All" | StageKey>("All");
   193→   193→  const [selectedId, setSelectedId] = useState<number | null>(null);
   194→   194→  const [regretOpen, setRegretOpen] = useState(false);
   195→   195→  const [regretBusy, setRegretBusy] = useState(false);
   196→   196→  const [spin, setSpin] = useState(false);
   197→   197→
   198→   198→  const load = useCallback(async (silent = false) => {
   199→   199→    if (!silent) setError(null);
   200→   200→    try {
   201→   201→      const [q, r] = await Promise.all([
   202→   202→        apiFetch<{ data: QueueRow[] }>("/api/evaluator/queue?pageSize=100"),
   203→   203→        apiFetch<{ data: RosterRow[] }>("/api/admin/applicants?pageSize=100"),
   204→   204→      ]);
   205→   205→      setQueue(q.data);
   206→   206→      setRoster(r.data);
   207→   207→      if (!silent) setError(null);
   208→   208→    } catch (e) {
   209→   209→      if (!silent) setError(e instanceof Error ? e.message : "Failed to load the review queue");
   210→   210→    }
   211→   211→  }, []);
   212→   212→
   213→   213→  useEffect(() => {
   214→   214→    void load();
   215→   215→  }, [load]);
   216→   216→
   217→   217→  // 15 s silent poll + refetch-on-focus (spec §13).
   218→   218→  useEffect(() => {
   219→   219→    const id = setInterval(() => {
   220→   220→      if (!document.hidden) void load(true);
   221→   221→    }, 15_000);
   222→   222→    return () => clearInterval(id);
   223→   223→  }, [load]);
   224→   224→  useFocusRefresh(() => void load(true));
   225→   225→
   226→   226→  const manualRefresh = useCallback(() => {
   227→   227→    setSpin(true);
   228→   228→    void load(true).finally(() => setTimeout(() => setSpin(false), 500));
   229→   229→  }, [load]);
   230→   230→
   231→   231→  const applications = useMemo(() => queue ?? [], [queue]);
   232→   232→  const filtered = useMemo(
   233→   233→    () => (qualifiedOnly ? applications.filter((r) => r.match?.verdict === "ALL_MET") : applications),
   234→   234→    [applications, qualifiedOnly]
   235→   235→  );
   236→   236→  const listRows = useMemo(
   237→   237→    () => (listStage === "All" ? filtered : filtered.filter((r) => stageForStatus(r.status) === listStage)),
   238→   238→    [filtered, listStage]
   239→   239→  );
   240→   240→  const rejectedIds = useMemo(
   241→   241→    () => filtered.filter((r) => stageForStatus(r.status) === "Rejected").map((r) => r.id),
   242→   242→    [filtered]
   243→   243→  );
   244→   244→
   245→   245→  const kanbanColumns = useMemo(() => {
   246→   246→    const map: Record<StageKey, QueueRow[]> = { "Applied": [], "Under Review": [], "Shortlisted": [], "Rejected": [] };
   247→   247→    for (const r of filtered) map[stageForStatus(r.status)].push(r);
   248→   248→    return map;
   249→   249→  }, [filtered]);
   250→   250→
   251→   251→  const rosterCards = useMemo(() => (roster ?? []).filter((r) => r.applicationCount === 0), [roster]);
   252→   252→
   253→   253→  const sendRegrets = async () => {
   254→   254→    setRegretBusy(true);
   255→   255→    try {
   256→   256→      const res = await apiFetch<{ summary: RegretSummary }>("/api/evaluator/applications/regrets", {
   257→   257→        method: "POST",
   258→   258→        body: { applicationIds: rejectedIds },
   259→   259→      });
   260→   260→      const s = res.summary;
   261→   261→      const parts = [
   262→   262→        `${s.sent} sent`,
   263→   263→        `${s.alreadySent} already sent`,
   264→   264→        s.shortlisted ? `${s.shortlisted} shortlisted skipped` : "",
   265→   265→        s.failed ? `${s.failed} failed` : "",
   266→   266→        s.notFound ? `${s.notFound} not found` : "",
   267→   267→      ].filter(Boolean);
   268→   268→      if (s.sent > 0) {
   269→   269→        toast.success(`Regret letters: ${s.sent} sent`, { description: parts.join(" · ") });
   270→   270→      } else {
   271→   271→        toast.info("No new regret letters sent", { description: parts.join(" · ") });
   272→   272→      }
   273→   273→      setRegretOpen(false);
   274→   274→      await load(true);
   275→   275→    } catch (e) {
   276→   276→      toast.error(e instanceof Error ? e.message : "Failed to send regret letters");
   277→   277→    } finally {
   278→   278→      setRegretBusy(false);
   279→   279→    }
   280→   280→  };
   281→   281→
   282→   282→  return (
   283→   283→    <div className="space-y-6">
   284→   284→      {/* Header + toolbar */}
   285→   285→      <PageHeader
   286→   286→        title="Review Queue"
   287→   287→        description="Screen applications, verify minimum qualifications, and advance candidates through the pipeline."
   288→   288→        actions={
   289→   289→          <>
   290→   290→            <span className="status-pill status-neutral num">
   291→   291→              {filtered.length} application{filtered.length === 1 ? "" : "s"}
   292→   292→            </span>
   293→   293→            <label className="focus-ring inline-flex min-h-[44px] cursor-pointer items-center gap-2 rounded-full border border-border bg-white px-3.5 text-sm text-stone transition-colors hover:border-ink/10">
   294→   294→              <Switch checked={qualifiedOnly} onCheckedChange={setQualifiedOnly} aria-label="Qualified only" />
   295→   295→              Qualified only
   296→   296→            </label>
   297→   297→            <div className="flex items-center gap-1 rounded-full bg-fog p-1" role="group" aria-label="View mode">
   298→   298→              <button
   299→   299→                type="button"
   300→   300→                className={`inline-flex min-h-[36px] items-center gap-1.5 rounded-full px-3 text-xs font-medium transition-colors ${mode === "kanban" ? "bg-ink text-white" : "text-stone hover:text-ink"}`}
   301→   301→                onClick={() => setMode("kanban")}
   302→   302→                aria-pressed={mode === "kanban"}
   303→   303→              >
   304→   304→                <LayoutGrid className="h-3.5 w-3.5" aria-hidden />
   305→   305→                Kanban
   306→   306→              </button>
   307→   307→              <button
   308→   308→                type="button"
   309→   309→                className={`inline-flex min-h-[36px] items-center gap-1.5 rounded-full px-3 text-xs font-medium transition-colors ${mode === "list" ? "bg-ink text-white" : "text-stone hover:text-ink"}`}
   310→   310→                onClick={() => setMode("list")}
   311→   311→                aria-pressed={mode === "list"}
   312→   312→              >
   313→   313→                <List className="h-3.5 w-3.5" aria-hidden />
   314→   314→                List
   315→   315→              </button>
   316→   316→            </div>
   317→   317→            <button type="button" className={ghostBtn} onClick={manualRefresh}>
   318→   318→              <RefreshCw className={`h-4 w-4 ${spin ? "animate-spin" : ""}`} aria-hidden />
   319→   319→              Refresh
   320→   320→            </button>
   321→   321→          </>
   322→   322→        }
   323→   323→      />
   324→   324→
   325→   325→      {/* Stage tabs (list mode only) */}
   326→   326→      {mode === "list" && (
   327→   327→        <div className="flex flex-wrap items-center gap-2">
   328→   328→          {(["All", ...PIPELINE_STAGES] as const).map((s) => {
   329→   329→            const count = s === "All" ? filtered.length : filtered.filter((r) => stageForStatus(r.status) === s).length;
   330→   330→            const active = listStage === s;
   331→   331→            return (
   332→   332→              <button
   333→   333→                key={s}
   334→   334→                type="button"
   335→   335→                className={`inline-flex min-h-[36px] items-center gap-2 rounded-full px-3.5 text-xs font-medium transition-colors ${active ? "bg-ink text-white" : "dlg-ghost"}`}
   336→   336→                onClick={() => setListStage(s)}
   337→   337→                aria-pressed={active}
   338→   338→              >
   339→   339→                {s !== "All" && <span className={dotClass(STAGE_DOT[s])} aria-hidden />}
   340→   340→                {s}
   341→   341→                <span className={`num rounded-full px-1.5 ${active ? "bg-white/20" : "bg-fog"}`}>{count}</span>
   342→   342→              </button>
   343→   343→            );
   344→   344→          })}
   345→   345→          {listStage === "Rejected" && rejectedIds.length > 0 && (
   346→   346→            <button
   347→   347→              type="button"
   348→   348→              className={ghostBtn + " border-[var(--bad)]/30 text-[var(--bad)]"}
   349→   349→              onClick={() => setRegretOpen(true)}
   350→   350→            >
   351→   351→              <MailX className="h-4 w-4" aria-hidden />
   352→   352→              Send regret letters ({rejectedIds.length})
   353→   353→            </button>
   354→   354→          )}
   355→   355→        </div>
   356→   356→      )}
   357→   357→
   358→   358→      {/* Error */}
   359→   359→      {error && (
   360→   360→        <div className="dlg-card p-8">
   361→   361→          <EmptyState
   362→   362→            icon={AlertTriangle}
   363→   363→            tone="rose"
   364→   364→            title="Couldn't load the review queue"
   365→   365→            description={error}
   366→   366→            action={
   367→   367→              <button type="button" className={ghostBtn} onClick={() => void load()}>
   368→   368→                Retry
   369→   369→              </button>
   370→   370→            }
   371→   371→          />
   372→   372→        </div>
   373→   373→      )}
   374→   374→
   375→   375→      {/* Loading skeleton */}
   376→   376→      {!error && (queue === null || roster === null) && (
   377→   377→        <>
   378→   378→          <SkeletonKpis count={4} />
   379→   379→          {mode === "kanban" ? <SkeletonKanban columns={5} /> : <SkeletonRows rows={8} rowClassName="h-16" />}
   380→   380→        </>
   381→   381→      )}
   382→   382→
   383→   383→      {/* Stage KPIs (wave-3) */}
   384→   384→      {!error && queue !== null && roster !== null && (
   385→   385→        <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   386→   386→          <KpiCard label="Applied" value={kanbanColumns["Applied"].length} icon={Inbox} tone="info" hint="awaiting start" />
   387→   387→          <KpiCard label="Under Review" value={kanbanColumns["Under Review"].length} icon={ScanSearch} tone="warn" hint="being screened" />
   388→   388→          <KpiCard label="Shortlisted" value={kanbanColumns["Shortlisted"].length} icon={Star} tone="ok" hint="advancing" />
   389→   389→          <KpiCard label="Rejected" value={kanbanColumns["Rejected"].length} icon={XCircle} tone="bad" hint="not qualified" />
   390→   390→        </div>
   391→   391→      )}
   392→   392→
   393→   393→      {/* Kanban */}
   394→   394→      {!error && queue !== null && roster !== null && mode === "kanban" && (
   395→   395→        <div className="flex snap-x snap-proximity gap-4 overflow-x-auto scroll-thin pb-2 lg:grid lg:grid-cols-5 lg:overflow-visible lg:pb-0">
   396→   396→          {/* Applicants column (roster) */}
   397→   397→          <KanbanColumn label="Applicants" icon={STAGE_ICON["Applicants"]} tone={STAGE_TONE["Applicants"]} count={rosterCards.length}>
   398→   398→            {rosterCards.length === 0 ? (
   399→   399→              <EmptyState icon={Users} tone="slate" title="No applicants" description="Registered applicants waiting to apply appear here." compact />
   400→   400→            ) : (
   401→   401→              rosterCards.map((r) => (
   402→   402→                <button
   403→   403→                  key={r.id}
   404→   404→                  type="button"
   405→   405→                  className={cardCls + " focus-ring w-full cursor-pointer p-4 text-left"}
   406→   406→                  onClick={() => navigate("candidate", { id: String(r.id) })}
   407→   407→                >
   408→   408→                  <div className="flex items-start gap-2.5">
   409→   409→                    <MonogramAvatar name={fullName(r)} />
   410→   410→                    <div className="min-w-0 flex-1">
   411→   411→                      <p className="min-w-0 truncate text-sm font-medium text-ink">{fullName(r)}</p>
   412→   412→                      <p className="mt-0.5 truncate text-xs text-stone">{r.emailAddress || "No email on record"}</p>
   413→   413→                    </div>
   414→   414→                  </div>
   415→   415→                  <div className="mt-2.5">
   416→   416→                    <StatusPill
   417→   417→                      status={r.isProfileComplete ? "Profile complete" : "Profile incomplete"}
   418→   418→                      variant={variantForCompletion(r.isProfileComplete)}
   419→   419→                    />
   420→   420→                  </div>
   421→   421→                </button>
   422→   422→              ))
   423→   423→            )}
   424→   424→          </KanbanColumn>
   425→   425→
   426→   426→          {/* Pipeline columns */}
   427→   427→          {PIPELINE_STAGES.map((stage) => {
   428→   428→            const rows = kanbanColumns[stage];
   429→   429→            return (

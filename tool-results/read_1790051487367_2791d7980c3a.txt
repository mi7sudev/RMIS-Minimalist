    80→// -----------------------------------------------------------------------------
    81→// CompletionRing — compact radial indicator (the record header's only graph).
    82→// Flat primary stroke on a hairline track; the percentage sits centered.
    83→// -----------------------------------------------------------------------------
    84→function CompletionRing({ percent, size = 72 }: { percent: number; size?: number }) {
    85→  const stroke = 6;
    86→  const r = (size - stroke) / 2;
    87→  const c = 2 * Math.PI * r;
    88→  const offset = c - (Math.min(100, Math.max(0, percent)) / 100) * c;
    89→  return (
    90→    <div className="relative shrink-0" style={{ width: size, height: size }}>
    91→      <svg width={size} height={size} className="-rotate-90" aria-hidden>
    92→        <circle
    93→          cx={size / 2}
    94→          cy={size / 2}
    95→          r={r}
    96→          fill="none"
    97→          strokeWidth={stroke}
    98→          className="stroke-border"
    99→        />
   100→        <motion.circle
   101→          cx={size / 2}
   102→          cy={size / 2}
   103→          r={r}
   104→          fill="none"
   105→          strokeWidth={stroke}
   106→          strokeLinecap="round"
   107→          className="stroke-primary"
   108→          strokeDasharray={c}
   109→          initial={{ strokeDashoffset: c }}
   110→          animate={{ strokeDashoffset: offset }}
   111→          transition={{ duration: 0.8, ease: [0.22, 1, 0.36, 1] }}
   112→        />
   113→      </svg>
   114→      <div className="absolute inset-0 grid place-items-center" aria-hidden>
   115→        <span className="text-[17px] font-semibold tabular-nums tracking-[-0.01em] text-foreground">
   116→          {percent}
   117→          <span className="text-[11px] font-medium text-muted-foreground">%</span>
   118→        </span>
   119→      </div>
   120→      <span className="sr-only" role="status">{`Profile ${percent} percent complete`}</span>
   121→    </div>
   122→  );
   123→}

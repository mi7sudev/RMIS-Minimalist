   113→/** Gradient monogram avatar (wave-3) with initials derived from the name. */
   114→function MonogramAvatar({ name, size = 36, warm = false }: { name: string; size?: number; warm?: boolean }) {
   115→  const initials = name
   116→    .split(/\s+/)
   117→    .filter(Boolean)
   118→    .slice(0, 2)
   119→    .map((w) => w[0]?.toUpperCase() ?? "")
   120→    .join("");
   121→  return (
   122→    <span aria-hidden="true">
   123→      <Monogram size={size} warm={warm}>{initials || "?"}</Monogram>
   124→    </span>
   125→  );
   126→}

   128→function s(v: unknown): string {
   129→  if (v === null || v === undefined) return "";
   130→  const t = String(v).trim();
   131→  return t;

   166→function CandidateModal({ id, onClose }: { id: number | null; onClose: () => void }) {
   167→  const [detail, setDetail] = useState<ApplicantDetail | null>(null);
   168→  const [error, setError] = useState<string | null>(null);
   169→
   170→  useEffect(() => {
   171→    if (id === null) {
   172→      setDetail(null);
   173→      setError(null);
   174→      return;
   175→    }
   176→    let alive = true;
   177→    apiFetch<ApplicantDetail>(`/api/admin/applicants/${id}`)
   178→      .then((d) => {
   179→        if (alive) setDetail(d);
   180→      })
   181→      .catch((e: unknown) => {
   182→        if (alive) setError(e instanceof Error ? e.message : "Failed to load the candidate");
   183→      });
   184→    return () => {
   185→      alive = false;
   186→    };
   187→  }, [id]);
   188→
   189→  const apps = detail?.applications ?? [];
   190→  const countBy = (stage: StageKey) => apps.filter((a) => stageForStatus(a.status) === stage).length;
   191→  const anyRejected = apps.some((a) => isRejectedStatus(a.status));
   192→
   193→  return (
   194→    <Dialog open={id !== null} onOpenChange={(v) => !v && onClose()}>
   195→      <DialogContent className="shadow-e4 sm:max-w-lg max-h-[92vh] overflow-y-auto scroll-thin">
   196→        <DialogHeader>
   197→          <DialogTitle className="font-display text-xl">Candidate</DialogTitle>
   198→          <DialogDescription>Quick identity and pipeline snapshot.</DialogDescription>
   199→        </DialogHeader>
   200→
   201→        {error ? (
   202→          <p className="py-4 text-sm text-[var(--bad)]">{error}</p>
   203→        ) : !detail ? (
   204→          <div className="space-y-3">
   205→            <Skeleton className="h-10 w-2/3" />
   206→            <Skeleton className="h-20 w-full rounded-[12px]" />
   207→          </div>
   208→        ) : (
   209→          <div className="space-y-4">
   210→            <div className="flex items-center gap-3">
   211→              <MonogramAvatar name={fullName(detail)} size={40} warm />
   212→              <div className="min-w-0">
   213→                <p className="truncate text-sm font-medium text-ink">{fullName(detail)}</p>
   214→                <p className="num text-xs text-pebble">#{detail.id}</p>
   215→              </div>
   216→              <span className="ml-auto shrink-0">
   217→                <StatusPill
   218→                  status={detail.isProfileComplete ? "Complete" : "Incomplete"}
   219→                  variant={variantForCompletion(detail.isProfileComplete)}
   220→                />
   221→              </span>
   222→            </div>
   223→
   224→            <div className="flex flex-wrap gap-2">
   225→              {detail.emailAddress && (
   226→                <a
   227→                  href={`mailto:${detail.emailAddress}`}
   228→                  className="focus-ring inline-flex min-h-[36px] items-center gap-1.5 rounded-full bg-fog px-3 text-xs text-ink transition-colors hover:bg-[#ececec]"
   229→                >
   230→                  <Mail className="h-3.5 w-3.5" aria-hidden />
   231→                  {detail.emailAddress}
   232→                </a>
   233→              )}
   234→              {detail.mobileNumber && (
   235→                <a
   236→                  href={`tel:${detail.mobileNumber}`}
   237→                  className="focus-ring inline-flex min-h-[36px] items-center gap-1.5 rounded-full bg-fog px-3 text-xs text-ink transition-colors hover:bg-[#ececec]"
   238→                >
   239→                  <Phone className="h-3.5 w-3.5" aria-hidden />
   240→                  {detail.mobileNumber}
   241→                </a>
   242→              )}
   243→            </div>
   244→
   245→            {/* Mini pipeline dots: Submitted → Review → Shortlisted */}
   246→            <div className="rounded-[12px] bg-fog p-4">
   247→              <div className="flex items-center justify-between gap-2">
   248→                {(["Applied", "Under Review", "Shortlisted"] as StageKey[]).map((stage, i) => (
   249→                  <div key={stage} className="flex flex-1 items-center gap-2">
   250→                    <div className="flex items-center gap-1.5">
   251→                      <span
   252→                        className={`stage-dot ${countBy(stage) > 0 ? dotClass(STAGE_DOT[stage]) : "dot-neutral"}`}
   253→                        aria-hidden
   254→                      />
   255→                      <span className="text-xs text-stone">{["Submitted", "Review", "Shortlisted"][i]}</span>
   256→                      <span className="num text-xs text-ink">{countBy(stage)}</span>
   257→                    </div>
   258→                  </div>
   259→                ))}
   260→                {anyRejected && <span className="status-pill status-bad shrink-0">Not Selected</span>}
   261→              </div>
   262→            </div>
   263→
   264→            <div>
   265→              <p className="text-xs font-medium text-stone">Education</p>
   266→              {detail.educations.length === 0 ? (
   267→                <p className="mt-1 text-xs text-pebble">No education entries.</p>
   268→              ) : (
   269→                <ul className="mt-1 space-y-1">
   270→                  {detail.educations.slice(0, 2).map((e, i) => (
   271→                    <li key={i} className="text-sm text-ink">
   272→                      {[s(e.degree) || s(e.course), s(e.schoolName)].filter(Boolean).join(" · ") || "—"}
   273→                    </li>
   274→                  ))}
   275→                </ul>
   276→              )}
   277→            </div>
   278→
   279→            <div className="flex flex-wrap gap-1.5">
   280→              <span className="status-pill status-neutral">
   281→                Documents: <span className="num">{detail.documents.length}</span>
   282→              </span>
   283→              <span className="status-pill status-neutral">
   284→                Applications: <span className="num">{detail.applicationCount ?? apps.length}</span>
   285→              </span>
   286→            </div>
   287→          </div>
   288→        )}
   289→
   290→        <DialogFooter>
   291→          <button type="button" className={ghostBtn} onClick={onClose}>
   292→            Close
   293→          </button>
   294→          {detail && (
   295→            <button type="button" className={ctaBtn} onClick={() => navigate("candidate", { id: String(detail.id) })}>
   296→              View full profile
   297→            </button>
   298→          )}
   299→        </DialogFooter>
   300→      </DialogContent>
   301→    </Dialog>
   302→  );
   303→}

   307→export default function Candidates() {
   308→  const { params } = useHashRoute();
   309→  const initialStatus = params.status === "complete" || params.status === "incomplete" ? params.status : "all";
   310→
   311→  const [rows, setRows] = useState<ApplicantRow[] | null>(null);
   312→  const [total, setTotal] = useState(0);
   313→  const [page, setPage] = useState(1);
   314→  const [error, setError] = useState<string | null>(null);
   315→  const [search, setSearch] = useState("");
   316→  const [debounced, setDebounced] = useState("");
   317→  const [status, setStatus] = useState<"all" | "complete" | "incomplete">(initialStatus as "all" | "complete" | "incomplete");
   318→  const [hasAccount, setHasAccount] = useState<"all" | "yes" | "no">("all");
   319→  const [mode, setMode] = useState<"list" | "kanban">("list");
   320→  const [modalId, setModalId] = useState<number | null>(null);
   321→  const [queue, setQueue] = useState<QueueRow[] | null>(null);
   322→  const [spin, setSpin] = useState(false);
   323→
   324→  const load = useCallback(
   325→    async (silent = false) => {
   326→      if (!silent) setError(null);
   327→      try {
   328→        const sp = new URLSearchParams({ page: String(page), pageSize: String(PAGE_SIZE) });
   329→        if (debounced) sp.set("search", debounced);
   330→        if (status !== "all") sp.set("status", status);
   331→        if (hasAccount !== "all") sp.set("hasAccount", hasAccount);
   332→        const res = await apiFetch<{ data: ApplicantRow[]; total: number }>(`/api/admin/applicants?${sp.toString()}`);
   333→        setRows(res.data);
   334→        setTotal(res.total);
   335→      } catch (e) {
   336→        if (!silent) setError(e instanceof Error ? e.message : "Failed to load candidates");
   337→      }
   338→    },
   339→    [page, debounced, status, hasAccount]
   340→  );
   341→
   342→  const loadQueue = useCallback(async (silent = true) => {
   343→    try {
   344→      const res = await apiFetch<{ data: QueueRow[] }>("/api/evaluator/queue?pageSize=100");
   345→      setQueue(res.data);
   346→    } catch {
   347→      /* silent — kanban is best-effort */
   348→    }
   349→  }, []);
   350→
   351→  useEffect(() => {
   352→    void load();
   353→  }, [load]);
   354→
   355→  useEffect(() => {
   356→    if (mode === "kanban") void loadQueue();
   357→  }, [mode, loadQueue]);
   358→
   359→  // 20 s silent poll + focus refresh.
   360→  useEffect(() => {
   361→    const id = setInterval(() => {
   362→      if (!document.hidden) {
   363→        void load(true);
   364→        if (mode === "kanban") void loadQueue();
   365→      }
   366→    }, 20_000);
   367→    return () => clearInterval(id);
   368→  }, [load, loadQueue, mode]);
   369→  useEffect(() => {
   370→    const onFocus = () => {
   371→      void load(true);
   372→      if (mode === "kanban") void loadQueue();
   373→    };
   374→    window.addEventListener("focus", onFocus);
   375→    document.addEventListener("visibilitychange", onFocus);
   376→    return () => {
   377→      window.removeEventListener("focus", onFocus);
   378→      document.removeEventListener("visibilitychange", onFocus);
   379→    };
   380→  }, [load, loadQueue, mode]);
   381→
   382→  // Debounced search (350 ms).
   383→  useEffect(() => {
   384→    const t = setTimeout(() => {
   385→      setDebounced(search.trim());
   386→      setPage(1);
   387→    }, 350);
   388→    return () => clearTimeout(t);
   389→  }, [search]);
   390→
   391→  const manualRefresh = useCallback(() => {
   392→    setSpin(true);
   393→    void load(true).finally(() => setTimeout(() => setSpin(false), 500));
   394→  }, [load]);
   395→
   396→  const list = rows ?? [];
   397→  const start = total === 0 ? 0 : (page - 1) * PAGE_SIZE + 1;
   398→  const end = Math.min(page * PAGE_SIZE, total);
   399→  const completeOnPage = list.filter((r) => r.isProfileComplete).length;
   400→  const accountsOnPage = list.filter((r) => r.hasAccount).length;
   401→
   402→  const kanban = useMemo(() => {
   403→    const map: Record<StageKey, QueueRow[]> = { "Applied": [], "Under Review": [], "Shortlisted": [], "Rejected": [] };
   404→    for (const r of queue ?? []) map[stageForStatus(r.status)].push(r);
   405→    return map;
   406→  }, [queue]);
   407→
   408→  return (
   409→    <div className="space-y-6">
   410→      <PageHeader
   411→        title="Candidates"
   412→        description="Registry of every candidate on file — profiles, logins, and pipeline activity."
   413→        actions={
   414→          <>
   415→            <div className="flex items-center gap-1 rounded-full bg-fog p-1" role="group" aria-label="View mode">
   416→              <button
   417→                type="button"
   418→                className={`inline-flex min-h-[36px] items-center gap-1.5 rounded-full px-3 text-xs font-medium transition-colors ${mode === "list" ? "bg-ink text-white" : "text-stone hover:text-ink"}`}
   419→                onClick={() => setMode("list")}
   420→                aria-pressed={mode === "list"}
   421→              >
   422→                <List className="h-3.5 w-3.5" aria-hidden />
   423→                List
   424→              </button>
   425→              <button
   426→                type="button"
   427→                className={`inline-flex min-h-[36px] items-center gap-1.5 rounded-full px-3 text-xs font-medium transition-colors ${mode === "kanban" ? "bg-ink text-white" : "text-stone hover:text-ink"}`}
   428→                onClick={() => setMode("kanban")}
   429→                aria-pressed={mode === "kanban"}
   430→              >
   431→                <LayoutGrid className="h-3.5 w-3.5" aria-hidden />
   432→                Kanban
   433→              </button>
   434→            </div>
   435→            <button type="button" className={ghostBtn} onClick={manualRefresh}>
   436→              <RefreshCw className={`h-4 w-4 ${spin ? "animate-spin" : ""}`} aria-hidden />
   437→              Refresh
   438→            </button>
   439→          </>
   440→        }
   441→      />
   442→
   443→      {/* KPI tiles */}
   444→      {!error && (rows === null ? (
   445→        <SkeletonKpis count={4} />
   446→      ) : (
   447→        <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   448→          <KpiCard label="Total on file" value={total} icon={Users} tone="info" hint="all registered candidates" />
   449→          <KpiCard label="Showing" value={`${start}–${end}`} icon={List} tone="neutral" hint={`of ${total}`} />
   450→          <KpiCard label="Complete profiles" value={completeOnPage} icon={UserCheck} tone="ok" hint="on this page" />
   451→          <KpiCard label="Has login" value={accountsOnPage} icon={KeyRound} tone="neutral" hint="on this page" />
   452→        </div>
   453→      ))}
   454→
   455→      {/* Filter bar — one card row */}
   456→      <div className="dlg-card flex flex-wrap items-center gap-3 px-4 py-3">
   457→        <div className="relative min-w-0 flex-1 basis-56">
   458→          <IconChip
   459→            icon={Search}
   460→            tone="slate"
   461→            size={28}
   462→            iconSize={14}
   463→            className="absolute left-2.5 top-1/2 -translate-y-1/2"
   464→          />
   465→          <Input
   466→            className="dlg-input min-h-[44px] pl-12"
   467→            placeholder="Search name, email, employee no., mobile…"
   468→            value={search}
   469→            onChange={(e) => setSearch(e.target.value)}
   470→            aria-label="Search candidates"
   471→          />
   472→        </div>
   473→        <Select
   474→          value={status}
   475→          onValueChange={(v) => {
   476→            setStatus(v as typeof status);
   477→            setPage(1);
   478→          }}
   479→        >
   480→          <SelectTrigger className="dlg-input min-h-[44px] w-40 shrink-0" aria-label="Profile status">
   481→            <SelectValue />
   482→          </SelectTrigger>
   483→          <SelectContent>
   484→            <SelectItem value="all">All profiles</SelectItem>
   485→            <SelectItem value="complete">Complete</SelectItem>
   486→            <SelectItem value="incomplete">Incomplete</SelectItem>
   487→          </SelectContent>
   488→        </Select>
   489→        <Select
   490→          value={hasAccount}
   491→          onValueChange={(v) => {
   492→            setHasAccount(v as typeof hasAccount);
   493→            setPage(1);
   494→          }}
   495→        >
   496→          <SelectTrigger className="dlg-input min-h-[44px] w-40 shrink-0" aria-label="Has account">
   497→            <SelectValue />
   498→          </SelectTrigger>
   499→          <SelectContent>
   500→            <SelectItem value="all">All accounts</SelectItem>
   501→            <SelectItem value="yes">Has login</SelectItem>
   502→            <SelectItem value="no">No login</SelectItem>
   503→          </SelectContent>
   504→        </Select>
   505→      </div>
   506→
   507→      {error ? (
   508→        <div className="dlg-card p-8">
   509→          <EmptyState
   510→            icon={AlertTriangle}
   511→            tone="rose"
   512→            title="Couldn't load candidates"
   513→            description={error}
   514→            action={
   515→              <button type="button" className={ghostBtn} onClick={() => void load()}>
   516→                Retry
   517→              </button>
   518→            }
   519→          />
   520→        </div>
   521→      ) : rows === null ? (
   522→        <SkeletonRows rows={6} rowClassName="h-16" />
   523→      ) : mode === "list" ? (
   524→        <div className="space-y-3">
   525→          {list.length === 0 ? (
   526→            <div className="dlg-card py-6">
   527→              <EmptyState
   528→                icon={Users}
   529→                title="No candidates match"
   530→                description="Try a different search, or clear the profile and account filters."
   531→              />
   532→            </div>

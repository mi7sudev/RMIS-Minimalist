    71→    71→function pageWindow(current: number, total: number): (number | "…")[] {
    72→    72→  if (total <= 7) return Array.from({ length: total }, (_, i) => i + 1);
    73→    73→  const out: (number | "…")[] = [1];
    74→    74→  const lo = Math.max(2, current - 1);
    75→    75→  const hi = Math.min(total - 1, current + 1);
    76→    76→  if (lo > 2) out.push("…");
    77→    77→  for (let i = lo; i <= hi; i++) out.push(i);
    78→    78→  if (hi < total - 1) out.push("…");
    79→    79→  out.push(total);
    80→    80→  return out;
    81→    81→}

    83→    83→/** % of the application window elapsed between publish and deadline (0–100). */
    84→    84→function deadlineProgress(job: JobWire): number {
    85→    85→  const start = new Date(job.publishDate ?? job.publishedAt).getTime();
    86→    86→  const end = job.deadlineDate ? new Date(job.deadlineDate).getTime() : Number.NaN;
    87→    87→  if (Number.isNaN(start) || Number.isNaN(end) || end <= start) return job.deadlineDate ? 100 : 0;
    88→    88→  const pct = ((Date.now() - start) / (end - start)) * 100;
    89→    89→  return Math.min(100, Math.max(0, pct));
    90→    90→}

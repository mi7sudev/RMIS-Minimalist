   122→// Heuristic: does this cell value look like a field label?
   123→// Labels are short, contain letters, don't contain digits, and often end with ":".
   124→function looksLikeLabel(s: string): boolean {
   125→  if (!s) return false;
   126→  const trimmed = s.replace(/:$/, "").trim();
   127→  if (trimmed.length === 0 || trimmed.length > 40) return false;
   128→  if (!/[a-zA-Z]/.test(trimmed)) return false; // must contain at least one letter
   129→  if (/\d/.test(trimmed)) return false; // labels don't contain digits
   130→  return true;
   131→}

   183→// Parse an Excel file (.xlsx/.xls) into a readable text representation.
   184→// The PDS (CSC Form 212) is a structured Excel form — we read all sheets,
   185→// extract cell values with their row/column context, and format as readable text
   186→// so the LLM can understand the form structure and extract structured data.
   187→//
   188→// Key transformations (each fixes a real extraction bug observed on the
   189→// CSC Form 212 PDS Excel template):
   190→//   1. Skip "Lookup" sheets — these are dropdown value sources (Yes/No,
   191→//      Civil Status, Gender, Country list) that pollute the text.
   192→//   2. Filter out cells whose value is a pure dropdown-list entry (country
   193→//      names, Yes/No, civil-status values). These appear as vertical lists in
   194→//      far-right columns and were pushing real education/work/training rows
   195→//      past the truncation limit, causing the LLM to return empty arrays.
   196→//   3. Deduplicate adjacent identical cells within a row — merged-cell ranges
   197→//      cause exceljs to return the same value in every cell of the merge, so
   198→//      a single "PERSONAL DATA SHEET" label would appear 14 times per row.
   199→//   4. Keep the "Label: Value" pairing only for genuine label cells (short,
   200→//      alphabetic, no digits) so tabular data (education/work tables) is
   201→//      preserved as pipe-separated values the LLM can parse column-by-column.
   202→//
   203→// Uses exceljs (maintained, no known advisories) instead of the vulnerable xlsx package.
   204→async function parseExcelToText(filePath: string): Promise<string> {
   205→  const abs = path.isAbsolute(filePath) ? filePath : path.join(process.cwd(), filePath);
   206→  const buffer = await fs.readFile(abs);
   207→
   208→  const workbook = new ExcelJS.Workbook();
   209→  // Cast to the exact declared parameter type: Buffer typings differ between
   210→  // runtimes (@types/node vs bun) but the runtime value is the same object.
   211→  await workbook.xlsx.load(buffer as unknown as Parameters<typeof workbook.xlsx.load>[0]);
   212→
   213→  const sections: string[] = [];
   214→
   215→  for (const sheet of workbook.worksheets) {
   216→    // Skip Lookup sheets — they only contain dropdown source values
   217→    // (Yes/No, Civil Status, Gender, Country list) that pollute extraction.
   218→    if (/^lookup$/i.test(sheet.name)) continue;
   219→    if (sheet.rowCount === 0) continue;
   220→
   221→    sections.push(`\n=== Sheet: ${sheet.name} ===`);
   222→
   223→    // Get the actual used range dimensions
   224→    const rowCount = sheet.actualRowCount;
   225→
   226→    for (let r = 1; r <= rowCount; r++) {
   227→      const row = sheet.getRow(r);
   228→      const rawCells: string[] = [];
   229→      let hasContent = false;
   230→
   231→      // Collect non-empty cell values across the row.
   232→      // NOTE: ExcelJS's `cell.text` getter throws "Cannot read properties of
   233→      // null (reading 'toString')" when cell.value is explicitly null (common
   234→      // in merged-cell regions of PDS / CSC Form 212 spreadsheets). We wrap
   235→      // it in try/catch and fall back to checking cell.value directly.
   236→      for (let c = 1; c <= row.cellCount; c++) {
   237→        const cell = row.getCell(c);
   238→        let value = "";
   239→        try {
   240→          value = cell.text || (typeof cell.value === "string" ? cell.value : "");
   241→        } catch {
   242→          // cell.text getter threw (null value in merged cell, etc.)
   243→          value = typeof cell.value === "string" ? cell.value : "";
   244→        }
   245→        const trimmed = String(value || "").trim();
   246→        rawCells.push(trimmed);
   247→        if (trimmed) hasContent = true;
   248→      }
   249→
   250→      if (!hasContent) continue;
   251→
   252→      // Filter out dropdown-noise cells (country names, Yes/No, civil-status
   253→      // values). These are embedded vertical lists in far-right columns that
   254→      // have nothing to do with the applicant's actual data.
   255→      let cells = rawCells.filter((c) => c && !isDropdownNoise(c));
   256→
   257→      // Skip rows that are now empty (pure dropdown-list rows like
   258→      // "Afghanistan", "Albania", ...).
   259→      if (cells.length === 0) continue;
   260→
   261→      // Deduplicate adjacent identical cells — merged-cell ranges cause
   262→      // exceljs to return the same value in every cell of the merge (e.g.
   263→      // "PERSONAL DATA SHEET" repeated 14 times across one row).
   264→      const deduped: string[] = [];
   265→      for (const c of cells) {
   266→        if (deduped[deduped.length - 1] !== c) deduped.push(c);
   267→      }
   268→      cells = deduped;
   269→
   270→      // Format the row.
   271→      // Only pair cells as "Label: Value" when the first cell genuinely looks
   272→      // like a label (short, alphabetic, no digits) and the next cell looks like a value.
   273→      // This prevents mangling tabular data (e.g. education tables) where every cell is a value.
   274→      const parts: string[] = [];
   275→      for (let c = 0; c < cells.length; c++) {
   276→        const cell = cells[c];
   277→        if (!cell) continue;
   278→        const next = cells[c + 1];
   279→
   280→        if (next && looksLikeLabel(cell) && !looksLikeLabel(next)) {
   281→          // Label: Value pair
   282→          parts.push(`${cell.replace(/:$/, "")}: ${next}`);
   283→          c++; // skip the value cell
   284→        } else {
   285→          // Standalone cell — add as-is
   286→          parts.push(cell);
   287→        }
   288→      }
   289→      if (parts.length > 0) {
   290→        sections.push(parts.join(" | "));
   291→      }
   292→    }
   293→  }
   294→
   295→  const text = sections.join("\n");
   296→  // Truncate to avoid exceeding LLM context limits (keep first ~16000 chars).
   297→  // Increased from 12000 → 16000 because the dedup + dropdown-noise filtering
   298→  // above significantly reduces text size, leaving room for the full PDS
   299→  // (4 sheets) without cutting off the education/work/training sections.
   300→  const MAX_CHARS = 16000;
   301→  if (text.length > MAX_CHARS) {
   302→    return text.slice(0, MAX_CHARS) + "\n\n[... document truncated ...]";
   303→  }
   304→  return text;
   305→}

   307→// Parse a PDF file into per-page text for the text-based LLM extraction path.
   308→// The VLM API rejects PDF file_url data URLs ("URL格式无效"), so we extract text
   309→// via unpdf and send it to the chat completions endpoint — same as Excel.
   310→//
   311→// Returns one string PER PAGE (mergePages: false) so the PDS extractor can
   312→// chunk multi-page documents instead of truncating them — applicants with
   313→// 100+ trainings/awards routinely submit 6-10 page PDS PDFs, and the old
   314→// single-string 12K truncation silently dropped every overflow page.
   315→async function parsePdfPages(filePath: string): Promise<string[]> {
   316→  const abs = path.isAbsolute(filePath) ? filePath : path.join(process.cwd(), filePath);
   317→  const buffer = await fs.readFile(abs);
   318→  // unpdf is a worker-free PDF text extractor built for bundled/server
   319→  // environments (pdfjs-dist's dynamic worker import fails under Turbopack).
   320→  const { text } = await unpdfExtractText(new Uint8Array(buffer), { mergePages: false });
   321→  const pages = Array.isArray(text) ? text : [String(text ?? "")];
   322→  return pages.map((p) => (p || "").trim());
   323→}

   325→// Legacy single-string variant — still used by non-PDS categories where the
   326→// document is small and a truncation guard is safe.
   327→async function parsePdfToText(filePath: string): Promise<string> {
   328→  const pages = await parsePdfPages(filePath);
   329→  const out = pages.join("\n\n").trim();
   330→  const MAX_CHARS = 12000;
   331→  if (out.length > MAX_CHARS) {
   332→    return out.slice(0, MAX_CHARS) + "\n\n[... document truncated ...]";
   333→  }
   334→  return out;
   335→}

   337→// Parse a Word (.doc/.docx) file into readable text for the text-based LLM.
   338→async function parseWordToText(filePath: string): Promise<string> {
   339→  const abs = path.isAbsolute(filePath) ? filePath : path.join(process.cwd(), filePath);
   340→  const buffer = await fs.readFile(abs);
   341→  const result = await mammoth.extractRawText({ buffer });
   342→  const text = (result?.value || "").trim();
   343→  const MAX_CHARS = 12000;
   344→  if (text.length > MAX_CHARS) {
   345→    return text.slice(0, MAX_CHARS) + "\n\n[... document truncated ...]";
   346→  }
   347→  return text;
   348→}

   352→async function readAsDataUrl(filePath: string): Promise<{ url: string; kind: ReturnType<typeof detectFileKind> }> {
   353→  const abs = path.isAbsolute(filePath) ? filePath : path.join(process.cwd(), filePath);
   354→  const buffer = await fs.readFile(abs);
   355→  const ext = path.extname(abs).toLowerCase();
   356→  const kind = detectFileKind(filePath);
   357→
   358→  let mime: string;
   359→  switch (kind) {
   360→    case "image":
   361→      mime = `image/${ext.replace(".", "") === "jpg" ? "jpeg" : ext.replace(".", "")}`;
   362→      break;
   363→    case "pdf":
   364→      mime = "application/pdf";
   365→      break;
   366→    case "word":
   367→      mime = ext === ".doc"
   368→        ? "application/msword"
   369→        : "application/vnd.openxmlformats-officedocument.wordprocessingml.document";
   370→      break;
   371→    default:
   372→      mime = "application/octet-stream";
   373→  }
   374→  return { url: `data:${mime};base64,${buffer.toString("base64")}`, kind };
   375→}

   483→function parseJsonResponse(content: string): unknown {
   484→  if (!content) throw new Error("Empty response from LLM/VLM");
   485→  // strip markdown fences if present
   486→  let cleaned = content.trim();
   487→  const fenceMatch = cleaned.match(/```(?:json)?\s*([\s\S]*?)```/);
   488→  if (fenceMatch) {
   489→    cleaned = fenceMatch[1].trim();
   490→  }
   491→  // find first { and last }
   492→  const first = cleaned.indexOf("{");
   493→  const last = cleaned.lastIndexOf("}");
   494→  if (first !== -1 && last !== -1 && last > first) {
   495→    cleaned = cleaned.slice(first, last + 1);
   496→  }
   497→  return JSON.parse(cleaned);
   498→}

   500→// ---- Parallel per-section PDS extraction ----
   501→//
   502→// The PDS (CSC Form 212) is a large 4-sheet Excel document. A single LLM call
   503→// to extract ALL sections (personal info + education + work + training +
   504→// eligibility + awards) takes 60+ seconds because the model must generate a
   505→// very large JSON output. This exceeds gateway/browser HTTP timeouts (typically
   506→// 30–60s), causing the client to receive a 504/timeout error even though the
   507→// server eventually completes successfully.
   508→//
   509→// FIX: Split the extraction into 4 focused parallel LLM calls, each asking for
   510→// only a subset of the data. Each call generates a much smaller JSON output, so
   511→// each completes in ~10–15s. Running them in parallel means the total wall-clock
   512→// time is the slowest call (~15s) instead of the sum (~60s) — comfortably under
   513→// any gateway timeout.
   514→//
   515→// The 4 groups are chosen to balance output size and keep related sections
   516→// together (e.g. work experience and eligibility are both on Sheet C2):
   517→//   1. Personal info  (Sheet C1)
   518→//   2. Educations     (Sheet C1)
   519→//   3. Work + Eligibility (Sheet C2)
   520→//   4. Training + Awards  (Sheet C3)
